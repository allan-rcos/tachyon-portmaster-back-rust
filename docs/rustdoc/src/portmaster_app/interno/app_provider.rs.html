25/// A implementação do provider. Privada: nenhum crate exporta impl.
26pub(crate) struct AppProviderImpl<D, I> {
27    /// O provider do `domain`, de onde saem os `TableModules`.
28    domain: D,
29    /// O provider da `infra`, de onde saem repositories, cache e leitura.
30    infra: I,
31}
32
33impl<D, I> AppProviderImpl<D, I> {
34    /// Guarda os subproviders das camadas de baixo.
35    ///
36    /// Não há recurso caro aqui: o pool e os caches nasceram no `register` da
37    /// `infra` e chegam dentro do `I`.
38    pub(crate) const fn new(domain: D, infra: I) -> Self {
39        Self { domain, infra }
40    }
41}
42
