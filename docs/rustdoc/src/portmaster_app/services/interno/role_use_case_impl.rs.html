24/// A implementação, genérica sobre os ports que consome.
25#[derive(Clone)]
26pub(crate) struct RoleUseCaseImpl<R, T, Q, F, C, U> {
27    /// Persistência de papéis.
28    roles: R,
29    /// As regras de papel.
30    role_tm: T,
31    /// Quem executa um DQL contra o banco.
32    queries: Q,
33    /// De onde os DQLs saem, já com os parâmetros.
34    dqls: F,
35    /// O cache de leitura, para o read-through e a invalidação.
36    cache: C,
37    /// Quem abre e fecha a transação.
38    unit_of_work: U,
39    /// A permissão exigida para create.
40    create_permission: RequiresPermission,
41    /// A permissão exigida para update.
42    update_permission: RequiresPermission,
43    /// A permissão exigida para read.
44    read_permission: RequiresPermission,
45}
46
47impl<R, T, Q, F, C, U> RoleUseCaseImpl<R, T, Q, F, C, U> {
48    /// Monta o caso de uso, declarando as permissões que ele exige.
49    pub(crate) const fn new(
50        roles: R,
51        role_tm: T,
52        queries: Q,
53        dqls: F,
54        cache: C,
55        unit_of_work: U,
56    ) -> Self {
57        Self {
58            roles,
59            role_tm,
60            queries,
61            dqls,
62            cache,
63            unit_of_work,
64            create_permission: RequiresPermission::new(PermissionSlug::ROLE_CREATE),
65            update_permission: RequiresPermission::new(PermissionSlug::ROLE_UPDATE_PERMISSIONS),
66            read_permission: RequiresPermission::new(PermissionSlug::ROLE_LIST),
67        }
68    }
69}
70
71impl<R, T, Q, F, C, U> RoleUseCase for RoleUseCaseImpl<R, T, Q, F, C, U>
72where
73    R: RoleRepository + Send + Sync,
74    T: RoleTM + Send + Sync,
75    Q: QueryRepository + Send + Sync,
76    F: QueryFactory + Send + Sync,
77    C: ReadCache + Send + Sync,
78    U: UnitOfWork + Send + Sync,
79{
80    async fn create(&self, command: CreateRoleCommand) -> Result<Box<dyn Role>, AppError> {
81        self.create_permission.authorize(&command.context)?;
82
83        let role = Transaction::run(&self.unit_of_work, async {
84            let role = self.role_tm.create(command.name, command.permissions)?;
85
86            self.roles.insert(role.as_ref()).await?;
87
88            Ok(role)
89        })
90        .await?;
91
92        ReadThrough::invalidate(&self.cache, Invalidation::ROLE_WRITE).await?;
93
94        Ok(role)
95    }
96
97    /// Substitui as permissões de um papel.
98    ///
99    /// A invalidação alcança `user:` e `account:` também: trocar as permissões
100    /// de um papel muda o que toda conta que o carrega pode fazer.
101    async fn update_permissions(
102        &self,
103        command: UpdateRolePermissionsCommand,
104    ) -> Result<Box<dyn Role>, AppError> {
105        self.update_permission.authorize(&command.context)?;
106
107        let role = Transaction::run(&self.unit_of_work, async {
108            let existing = self
109                .roles
110                .find_by_id(&command.id)
111                .await?
112                .ok_or_else(|| AppError::missing("papel", &command.id))?;
113
114            let updated = self
115                .role_tm
116                .update_permissions(existing.as_ref(), command.permissions)?;
117
118            self.roles.update(updated.as_ref()).await?;
119
120            Ok(updated)
121        })
122        .await?;
123
124        ReadThrough::invalidate(&self.cache, Invalidation::ROLE_WRITE).await?;
125
126        Ok(role)
127    }
128
129    async fn get(&self, query: GetRoleQuery) -> Result<RoleViewItem, AppError> {
130        self.read_permission.authorize(&query.context)?;
131
132        let key = CacheKey::of(CacheKey::ROLE, "get", &[&query.id]);
133
134        ReadThrough::cached(&self.cache, &key, async {
135            let dql = self.dqls.get_role(&query.id)?;
136
137            Transaction::run(&self.unit_of_work, async {
138                self.queries
139                    .run(dql)
140                    .await?
141                    .ok_or_else(|| AppError::missing("papel", &query.id))
142            })
143            .await
144        })
145        .await
146    }
147
148    async fn list(&self, query: ListRolesQuery) -> Result<RoleListView, AppError> {
149        self.read_permission.authorize(&query.context)?;
150
151        let key = CacheKey::of(
152            CacheKey::ROLE,
153            "list",
154            &[
155                &query.limit.unwrap_or_default().to_string(),
156                query.cursor.as_deref().unwrap_or_default(),
157                query.search.as_deref().unwrap_or_default(),
158            ],
159        );
160
161        ReadThrough::cached(&self.cache, &key, async {
162            let dql = self.dqls.list_roles(ListParams {
163                cursor: query.cursor.clone(),
164                limit: query.limit,
165                search: query.search.clone(),
166            });
167
168            Transaction::run(&self.unit_of_work, async {
169                Ok(self.queries.run(dql).await?)
170            })
171            .await
172        })
173        .await
174    }
