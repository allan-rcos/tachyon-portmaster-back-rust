12/// O que a apresentação pode pedir sobre papéis.
13#[trait_variant::make(Send)]
14pub trait RoleService {
15    /// Registra, no boot, as permissões que este serviço exige.
16    ///
17    /// Os slugs são `const` privadas da implementação e **não** saem dela: quem
18    /// os compara com o `UserContext` é o próprio caso de uso, e não há segundo
19    /// lugar no sistema que precise vê-los. O que atravessa esta fronteira é a
20    /// ação de registrar, nunca a lista — é o molde do `declarePermission` do
21    /// PHP, onde a permissão pertence a exatamente um caso de uso.
22    async fn declare_permissions<M: MetadataService + Send + Sync>(
23        &self,
24        registrar: &M,
25    ) -> Result<(), RoleError>;
26
27    /// Cria e devolve o papel.
28    async fn create(&self, command: CreateRoleCommand) -> Result<Box<dyn Role>, RoleError>;
29
30    /// Substitui as permissões e devolve o papel atualizado.
31    async fn update_permissions(
32        &self,
33        command: UpdateRolePermissionsCommand,
34    ) -> Result<Box<dyn Role>, RoleError>;
35
36    /// Lê um papel.
37    async fn get(&self, query: GetRoleQuery) -> Result<RoleViewItem, RoleError>;
38
39    /// Lista papéis.
40    async fn list(&self, query: ListRolesQuery) -> Result<RoleListView, RoleError>;
41}
