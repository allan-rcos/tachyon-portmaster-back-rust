17/// O prefixo de toda leitura deste serviço.
18const CACHE_GROUP: &str = "account";
19
20/// A chave do perfil de quem está na sessão.
21/// A implementação, genérica sobre os ports que consome.
22#[derive(Clone)]
23pub(crate) struct AccountUseCaseImpl<R, T, A, Q, C> {
24    /// Persistência de usuários.
25    users: R,
26    /// As regras de usuário — quem constrói e valida.
27    user_tm: T,
28    /// As regras de credencial.
29    auth_tm: A,
30    /// Quem executa um DQL contra o banco.
31    queries: Q,
32    /// O cache do lado de leitura.
33    views: C,
34}
35
36impl<R, T, A, Q, C> AccountUseCaseImpl<R, T, A, Q, C> {
37    /// Monta o caso de uso.
38    pub(crate) const fn new(users: R, user_tm: T, auth_tm: A, queries: Q, views: C) -> Self {
39        Self {
40            users,
41            user_tm,
42            auth_tm,
43            queries,
44            views,
45        }
46    }
47}
48
49impl<R, T, A, Q, C> AccountUseCase for AccountUseCaseImpl<R, T, A, Q, C>
50where
51    R: UserRepository + Send + Sync,
52    T: UserTM + Send + Sync,
53    A: AuthTM + Send + Sync,
54    Q: QueryRepository + Send + Sync,
55    C: ViewCacheRepository + Send + Sync,
56{
57    /// O perfil de quem está na sessão.
58    ///
59    /// Não confere permissão: a conta é do próprio usuário, e a sessão já provou
60    /// quem ele é. Responde `InvalidCredentials` se ela não existe mais — a
61    /// conta some entre a emissão do token e o pedido quando o usuário é
62    /// removido, e o token continua assinado sem descrever ninguém.
63    async fn get(&self, query: GetAccountQuery) -> Result<AccountView, AccountError> {
64        let dql = dql::get_account(&query.context.id)?;
65        let key = dql.cache_key();
66
67        if let Some(hit) = self.views.get(CACHE_GROUP, &key).await? {
68            return Ok(hit);
69        }
70
71        let view = MasterScope::run(|uow| async move {
72            let Some(view) = self.queries.run(dql).await? else {
73                return Err(AccountError::InvalidCredentials);
74            };
75
76            uow.commit().await?;
77
78            Ok(view)
79        })
80        .await?;
81
82        // Falhar ao guardar não invalida a resposta: o cliente já tem o
83        // dado correto, e o único prejuízo é o próximo pedido recalcular.
84        self.views.put(CACHE_GROUP, &key, &view).await?;
85
86        Ok(view)
87    }
88
89    async fn update(&self, command: UpdateAccountCommand) -> Result<Box<dyn User>, AccountError> {
90        let user = MasterScope::run(|uow| async move {
91            let Some(existing) = self.users.find_by_id(&command.context.id).await? else {
92                return Err(AccountError::InvalidCredentials);
93            };
94
95            let updated = self
96                .user_tm
97                .update(existing.as_ref(), command.name, command.email)?;
98
99            self.users.update(updated.as_ref()).await?;
100
101            uow.commit().await?;
102
103            Ok(updated)
104        })
105        .await?;
106
107        self.views.invalidate(CACHE_GROUP).await?;
108
109        Ok(user)
110    }
111
112    /// Troca a senha do próprio usuário.
113    ///
114    /// A senha **atual** é exigida mesmo com a sessão válida: um token roubado
115    /// não deve bastar para trocar a senha e expulsar o dono.
116    async fn change_password(&self, command: ChangePasswordCommand) -> Result<(), AccountError> {
117        MasterScope::run(|uow| async move {
118            let Some(existing) = self.users.find_by_id(&command.context.id).await? else {
119                return Err(AccountError::InvalidCredentials);
120            };
121
122            self.auth_tm
123                .login(existing.as_ref(), &command.current_password)?;
124
125            let updated = self
126                .user_tm
127                .change_password(existing.as_ref(), command.new_password)?;
128
129            self.users.update(updated.as_ref()).await?;
130
131            uow.commit().await?;
132
133            Ok(())
134        })
135        .await?;
136
137        self.views.invalidate(CACHE_GROUP).await?;
138
139        Ok(())
140    }
