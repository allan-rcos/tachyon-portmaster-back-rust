1//! A orquestração de usuários.
2//!
3//! ## As permissões são privadas
4//!
5//! Os slugs abaixo são **contrato**: já existem em papéis gravados no banco de
6//! quem roda a versão PHP, e renomear qualquer um revoga silenciosamente o
7//! acesso de quem o tinha. São `const` privadas porque uma permissão pertence a
8//! exatamente um caso de uso — é ele quem a compara com o `UserContext`, e não
9//! há segundo lugar no sistema que precise vê-la. O boot as registra chamando
10//! `declare_permissions`, sem nunca lê-las.

50/// A implementação, genérica sobre os ports que consome.
51#[derive(Clone)]
52pub(crate) struct UserUseCaseImpl<UR, RR, T, Q, C> {
53    /// Persistência de usuários.
54    users: UR,
55    /// Persistência de papéis.
56    roles: RR,
57    /// As regras de usuário — quem constrói e valida.
58    user_tm: T,
59    /// Quem executa um DQL contra o banco.
60    queries: Q,
61    /// O cache do lado de leitura.
62    views: C,
63}

66    /// Monta o caso de uso.

78impl<UR, RR, T, Q, C> UserUseCaseImpl<UR, RR, T, Q, C>
79where
80    RR: RoleRepository + Send + Sync,
81{
82    /// Carrega os papéis pedidos, recusando um id que não existe.
83    ///
84    /// O PHP sincronizava os ids direto e deixava a chave estrangeira reclamar —
85    /// o que devolvia erro de banco para o que é, na verdade, um id errado no
86    /// corpo do request. Carregar antes transforma isso num 404 com o id que
87    /// falhou.
88    async fn resolve_roles(&self, ids: &[String]) -> Result<Vec<Box<dyn Role>>, UserError> {
89        let mut roles = Vec::with_capacity(ids.len());
90
91        for id in ids {
92            let role = self
93                .roles
94                .find_by_id(id)
95                .await?
96                .ok_or_else(|| UserError::MissingRole(id.clone()))?;
97
98            roles.push(role);
99        }
100
101        Ok(roles)
102    }
103}

105impl<UR, RR, T, Q, C> UserUseCase for UserUseCaseImpl<UR, RR, T, Q, C>
106where
107    UR: UserRepository + Send + Sync,
108    RR: RoleRepository + Send + Sync,
109    T: UserTM + Send + Sync,
110    Q: QueryRepository + Send + Sync,
111    C: ViewCacheRepository + Send + Sync,
112{
113    async fn declare_permissions<M: MetadataUseCase + Send + Sync>(
114        &self,
115        registrar: &M,
116    ) -> Result<(), UserError> {
117        for slug in [
118            CHANGE_PASSWORD,
119            CREATE,
120            DELETE,
121            GET,
122            LIST,
123            UPDATE,
124            UPDATE_ROLES,
125        ] {
126            registrar
127                .register_permission(RegisterPermissionCommand {
128                    slug: slug.to_owned(),
129                })
130                .await?;
131        }
132
133        Ok(())
134    }
135
136    /// Cria um usuário e liga os papéis dele.
137    ///
138    /// O e-mail é único, e a duplicidade é descoberta **aqui** em vez de deixar
139    /// o índice reclamar: assim o cliente recebe um conflito com sentido, e não
140    /// um erro de banco.
141    ///
142    /// Os papéis são uma tabela de ligação à parte, então `insert` grava a linha
143    /// de `users` e o vínculo precisa do seu próprio comando.
144    async fn create(&self, command: CreateUserCommand) -> Result<Box<dyn User>, UserError> {
145        if !command.context.has_permission(CREATE) {
146            return Err(AppError::permission_denied(CREATE).into());
147        }
148
149        let user = MasterScope::run(|uow| async move {
150            if self.users.find_by_email(&command.email).await?.is_some() {
151                return Err(UserError::EmailTaken);
152            }
153
154            let roles = self.resolve_roles(&command.role_ids).await?;
155            let role_ids: Vec<String> = roles.iter().map(|role| role.id().to_owned()).collect();
156
157            let user = self.user_tm.create(
158                command.name,
159                command.email,
160                command.initial_password,
161                roles,
162            )?;
163
164            self.users.insert(user.as_ref()).await?;
165
166            self.users.sync_roles(user.id(), &role_ids).await?;
167
168            uow.commit().await?;
169
170            Ok(user)
171        })
172        .await?;
173
174        self.views.invalidate(CACHE_GROUP).await?;
175
176        Ok(user)
177    }
178
179    async fn update(&self, command: UpdateUserCommand) -> Result<Box<dyn User>, UserError> {
180        if !command.context.has_permission(UPDATE) {
181            return Err(AppError::permission_denied(UPDATE).into());
182        }
183
184        let user = MasterScope::run(|uow| async move {
185            let Some(existing) = self.users.find_by_id(&command.id).await? else {
186                return Err(UserError::Missing(command.id));
187            };
188
189            let updated = self
190                .user_tm
191                .update(existing.as_ref(), command.name, command.email)?;
192
193            self.users.update(updated.as_ref()).await?;
194
195            uow.commit().await?;
196
197            Ok(updated)
198        })
199        .await?;
200
201        self.views.invalidate(CACHE_GROUP).await?;
202
203        Ok(user)
204    }
205
206    async fn update_roles(
207        &self,
208        command: UpdateUserRolesCommand,
209    ) -> Result<Box<dyn User>, UserError> {
210        if !command.context.has_permission(UPDATE_ROLES) {
211            return Err(AppError::permission_denied(UPDATE_ROLES).into());
212        }
213
214        let user = MasterScope::run(|uow| async move {
215            let Some(existing) = self.users.find_by_id(&command.id).await? else {
216                return Err(UserError::Missing(command.id));
217            };
218
219            let roles = self.resolve_roles(&command.role_ids).await?;
220            let role_ids: Vec<String> = roles.iter().map(|role| role.id().to_owned()).collect();
221
222            let updated = self.user_tm.update_roles(existing.as_ref(), roles)?;
223
224            self.users.sync_roles(&command.id, &role_ids).await?;
225
226            uow.commit().await?;
227
228            Ok(updated)
229        })
230        .await?;
231
232        self.views.invalidate(CACHE_GROUP).await?;
233
234        Ok(user)
235    }

237    /// Redefine a senha de outro usuário.
238    ///
239    /// **Sem pedir a senha atual**: quem redefine a senha de outro não a
240    /// conhece. É a permissão que autoriza, não o conhecimento do segredo — ao
241    /// contrário de `AccountUseCase::change_password`, que rege a própria conta.
242    async fn reset_password(&self, command: ResetUserPasswordCommand) -> Result<(), UserError> {
243        if !command.context.has_permission(CHANGE_PASSWORD) {
244            return Err(AppError::permission_denied(CHANGE_PASSWORD).into());
245        }
246
247        MasterScope::run(|uow| async move {
248            let Some(existing) = self.users.find_by_id(&command.id).await? else {
249                return Err(UserError::Missing(command.id));
250            };
251
252            let updated = self
253                .user_tm
254                .change_password(existing.as_ref(), command.new_password)?;
255
256            self.users.update(updated.as_ref()).await?;
257
258            uow.commit().await?;
259
260            Ok(())
261        })
262        .await?;
263
264        self.views.invalidate(CACHE_GROUP).await?;
265
266        Ok(())
267    }

269    async fn delete(&self, command: DeleteUserCommand) -> Result<(), UserError> {
270        if !command.context.has_permission(DELETE) {
271            return Err(AppError::permission_denied(DELETE).into());
272        }
273
274        MasterScope::run(|uow| async move {
275            if self.users.find_by_id(&command.id).await?.is_none() {
276                return Err(UserError::Missing(command.id));
277            }
278
279            self.users.delete(&command.id).await?;
280
281            uow.commit().await?;
282
283            Ok(())
284        })
285        .await?;
286
287        self.views.invalidate(CACHE_GROUP).await?;
288
289        Ok(())
290    }

292    /// Um usuário com os papéis dele, pelo lado de leitura.
293    ///
294    /// Usa a mesma consulta de `GET /account`: um usuário com os papéis dele é o
295    /// mesmo recorte, seja o próprio ou outro. A chave é outra — `user:` e não
296    /// `account:` — porque é este serviço que a derruba.
297    async fn get(&self, query: GetUserQuery) -> Result<AccountView, UserError> {
298        if !query.context.has_permission(GET) {
299            return Err(AppError::permission_denied(GET).into());
300        }
301
302        let dql = dql::get_account(&query.id)?;
303        let key = dql.cache_key();
304
305        if let Some(hit) = self.views.get(CACHE_GROUP, &key).await? {
306            return Ok(hit);
307        }
308
309        let missing = query.id.clone();
310
311        let view = MasterScope::run(|uow| async move {
312            let Some(view) = self.queries.run(dql).await? else {
313                return Err(UserError::Missing(missing));
314            };
315
316            uow.commit().await?;
317
318            Ok(view)
319        })
320        .await?;
321
322        // Falhar ao guardar não invalida a resposta: o cliente já tem o
323        // dado correto, e o único prejuízo é o próximo pedido recalcular.
324        self.views.put(CACHE_GROUP, &key, &view).await?;
325
326        Ok(view)
327    }
328
329    async fn list(&self, query: ListUsersQuery) -> Result<UserListView, UserError> {
330        if !query.context.has_permission(LIST) {
331            return Err(AppError::permission_denied(LIST).into());
332        }
333
334        let dql = dql::list_users(query.page, query.limit);
335        let key = dql.cache_key();
336
337        if let Some(hit) = self.views.get(CACHE_GROUP, &key).await? {
338            return Ok(hit);
339        }
340
341        let view = MasterScope::run(|uow| async move {
342            let view = self.queries.run(dql).await?;
343
344            uow.commit().await?;
345
346            Ok::<_, UserError>(view)
347        })
348        .await?;
349
350        // Falhar ao guardar não invalida a resposta: o cliente já tem o
351        // dado correto, e o único prejuízo é o próximo pedido recalcular.
352        self.views.put(CACHE_GROUP, &key, &view).await?;
353
354        Ok(view)
355    }
