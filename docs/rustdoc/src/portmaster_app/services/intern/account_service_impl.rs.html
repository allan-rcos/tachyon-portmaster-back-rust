17/// O prefixo de toda leitura deste serviço.
18const CACHE_GROUP: &str = "account";
19
20/// Monta o caso de uso de conta.
21///
22/// Os ports chegam injetados e o que sai é o contrato: o tipo concreto não tem
23/// nome fora deste arquivo, então nada além do provider consegue depender do
24/// formato dele.
25pub(crate) fn account_service<R, T, A, Q, C>(
26    users: R,
27    user_tm: T,
28    auth_tm: A,
29    queries: Q,
30    views: C,
31) -> impl AccountService + Sync + Clone + use<R, T, A, Q, C> + 'static
32where
33    R: UserRepository + Send + Sync + Clone + 'static,
34    T: UserTM + Send + Sync + Clone + 'static,
35    A: AuthTM + Send + Sync + Clone + 'static,
36    Q: QueryRepository + Send + Sync + Clone + 'static,
37    C: ViewCacheRepository + Send + Sync + Clone + 'static,
38{
39    AccountServiceImpl {
40        users,
41        user_tm,
42        auth_tm,
43        queries,
44        views,
45    }
46}

48/// A chave do perfil de quem está na sessão.
49/// A implementação, genérica sobre os ports que consome.
50#[derive(Clone)]
51struct AccountServiceImpl<R, T, A, Q, C> {
52    /// Persistência de usuários.
53    users: R,
54    /// As regras de usuário — quem constrói e valida.
55    user_tm: T,
56    /// As regras de credencial.
57    auth_tm: A,
58    /// Quem executa um DQL contra o banco.
59    queries: Q,
60    /// O cache do lado de leitura.
61    views: C,
62}
63
64impl<R, T, A, Q, C> AccountService for AccountServiceImpl<R, T, A, Q, C>
65where
66    R: UserRepository + Send + Sync,
67    T: UserTM + Send + Sync,
68    A: AuthTM + Send + Sync,
69    Q: QueryRepository + Send + Sync,
70    C: ViewCacheRepository + Send + Sync,
71{
72    /// O perfil de quem está na sessão.
73    ///
74    /// Não confere permissão: a conta é do próprio usuário, e a sessão já provou
75    /// quem ele é. Responde `InvalidCredentials` se ela não existe mais — a
76    /// conta some entre a emissão do token e o pedido quando o usuário é
77    /// removido, e o token continua assinado sem descrever ninguém.
78    async fn get(&self, query: GetAccountQuery) -> Result<AccountView, AccountError> {
79        let dql = dql::get_account(&query.context.id)?;
80        let key = dql.cache_key();
81
82        if let Some(hit) = self.views.get(CACHE_GROUP, &key).await? {
83            return Ok(hit);
84        }
85
86        let view = MasterScope::run(|uow| async move {
87            let Some(view) = self.queries.run(dql).await? else {
88                return Err(AccountError::InvalidCredentials);
89            };
90
91            uow.commit().await?;
92
93            Ok(view)
94        })
95        .await?;
96
97        // Falhar ao guardar não invalida a resposta: o cliente já tem o
98        // dado correto, e o único prejuízo é o próximo pedido recalcular.
99        self.views.put(CACHE_GROUP, &key, &view).await?;
100
101        Ok(view)
102    }
103
104    async fn update(&self, command: UpdateAccountCommand) -> Result<Box<dyn User>, AccountError> {
105        let user = MasterScope::run(|uow| async move {
106            let Some(existing) = self.users.find_by_id(&command.context.id).await? else {
107                return Err(AccountError::InvalidCredentials);
108            };
109
110            let updated = self
111                .user_tm
112                .update(existing.as_ref(), command.name, command.email)?;
113
114            self.users.update(updated.as_ref()).await?;
115
116            uow.commit().await?;
117
118            Ok(updated)
119        })
120        .await?;
121
122        self.views.invalidate(CACHE_GROUP).await?;
123
124        Ok(user)
125    }
126
127    /// Troca a senha do próprio usuário.
128    ///
129    /// A senha **atual** é exigida mesmo com a sessão válida: um token roubado
130    /// não deve bastar para trocar a senha e expulsar o dono.
131    async fn change_password(&self, command: ChangePasswordCommand) -> Result<(), AccountError> {
132        MasterScope::run(|uow| async move {
133            let Some(existing) = self.users.find_by_id(&command.context.id).await? else {
134                return Err(AccountError::InvalidCredentials);
135            };
136
137            self.auth_tm
138                .login(existing.as_ref(), &command.current_password)?;
139
140            let updated = self
141                .user_tm
142                .change_password(existing.as_ref(), command.new_password)?;
143
144            self.users.update(updated.as_ref()).await?;
145
146            uow.commit().await?;
147
148            Ok(())
149        })
150        .await?;
151
152        self.views.invalidate(CACHE_GROUP).await?;
153
154        Ok(())
155    }
156}
