1//! A orquestração de carga e telemetria.
2//!
3//! ## As permissões são privadas
4//!
5//! Os slugs abaixo são **contrato**: já existem em papéis gravados no banco de
6//! quem roda a versão PHP, e renomear qualquer um revoga silenciosamente o
7//! acesso de quem o tinha. São `const` privadas porque uma permissão pertence a
8//! exatamente um caso de uso — é ele quem a compara com o `UserContext`, e não
9//! há segundo lugar no sistema que precise vê-la. O boot as registra chamando
10//! `declare_permissions`, sem nunca lê-las.
11

25/// Embarcar carga.
26const LOAD: &str = "manifest:load";
27/// Desembarcar carga.
28const UNLOAD: &str = "manifest:unload";
29
30/// O prefixo que um embarque ou desembarque derruba.
31///
32/// É o do contêiner, não o deste serviço: quem muda é o peso, o status e o
33/// resumo de carga, e todos são leitura de contêiner. Este serviço não tem
34/// leitura própria a invalidar.
35const CONTAINER_CACHE_GROUP: &str = "container";
36
37/// Monta o caso de uso de manifesto.
38///
39/// Os ports chegam injetados e o que sai é o contrato: o tipo concreto não tem
40/// nome fora deste arquivo, então nada além do provider consegue depender do
41/// formato dele.
42pub(crate) fn manifest_service<CR, PR, MR, T, C>(
43    containers: CR,
44    products: PR,
45    manifest: MR,
46    manifest_tm: T,
47    views: C,
48) -> impl ManifestService + Sync + Clone + use<CR, PR, MR, T, C> + 'static
49where
50    CR: ContainerRepository + Send + Sync + Clone + 'static,
51    PR: ProductRepository + Send + Sync + Clone + 'static,
52    MR: ManifestRepository + Send + Sync + Clone + 'static,
53    T: ManifestTM + Send + Sync + Clone + 'static,
54    C: ViewCacheRepository + Send + Sync + Clone + 'static,
55{
56    ManifestServiceImpl {
57        containers,
58        products,
59        manifest,
60        manifest_tm,
61        views,
62    }
63}

65/// A implementação, genérica sobre os ports que consome.
66#[derive(Clone)]
67struct ManifestServiceImpl<CR, PR, MR, T, C> {
68    /// Persistência de contêineres.
69    containers: CR,
70    /// Persistência de produtos.
71    products: PR,
72    /// Persistência do manifesto e da telemetria.
73    manifest: MR,
74    /// As regras de embarque e desembarque.
75    manifest_tm: T,
76    /// O cache do lado de leitura.
77    views: C,
78}
79
80impl<CR, PR, MR, T, C> ManifestServiceImpl<CR, PR, MR, T, C>
81where
82    CR: ContainerRepository + Send + Sync,
83    PR: ProductRepository + Send + Sync,
84    MR: ManifestRepository + Send + Sync,
85    T: ManifestTM + Send + Sync,
86    C: ViewCacheRepository + Send + Sync,
87{
88    /// A moldura de embarque e desembarque.
89    ///
90    /// Carrega os três envolvidos, pede a mudança ao `TableModule` e a persiste.
91    /// O `apply` é o único ponto de diferença entre os dois casos.
92    ///
93    /// A carga atual pode não existir — é o primeiro embarque daquele produto.
94    /// Ausência aqui é **dado, não falha**.
95    ///
96    /// A mudança é consumida depois de gravada: o contêiner que sai daqui é o
97    /// mesmo que foi persistido, sem uma segunda leitura que poderia divergir.
98    async fn r#move(
99        &self,
100        command: MoveItemCommand,
101        apply: impl Fn(
102            &T,
103            &dyn Container,
104            &dyn Product,
105            f64,
106            Option<&dyn ManifestCargo>,
107        ) -> Result<Box<dyn ManifestChange>, ManifestError>,
108    ) -> Result<Box<dyn Container>, ManifestError> {
109        let container = MasterScope::run(|uow| async move {
110            let Some(container) = self.containers.find_by_id(&command.container_id).await? else {
111                return Err(ManifestError::MissingContainer(command.container_id));
112            };
113
114            let Some(product) = self.products.find_by_id(&command.product_id).await? else {
115                return Err(ManifestError::MissingProduct(command.product_id));
116            };
117
118            let current = self
119                .manifest
120                .find_cargo(&command.container_id, &command.product_id)
121                .await?;
122
123            let change = apply(
124                &self.manifest_tm,
125                container.as_ref(),
126                product.as_ref(),
127                command.quantity,
128                current.as_deref(),
129            )?;
130
131            self.persist(change.as_ref()).await?;
132
133            uow.commit().await?;
134
135            Ok(change.into_container())
136        })
137        .await?;
138
139        self.views.invalidate(CONTAINER_CACHE_GROUP).await?;
140
141        Ok(container)
142    }
143
144    /// Grava a mudança que o `domain` descreveu.
145    ///
146    /// Os três ramos são exclusivos e vêm do próprio `ManifestChange`: limpar o
147    /// manifesto inteiro (o desembarque zerou o contêiner), apagar a linha (o
148    /// produto saiu por completo) ou gravá-la (embarque, ou desembarque
149    /// parcial). Decidir aqui qual dos três seria reimplementar a regra do
150    /// domínio numa segunda cópia.
151    async fn persist(&self, change: &dyn ManifestChange) -> Result<(), ManifestError> {
152        let container_id = change.container().id().to_owned();
153
154        self.containers.update(change.container()).await?;
155
156        if change.clear_manifest() {
157            self.manifest.clear_manifest(&container_id).await?;
158        } else if let Some(cargo) = change.cargo() {
159            self.manifest.upsert_cargo(cargo).await?;
160        } else {
161            self.manifest
162                .delete_cargo(&container_id, change.product_id())
163                .await?;
164        }
165
166        self.manifest
167            .insert_telemetry(
168                &container_id,
169                change.event(),
170                Some(&format!("Product {}", change.product_id())),
171            )
172            .await?;
173
174        Ok(())
175    }
176}
177
178impl<CR, PR, MR, T, C> ManifestService for ManifestServiceImpl<CR, PR, MR, T, C>
179where
180    CR: ContainerRepository + Send + Sync,
181    PR: ProductRepository + Send + Sync,
182    MR: ManifestRepository + Send + Sync,
183    T: ManifestTM + Send + Sync,
184    C: ViewCacheRepository + Send + Sync,
185{
186    async fn declare_permissions<M: MetadataService + Send + Sync>(
187        &self,
188        registrar: &M,
189    ) -> Result<(), ManifestError> {
190        for slug in [LOAD, UNLOAD] {
191            registrar
192                .register_permission(RegisterPermissionCommand {
193                    slug: slug.to_owned(),
194                })
195                .await?;
196        }
197
198        Ok(())
199    }
200
201    async fn load(&self, command: MoveItemCommand) -> Result<Box<dyn Container>, ManifestError> {
202        if !command.context.has_permission(LOAD) {
203            return Err(AppError::permission_denied(LOAD).into());
204        }
205
206        self.r#move(command, |tm, container, product, quantity, current| {
207            Ok(tm.load(container, product, quantity, current)?)
208        })
209        .await
210    }
211
212    async fn unload(&self, command: MoveItemCommand) -> Result<Box<dyn Container>, ManifestError> {
213        if !command.context.has_permission(UNLOAD) {
214            return Err(AppError::permission_denied(UNLOAD).into());
215        }
216
217        self.r#move(command, |tm, container, product, quantity, current| {
218            Ok(tm.unload(container, product, quantity, current)?)
219        })
220        .await
221    }
222}
