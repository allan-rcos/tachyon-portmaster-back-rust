1//! A orquestração de marcadores.
2
3use portmaster_domain::table_modules::{MarkerGroupTM, MarkerTM};
4use portmaster_infra::repository::{MarkerGroupRepository, MarkerRepository};
5
6use crate::commands::marker::{RegisterMarkerGroupCommand, SetMarkerCommand};
7use crate::error::MarkerError;
8use crate::queries::marker::GetMarkerQuery;
9use crate::services::MarkUseCase;
10
11/// A implementação, genérica sobre os ports que consome.
12#[derive(Clone)]
13pub(crate) struct MarkUseCaseImpl<T, G, R, GR> {
14    /// As regras de marcador.
15    marker_tm: T,
16    /// As regras de grupo de marcador.
17    marker_group_tm: G,
18    /// Persistência dos marcadores.
19    markers: R,
20    /// Persistência dos grupos.
21    groups: GR,
22}
23
24impl<T, G, R, GR> MarkUseCaseImpl<T, G, R, GR> {
25    /// Monta o caso de uso.
26    pub(crate) const fn new(marker_tm: T, marker_group_tm: G, markers: R, groups: GR) -> Self {
27        Self {
28            marker_tm,
29            marker_group_tm,
30            markers,
31            groups,
32        }
33    }
34}
35
36impl<T, G, R, GR> MarkUseCase for MarkUseCaseImpl<T, G, R, GR>
37where
38    T: MarkerTM + Send + Sync,
39    G: MarkerGroupTM + Send + Sync,
40    R: MarkerRepository + Send + Sync,
41    GR: MarkerGroupRepository + Send + Sync,
42{
43    /// Declara um grupo de marcador.
44    ///
45    /// Sem transação: os grupos vivem num registro em memória, e envolvê-los
46    /// numa transação abriria uma conexão de banco para não consultar banco
47    /// nenhum. Sem checagem de permissão: quem chama é o boot, e não há
48    /// chamador.
49    async fn register_group(&self, command: RegisterMarkerGroupCommand) -> Result<(), MarkerError> {
50        let group = self.marker_group_tm.create(command.slug)?;
51
52        self.groups.register(group.as_ref()).await?;
53
54        Ok(())
55    }
56
57    /// Grava ou apaga um marcador.
58    ///
59    /// As regras de transição — remarcar o que já vale, revalidar o que foi
60    /// invalidado — são da `infra`, que é dona do estado. Reimplementá-las aqui
61    /// daria ao sistema duas opiniões sobre a mesma coisa.
62    async fn set(&self, command: SetMarkerCommand) -> Result<(), MarkerError> {
63        let marker = self
64            .marker_tm
65            .create(command.group, &command.value, command.flag)?;
66
67        self.markers
68            .put(marker.as_ref(), command.ttl_seconds)
69            .await?;
70
71        Ok(())
72    }
73
74    /// Cria um marcador só para chegar ao digest: é o `TableModule` que sabe
75    /// reduzir o valor em claro à chave, e duplicar essa conversão aqui faria
76    /// as duas divergirem no dia em que o hash mudasse.
77    async fn is_valid(&self, query: GetMarkerQuery) -> Result<bool, MarkerError> {
78        let marker = self.marker_tm.create(query.group, &query.value, false)?;
79
80        Ok(self.markers.is_valid(marker.group(), marker.key()).await?)
81    }
82}
