1//! As permissões que o serviço de contêineres exige.
2
3/// Os slugs de permissão do serviço de contêineres.
4///
5/// São contrato com os papéis já gravados — ver [`ProductPermissions`] para o
6/// racional da divisão por serviço.
7///
8/// [`ProductPermissions`]: crate::services::product_permissions::ProductPermissions
9pub(crate) struct ContainerPermissions;
10
11impl ContainerPermissions {
12    /// Registrar um contêiner.
13    pub(crate) const CREATE: &str = "container:create";
14    /// Remover um contêiner.
15    pub(crate) const DELETE: &str = "container:delete";
16    /// Despachar um contêiner.
17    pub(crate) const DISPATCH: &str = "container:dispatch";
18    /// Ler contêineres.
19    pub(crate) const READ: &str = "container:read";
20    /// Selar um contêiner.
21    pub(crate) const SEAL: &str = "container:seal";
22    /// Ler o resumo de carga e telemetria.
23    pub(crate) const SUMMARY: &str = "container:summary";
24    /// Alterar um contêiner.
25    pub(crate) const UPDATE: &str = "container:update";
26
27    /// O que o boot registra por este serviço.
28    pub(crate) const ALL: &[&str] = &[
29        Self::CREATE,
30        Self::DELETE,
31        Self::DISPATCH,
32        Self::READ,
33        Self::SEAL,
34        Self::SUMMARY,
35        Self::UPDATE,
36    ];
37}
