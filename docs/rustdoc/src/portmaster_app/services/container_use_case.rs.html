3use crate::commands::container::ContainerCommand;
4use crate::commands::container::CreateContainerCommand;
5use crate::commands::container::UpdateContainerCommand;
6use crate::error::AppError;
7use crate::queries::container::GetContainerQuery;
8use crate::queries::container::ListContainerSummariesQuery;
9use crate::queries::container::ListContainersQuery;
10use portmaster_domain::models::Container;
11use portmaster_infra::query::views::{
12    ContainerListView, ContainerSummaryListView, ContainerViewItem,
13};
14
15/// O que a apresentação pode pedir sobre contêineres.
16#[trait_variant::make(Send)]
17pub trait ContainerUseCase {
18    /// Registra e devolve o contêiner criado.
19    async fn create(&self, command: CreateContainerCommand)
20        -> Result<Box<dyn Container>, AppError>;
21
22    /// Altera a capacidade e devolve o contêiner atualizado.
23    async fn update(&self, command: UpdateContainerCommand)
24        -> Result<Box<dyn Container>, AppError>;
25
26    /// Remove — soft-delete.
27    async fn delete(&self, command: ContainerCommand) -> Result<(), AppError>;
28
29    /// Sela: exige estar carregando e ter ao menos 10% da capacidade.
30    async fn seal(&self, command: ContainerCommand) -> Result<(), AppError>;
31
32    /// Despacha: exige estar selado, e é o que impede despachar duas vezes.
33    async fn dispatch(&self, command: ContainerCommand) -> Result<(), AppError>;
34
35    /// Lê um contêiner.
36    async fn get(&self, query: GetContainerQuery) -> Result<ContainerViewItem, AppError>;
37
38    /// Lista contêineres.
39    async fn list(&self, query: ListContainersQuery) -> Result<ContainerListView, AppError>;
40
41    /// Lista contêineres com carga e telemetria recente.
42    async fn list_summaries(
43        &self,
44        query: ListContainerSummariesQuery,
45    ) -> Result<ContainerSummaryListView, AppError>;
46}
