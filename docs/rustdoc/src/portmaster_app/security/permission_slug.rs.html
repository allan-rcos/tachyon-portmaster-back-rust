1//! Os slugs de permissão do sistema.
2//!
3//! São **contrato**: já existem em papéis gravados no banco de quem roda a
4//! versão PHP. Renomear qualquer um destes revoga silenciosamente o acesso de
5//! quem o tinha.
6//!
7//! Namespace no molde do `Base62`: o nome do tipo é o que dá sentido às
8//! constantes, e `PermissionSlug::CONTAINER_CREATE` diz de onde o valor vem.
9
10/// Os slugs de permissão, literalmente como o PHP os tinha.
11pub struct PermissionSlug;
12
13impl PermissionSlug {
14    /// Registrar um contêiner.
15    pub const CONTAINER_CREATE: &str = "container:create";
16    /// Remover um contêiner.
17    pub const CONTAINER_DELETE: &str = "container:delete";
18    /// Despachar um contêiner.
19    pub const CONTAINER_DISPATCH: &str = "container:dispatch";
20    /// Ler contêineres.
21    pub const CONTAINER_READ: &str = "container:read";
22    /// Selar um contêiner.
23    pub const CONTAINER_SEAL: &str = "container:seal";
24    /// Ler o resumo de carga e telemetria.
25    pub const CONTAINER_SUMMARY: &str = "container:summary";
26    /// Alterar um contêiner.
27    pub const CONTAINER_UPDATE: &str = "container:update";
28    /// Embarcar carga.
29    pub const MANIFEST_LOAD: &str = "manifest:load";
30    /// Desembarcar carga.
31    pub const MANIFEST_UNLOAD: &str = "manifest:unload";
32    /// Ler o painel do pátio.
33    pub const METRICS_READ: &str = "metrics:read";
34    /// Listar as permissões registradas.
35    pub const PERMISSION_LIST: &str = "permission:list";
36    /// Cadastrar um produto.
37    pub const PRODUCT_CREATE: &str = "product:create";
38    /// Remover um produto.
39    pub const PRODUCT_DELETE: &str = "product:delete";
40    /// Ler produtos.
41    pub const PRODUCT_READ: &str = "product:read";
42    /// Alterar um produto.
43    pub const PRODUCT_UPDATE: &str = "product:update";
44    /// Criar um papel.
45    pub const ROLE_CREATE: &str = "role:create";
46    /// Ler papéis.
47    pub const ROLE_LIST: &str = "role:list";
48    /// Trocar as permissões de um papel.
49    pub const ROLE_UPDATE_PERMISSIONS: &str = "role:update-permissions";
50    /// Redefinir a senha de outro usuário.
51    pub const USER_CHANGE_PASSWORD: &str = "user:change-password";
52    /// Cadastrar um usuário.
53    pub const USER_CREATE: &str = "user:create";
54    /// Remover um usuário.
55    pub const USER_DELETE: &str = "user:delete";
56    /// Ler um usuário.
57    pub const USER_GET: &str = "user:get";
58    /// Listar usuários.
59    pub const USER_LIST: &str = "user:list";
60    /// Alterar um usuário.
61    pub const USER_UPDATE: &str = "user:update";
62    /// Trocar os papéis de um usuário.
63    pub const USER_UPDATE_ROLES: &str = "user:update-roles";
64}
