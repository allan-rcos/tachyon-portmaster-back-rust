5use crate::event::meta_event::MetaEvent;

7/// Abre o escopo da pilha e consulta o que foi registrado nele.
8///
9/// Público de propósito, e é a metade pública do par: quem **emite** é um caso
10/// de uso, quem **lê** é a apresentação. Um middleware abre o escopo, o caso de
11/// uso emite lá dentro, e outro middleware pergunta o que saiu — sem que nenhum
12/// dos três conheça os outros.
13///
14/// ## Sem escopo, nada acontece
15///
16/// As três operações são idempotentes quanto à existência do escopo: fora dele,
17/// [`flush`](Self::flush) não faz nada e [`captured`](Self::captured) responde
18/// `false`. É o que permite chamar um caso de uso de um teste, de um comando de
19/// linha ou de uma tarefa de fundo sem montar middleware nenhum.
20pub trait MetaEventStackSubscriber {
21    /// Roda `future` com uma pilha vazia instalada na tarefa.
22    ///
23    /// Um `task_local` não pode ser escrito de dentro, só envolvido — então esta
24    /// é a única forma de a pilha existir. Aninhar dois escopos instala uma
25    /// pilha nova: o de dentro não enxerga o que o de fora registrou, e o de
26    /// fora não recebe o que o de dentro emitiu.
27    ///
28    /// Não é `async fn` porque o futuro que embrulha já vem pronto de baixo, e
29    /// devolvê-lo direto é o que dispensa a variante `Send` gerada — o `Send`
30    /// sai daqui declarado.
31    fn scope<F>(&self, future: F) -> impl Future<Output = F::Output> + Send
32    where
33        F: Future + Send;
34
35    /// Esquece um evento, ou a pilha inteira.
36    ///
37    /// Com `Some`, só aquele evento deixa de estar registrado; com `None`, a
38    /// pilha volta a ser o que era quando o escopo abriu.
39    fn flush(&self, event: Option<MetaEvent>);
40
41    /// Se o evento foi registrado nesta tarefa.
42    fn captured(&self, event: MetaEvent) -> bool;
43}
