1//! A pilha de eventos da tarefa corrente.
2//!
3//! > **Lacuna conhecida do `lint-exports`:** o `STACK` abaixo nasce de
4//! > `tokio::task_local!`, e itens gerados por macro são invisíveis ao `syn`.
5//! > O arquivo exporta dois itens na prática, não um — o `MetaEventStack` e o
6//! > task-local que só ele alcança.
7//!
8//! ## A pilha é uma máscara de bits
9//!
10//! Um `u8`, um bit por evento. Não há alocação, não há `RefCell` e não há
11//! rastreio de empréstimo em execução: um `Cell` de um valor `Copy` se lê e se
12//! escreve inteiro.
13//!
14//! O preço é o teto de oito eventos e a perda da multiplicidade — emitir o mesmo
15//! evento duas vezes é indistinguível de emiti-lo uma. Nenhuma das duas coisas
16//! importa para quem pergunta "isto aconteceu?", que é a única pergunta que os
17//! `captured` do sistema fazem.
18//!
19//! A largura é **privada a este arquivo**. Nem o enum, nem os dois traits, nem
20//! os middlewares, nem os casos de uso a enxergam — trocar `u8` por `u16` é uma
21//! edição local, e a [`bit`] logo abaixo garante que ela não seja esquecida.
22
23use std::cell::Cell;
24use std::future::Future;
25
26use crate::event::meta_event::MetaEvent;
27use crate::event::meta_event_stack_publisher::MetaEventStackPublisher;
28use crate::event::meta_event_stack_subscriber::MetaEventStackSubscriber;
29
30tokio::task_local! {
31    /// Os eventos registrados nesta tarefa, um bit por evento.
32    ///
33    /// Interno a este módulo: de fora se chega pelos dois traits, e nenhum deles
34    /// revela que existe uma máscara aqui.
35    static STACK: Cell<u8>;
36}
37
38/// O bit de cada evento.
39///
40/// O `match` é exaustivo de propósito, e é a guarda do teto: acrescentar uma
41/// variante a [`MetaEvent`] derruba a compilação **aqui**, que é o único lugar
42/// que sabe a largura da máscara. É o que transforma o nono evento num
43/// alargamento consciente de `u8` para `u16`, em vez de um bit que sai do byte
44/// em silêncio.
45const fn bit(event: MetaEvent) -> u8 {
46    match event {
47        MetaEvent::ViewCacheHit => 1 << 0,
48    }
49}
50
51/// A pilha de eventos, vista dos dois lados.
52///
53/// ZST: não guarda nada, porque o estado é da tarefa e não dele. Injetá-lo custa
54/// o mesmo que não injetar, e o que se ganha é o caso de uso depender de um
55/// contrato em vez de alcançar um task-local de outro módulo.
56#[derive(Debug, Clone, Copy, Default)]
57pub(crate) struct MetaEventStack;
58
59impl MetaEventStack {
60    /// Aplica `change` à máscara, se houver escopo.
61    ///
62    /// O `Err` do `try_with` é descartado, e é só isto que faz toda operação
63    /// desta pilha virar no-op fora de um escopo aberto.
64    fn update(change: impl FnOnce(u8) -> u8) {
65        let _ = STACK.try_with(|mask| mask.set(change(mask.get())));
66    }
67}
68
69impl MetaEventStackSubscriber for MetaEventStack {
70    fn scope<F>(&self, future: F) -> impl Future<Output = F::Output> + Send
71    where
72        F: Future + Send,
73    {
74        STACK.scope(Cell::new(0), future)
75    }
76
77    fn flush(&self, event: Option<MetaEvent>) {
78        Self::update(|mask| event.map_or(0, |event| mask & !bit(event)));
79    }
80
81    fn captured(&self, event: MetaEvent) -> bool {
82        STACK
83            .try_with(|mask| mask.get() & bit(event) != 0)
84            .unwrap_or(false)
85    }
86}
87
88impl MetaEventStackPublisher for MetaEventStack {
89    fn emit(&self, event: MetaEvent) {
90        Self::update(|mask| mask | bit(event));
91    }
92}
