18/// Os factories dos services.
19///
20/// A apresentação recebe **contrato**: nenhum tipo concreto de service
21/// atravessa esta fronteira. Ver
22/// `docs/adr/0011-static-providers-one-per-directory.md`.
23///
24/// Oito dos dez devolvem `Result`, e dois não. A divisão vem de baixo: quem
25/// depende do pool pode falhar se o boot ainda não tiver instalado os segredos
26/// do banco, e quem só toca memória não tem como. Todas as chamadas acontecem
27/// no boot, dentro de funções que já devolvem `anyhow::Result`.
28pub struct AppProvider;
29
30impl AppProvider {
31    /// Inicializa o sistema inteiro.
32    ///
33    /// Instala a configuração das camadas de baixo em quem a consome, confirma
34    /// que o banco responde e declara o catálogo. O `main` chama isto e depois
35    /// monta o router, sem conhecer `domain` nem `infra`.
36    ///
37    /// ## O catálogo de permissões nasce aqui, e esta função não conhece um slug
38    ///
39    /// Cada serviço declara as suas, e o que este boot faz é pedir que declarem.
40    /// É o molde do `declarePermission` do PHP: a permissão pertence a exatamente
41    /// um caso de uso, e é ele quem a nomeia. Acrescentar um serviço é acrescentar
42    /// uma linha aqui — não editar uma lista central que ninguém lembra de manter.
43    ///
44    /// O catálogo precisa existir antes da primeira requisição: o `POST /setup`
45    /// concede ao primeiro papel tudo que estiver registrado. Falhar aqui derruba o
46    /// boot de propósito — um sistema que subiu com o catálogo pela metade daria 403
47    /// em endpoints que deveriam funcionar, e a causa seria invisível.
48    ///
49    /// Registrar no boot, e não no construtor de cada caso de uso como o PHP fazia,
50    /// evita repetir o registro a cada requisição: os services são reconstruídos
51    /// o tempo todo, o catálogo não.
52    ///
53    /// **Grupo de marcador não entra aqui.** Quem usa um grupo é a apresentação, e é
54    /// ela quem o declara — esta camada não sabe o que é sessão.
55    ///
56    /// É o único ponto em que um erro tipado vira `anyhow`: aqui é wiring de root, e
57    /// o que sai daqui vai para o `main`.
58    pub async fn boot(secrets: &AppSecrets) -> anyhow::Result<()> {
59        DomainProvider::install_identity(secrets.domain);
60        InfraProvider::install_database(&secrets.infra)?;
61        InfraProvider::check_database().await?;
62
63        let metadata = Self::metadata_service();
64
65        metadata
66            .declare_permissions()
67            .await
68            .context("falha ao registrar as permissões de metadados")?;
69
70        Self::container_service()?
71            .declare_permissions(&metadata)
72            .await
73            .context("falha ao registrar as permissões de contêiner")?;
74
75        Self::manifest_service()?
76            .declare_permissions(&metadata)
77            .await
78            .context("falha ao registrar as permissões de manifesto")?;
79
80        Self::metrics_service()?
81            .declare_permissions(&metadata)
82            .await
83            .context("falha ao registrar as permissões do painel")?;
84
85        Self::product_service()?
86            .declare_permissions(&metadata)
87            .await
88            .context("falha ao registrar as permissões de produto")?;
89
90        Self::role_service()?
91            .declare_permissions(&metadata)
92            .await
93            .context("falha ao registrar as permissões de papel")?;
94
95        Self::user_service()?
96            .declare_permissions(&metadata)
97            .await
98            .context("falha ao registrar as permissões de usuário")?;
99
100        Ok(())
101    }
