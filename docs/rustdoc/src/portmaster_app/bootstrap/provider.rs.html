11/// Os factories dos services.
12///
13/// Cada método devolve `impl Trait` — contrato, nunca tipo concreto. O tipo real
14/// é innomeável: só existe depois da monomorfização, e é exatamente por isso que
15/// a apresentação não consegue depender dele.
16///
17/// Os services são reconstruídos a cada chamada. Isso é barato de propósito:
18/// eles não guardam estado, e os recursos caros — pool, caches — nasceram uma
19/// vez no [`crate::register()`] e chegam por clone.
20pub trait AppProvider {
21    /// A conta do próprio usuário.
22    fn account_service(&self) -> impl AccountService + Sync + Clone + use<Self> + 'static;
23
24    /// Contêineres.
25    fn container_service(&self) -> impl ContainerService + Sync + Clone + use<Self> + 'static;
26
27    /// Carga e telemetria.
28    fn manifest_service(&self) -> impl ManifestService + Sync + Clone + use<Self> + 'static;
29
30    /// A primitiva de marcação — sessão de refresh é um uso dela.
31    fn mark_service(&self) -> impl MarkService + Sync + Clone + use<Self> + 'static;
32
33    /// Metadados de sistema.
34    fn metadata_service(&self) -> impl MetadataService + Sync + Clone + use<Self> + 'static;
35
36    /// O painel do pátio.
37    fn metrics_service(&self) -> impl MetricsService + Sync + Clone + use<Self> + 'static;
38
39    /// Produtos.
40    fn product_service(&self) -> impl ProductService + Sync + Clone + use<Self> + 'static;
41
42    /// Papéis.
43    fn role_service(&self) -> impl RoleService + Sync + Clone + use<Self> + 'static;
44
45    /// Login, validação de sessão e o setup inicial.
46    fn session_service(&self) -> impl SessionService + Sync + Clone + use<Self> + 'static;
47
48    /// Usuários.
49    fn user_service(&self) -> impl UserService + Sync + Clone + use<Self> + 'static;
50
51    /// Fábrica de loggers, para a apresentação.
52    fn logger_factory(&self) -> impl LoggerFactory + use<Self> + 'static;
53
54    /// Gerador de id opaco, para o refresh token.
55    fn random_id_generator(&self) -> impl RandomIdGenerator + use<Self>;
56
57    /// Gerador de id ordenável, para o `request_id`.
58    fn sequential_id_generator(&self) -> impl SequentialIdGenerator + use<Self>;
59}
