8/// Inicializa o sistema inteiro e devolve o provider da aplicação.
9///
10/// Encadeia os `register` das camadas de baixo e **embute** os subproviders.
11/// Não há composition root: o `main` chama isto e recebe algo pronto, sem
12/// conhecer `domain` nem `infra`.
13///
14/// ## O que acontece aqui e em nenhum outro lugar
15///
16/// O **catálogo de permissões** e o **grupo de marcador da sessão** são
17/// registrados no boot. Os dois precisam existir antes da primeira requisição:
18/// o `POST /setup` concede ao primeiro papel tudo que estiver registrado, e o
19/// repositório de marcadores recusa marcar num grupo desconhecido.
20///
21/// Registrar aqui, e não no construtor de cada caso de uso como o PHP fazia,
22/// evita repetir o registro a cada requisição — os casos de uso são
23/// reconstruídos o tempo todo, o catálogo não.
24pub async fn register(secrets: AppSecrets) -> anyhow::Result<impl AppProvider> {
25    let domain = portmaster_domain::register(secrets.domain);
26    let infra = portmaster_infra::register(secrets.infra).await?;
27
28    declare_metadata(&domain, &infra).await?;
29
30    Ok(AppProviderImpl::new(domain, infra))
31}
