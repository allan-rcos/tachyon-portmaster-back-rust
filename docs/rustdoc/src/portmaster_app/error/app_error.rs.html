1//! A falha que qualquer caso de uso pode devolver.

3use portmaster_domain::error::FieldError;
4
5use crate::error::AppErrorKind;

7/// O que pode dar errado em **qualquer** caso de uso.
8///
9/// São as três recusas que não pertencem a serviço nenhum: o corpo que não passa
10/// pela forma, a permissão que falta e a infraestrutura que caiu. Tudo o mais —
11/// "produto não encontrado", "esse contêiner não está selado" — é do serviço que
12/// levanta, e mora no erro dele.
13///
14/// As variantes descrevem a falha no vocabulário do sistema, não no do protocolo
15/// por onde ela sairá. Quem precisa decidir em cima do erro sem casar variante
16/// por variante usa [`AppError::kind`]; quem precisa de um status HTTP é o
17/// `api-http`, e a tradução mora lá.
18#[derive(Debug, thiserror::Error)]
19pub enum AppError {
20    /// Campos inválidos, em lote.
21    ///
22    /// Lista e não primeiro-erro: quem preencheu um formulário quer corrigir
23    /// tudo de uma vez, não descobrir um problema por requisição.
24    #[error("dados inválidos")]
25    Validation(Vec<FieldError>),
26
27    /// Falta a permissão exigida.
28    ///
29    /// O slug viaja para o log. O `api-http` **não** o coloca no corpo — é o
30    /// comportamento que o PHP tinha, e dizer ao cliente qual permissão faltou
31    /// descreve para ele o mapa de autorização do sistema.
32    #[error("permissão negada: {permission}")]
33    PermissionDenied {
34        /// O slug que faltava.
35        permission: &'static str,
36    },
37
38    /// Falha de I/O ou de infraestrutura.
39    ///
40    /// Opaca de propósito: "o banco recusou a conexão" não é uma regra que o
41    /// cliente possa satisfazer, e detalhar o motivo só vazaria topologia.
42    #[error(transparent)]
43    Infra(#[from] anyhow::Error),
44}
45
46impl AppError {
47    /// A recusa de quem não tem a permissão exigida.
48    ///
49    /// O slug é `&'static str` e não `String`: ele é uma constante do binário,
50    /// declarada pelo serviço que a exige, e alocá-la a cada recusa seria
51    /// trabalho por nada.
52    pub const fn permission_denied(permission: &'static str) -> Self {
53        Self::PermissionDenied { permission }
54    }
55
56    /// De que natureza é esta falha.
57    ///
58    /// O agrupamento é o contrato estável desta camada; quem decide em cima do
59    /// erro não precisa casar variante por variante.
60    pub const fn kind(&self) -> AppErrorKind {
61        match *self {
62            Self::Validation(_) => AppErrorKind::Validation,
63            Self::PermissionDenied { .. } => AppErrorKind::Authorization,
64            Self::Infra(_) => AppErrorKind::Internal,
65        }
66    }
67}
