1//! Persistência de grupos de marcador em memória.
2
3use portmaster_domain::domain::MarkerGroup;
4
5use crate::repository::MarkerGroupRepository;
6use crate::scope::memory::memory_store::MemoryStore;
7
8/// O grupo deste repositório — o que num repositório do `MariaDB` seria a tabela.
9const GROUP: &str = "marker-group";
10
11/// O repositório de grupos de marcador.
12///
13/// Vive num store próprio, e não no mesmo das permissões: se dividissem o
14/// armazenamento, `PermissionRepository::all()` devolveria `refresh-token` junto
15/// das permissões, e o papel que o `POST /setup` cria — que recebe tudo que
16/// estiver registrado — ganharia uma concessão que ninguém declarou. O schema
17/// anterior os separava em duas tabelas pelo mesmo motivo.
18#[derive(Clone)]
19pub struct MarkerGroupMemoryRepository<S> {
20    /// De onde a memória da tarefa vem.
21    store: S,
22}
23
24impl<S> MarkerGroupMemoryRepository<S> {
25    /// Monta o repositório.
26    pub(crate) const fn new(store: S) -> Self {
27        Self { store }
28    }
29}
30
31impl<S: MemoryStore + Send + Sync> MarkerGroupRepository for MarkerGroupMemoryRepository<S> {
32    async fn register(&self, group: &dyn MarkerGroup) -> anyhow::Result<()> {
33        self.store.put(GROUP, group.slug(), Vec::new(), None).await
34    }
35
36    async fn has(&self, slug: &str) -> anyhow::Result<bool> {
37        Ok(self.store.get(GROUP, slug).await?.is_some())
38    }
39}
