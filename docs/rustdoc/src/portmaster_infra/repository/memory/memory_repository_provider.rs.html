1//! Quem serve os repositórios em memória.
2
3use crate::repository::memory::marker_group_repository::marker_group_repository;
4use crate::repository::memory::marker_repository::marker_repository;
5use crate::repository::memory::permission_repository::permission_repository;
6use crate::repository::memory::view_cache_repository::view_cache_repository;
7use crate::repository::{
8    MarkerGroupRepository, MarkerRepository, PermissionRepository, ViewCacheRepository,
9};
10use crate::scope::ScopeProvider;
11
12/// Os repositórios cujo armazenamento é a memória do processo.
13///
14/// Infalíveis, ao contrário dos do `MariaDB`: um mapa em memória não depende de
15/// segredo nenhum para existir, então a assinatura não pede um `?` a quem não
16/// pode falhar.
17pub(crate) struct MemoryRepositoryProvider;
18
19impl MemoryRepositoryProvider {
20    /// O catálogo de permissões.
21    pub(crate) fn permission() -> impl PermissionRepository + Sync + Clone + use<> + 'static {
22        permission_repository(ScopeProvider::permissions())
23    }
24
25    /// O registro de grupos de marcador.
26    pub(crate) fn marker_group() -> impl MarkerGroupRepository + Sync + Clone + use<> + 'static {
27        marker_group_repository(ScopeProvider::marker_groups())
28    }
29
30    /// Os marcadores, já ligados ao registro de grupos que os valida.
31    pub(crate) fn marker() -> impl MarkerRepository + Sync + Clone + use<> + 'static {
32        marker_repository(ScopeProvider::markers(), Self::marker_group())
33    }
34
35    /// O cache do lado de leitura.
36    pub(crate) fn view_cache() -> impl ViewCacheRepository + Sync + Clone + use<> + 'static {
37        view_cache_repository(ScopeProvider::views())
38    }
39}
