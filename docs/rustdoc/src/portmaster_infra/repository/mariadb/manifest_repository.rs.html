13/// A linha de manifesto de um produto num contêiner.
14const FIND_CARGO: &str = "SELECT container_id, product_id, quantity, weight, created_at \
15                          FROM `container_items` WHERE container_id = ? AND product_id = ?";
16
17/// Upsert num comando só: o par (contêiner, produto) é a chave primária, então
18/// embarcar de novo o mesmo produto atualiza a linha em vez de duplicá-la — sem
19/// um `SELECT` antes para descobrir qual dos dois casos é.
20const UPSERT_CARGO: &str =
21    "INSERT INTO `container_items` (container_id, product_id, quantity, weight) \
22     VALUES (?, ?, ?, ?) \
23     ON DUPLICATE KEY UPDATE quantity = VALUES(quantity), weight = VALUES(weight)";
24
25/// `DELETE` de verdade: carga é entidade fraca. Não há o que preservar numa linha
26/// que só existia enquanto o produto estava no contêiner.
27const DELETE_CARGO: &str =
28    "DELETE FROM `container_items` WHERE container_id = ? AND product_id = ?";
29
30/// Esvazia o manifesto inteiro de um contêiner.
31const CLEAR_MANIFEST: &str = "DELETE FROM `container_items` WHERE container_id = ?";
32
33/// Registra o evento de embarque ou desembarque.
34///
35/// Usa `UTC_TIMESTAMP()` e não `NOW()`: o `NOW()` devolve o fuso da **sessão**,
36/// e um horário de telemetria que depende de como a conexão foi aberta não é um
37/// horário. A sessão do pool já é fixada em `+00:00` (ver
38/// [`pool`](crate::database::pool)), o que faria os dois coincidirem — mas
39/// escrever o que se quer dizer é o que mantém a linha correta se alguém um dia
40/// rodar este SQL por outro caminho.
41const INSERT_TELEMETRY: &str =
42    "INSERT INTO `telemetry_logs` (container_id, event, description, timestamp) \
43     VALUES (?, ?, ?, UTC_TIMESTAMP())";
44
45/// O repositório de manifesto.
46pub struct ManifestMariadbRepository;
47
48impl ManifestMariadbRepository {
49    /// Monta o repositório.
50    pub(crate) const fn new() -> Self {
51        Self
52    }
53}
54
55impl ManifestRepository for ManifestMariadbRepository {
56    async fn find_cargo(
57        &self,
58        container_id: &str,
59        product_id: &str,
60    ) -> anyhow::Result<Option<Box<dyn ManifestCargo>>> {
61        let raw_container = Codec::decode_id(container_id)?;
62        let raw_product = Codec::decode_id(product_id)?;
63        let mut transaction = MariadbUnitOfWork::current().await?;
64
65        let row: Option<ManifestCargoRow> = sqlx::query_as(FIND_CARGO)
66            .bind(raw_container)
67            .bind(raw_product)
68            .fetch_optional(&mut **transaction.as_mut())
69            .await
70            .with_context(|| {
71                format!("falha ao buscar a carga de {product_id} no contêiner {container_id}")
72            })?;
73
74        Ok(row
75            .map(ManifestCargoEntity::from_row)
76            .map(|entity| Box::new(entity) as Box<dyn ManifestCargo>))
77    }
78
79    async fn upsert_cargo(&self, cargo: &dyn ManifestCargo) -> anyhow::Result<()> {
80        let entity = ManifestCargoEntity::from_domain(cargo)?;
81        let mut transaction = MariadbUnitOfWork::current().await?;
82
83        sqlx::query(UPSERT_CARGO)
84            .bind(entity.raw_container_id())
85            .bind(entity.raw_product_id())
86            .bind(entity.quantity())
87            .bind(entity.weight())
88            .execute(&mut **transaction.as_mut())
89            .await
90            .context("falha ao gravar a linha do manifesto")?;
91
92        Ok(())
93    }
94
95    async fn delete_cargo(&self, container_id: &str, product_id: &str) -> anyhow::Result<()> {
96        let raw_container = Codec::decode_id(container_id)?;
97        let raw_product = Codec::decode_id(product_id)?;
98        let mut transaction = MariadbUnitOfWork::current().await?;
99
100        sqlx::query(DELETE_CARGO)
101            .bind(raw_container)
102            .bind(raw_product)
103            .execute(&mut **transaction.as_mut())
104            .await
105            .context("falha ao remover a linha do manifesto")?;
106
107        Ok(())
108    }
109
110    async fn clear_manifest(&self, container_id: &str) -> anyhow::Result<()> {
111        let raw_container = Codec::decode_id(container_id)?;
112        let mut transaction = MariadbUnitOfWork::current().await?;
113
114        sqlx::query(CLEAR_MANIFEST)
115            .bind(raw_container)
116            .execute(&mut **transaction.as_mut())
117            .await
118            .with_context(|| format!("falha ao limpar o manifesto do contêiner {container_id}"))?;
119
120        Ok(())
121    }
122
123    async fn insert_telemetry(
124        &self,
125        container_id: &str,
126        event: TelemetryEvent,
127        description: Option<&str>,
128    ) -> anyhow::Result<()> {
129        let raw_container = Codec::decode_id(container_id)?;
130        let mut transaction = MariadbUnitOfWork::current().await?;
131
132        sqlx::query(INSERT_TELEMETRY)
133            .bind(raw_container)
134            .bind(event.as_i32())
135            .bind(description)
136            .execute(&mut **transaction.as_mut())
137            .await
138            .with_context(|| {
139                format!("falha ao registrar telemetria do contêiner {container_id}")
140            })?;
141
142        Ok(())
143    }
144}
