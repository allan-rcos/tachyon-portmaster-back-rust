14/// Monta o repositório de usuários sobre o de papéis.
15///
16/// O de papéis chega injetado porque um usuário não está completo sem eles: os
17/// papéis decidem o que ele pode fazer, e devolvê-lo sem papéis faria toda
18/// verificação de autorização falhar em silêncio.
19pub(super) fn user_repository<R, T>(
20    roles: R,
21    transactions: T,
22) -> impl UserRepository + Sync + Clone + use<R, T> + 'static
23where
24    R: RoleRepository + Send + Sync + Clone + 'static,
25    T: MySqlTransaction + Send + Sync + Clone + 'static,
26{
27    UserMariadbRepository {
28        roles,
29        transactions,
30    }
31}
32
33/// O repositório de usuários, genérico sobre o de papéis.
34#[derive(Clone)]
35struct UserMariadbRepository<R, T> {
36    /// De onde os papéis do usuário são lidos, na mesma leitura.
37    roles: R,
38    /// De onde a transação da tarefa vem.
39    transactions: T,
40}
41
42impl<R: RoleRepository, T> UserMariadbRepository<R, T> {
43    /// Completa a entity buscando os papéis do usuário.
44    ///
45    /// A busca dos papéis acontece **depois** de soltar o empréstimo da
46    /// transação: as duas consultas usam a mesma conexão, e segurá-la durante a
47    /// segunda travaria a si mesma.
48    async fn hydrate(&self, entity: UserEntity) -> anyhow::Result<UserEntity> {
49        let roles = self.roles.find_by_user_id(entity.id()).await?;
50        Ok(entity.with_roles(roles))
51    }
52}
53
54impl<R: RoleRepository + Send + Sync, T: MySqlTransaction + Send + Sync> UserRepository
55    for UserMariadbRepository<R, T>
56{
57    /// `LIMIT 1` e não `COUNT(*)`: a pergunta é "existe algum", e contar todos
58    /// para descobrir isso varre a tabela inteira à toa.
59    async fn has_any(&self) -> anyhow::Result<bool> {
60        let mut transaction = self.transactions.transaction().await?;
61
62        let found: Option<i64> = transaction
63            .exec_first("SELECT 1 FROM `users` WHERE deleted_at IS NULL LIMIT 1", ())
64            .await
65            .context("falha ao verificar se há algum usuário")?;
66
67        Ok(found.is_some())
68    }
69
70    /// Busca por id, já filtrando o soft-delete.
71    async fn find_by_id(&self, id: &str) -> anyhow::Result<Option<Box<dyn User>>> {
72        let raw_id = Codec::decode_id(id)?;
73
74        let entity: Option<UserEntity> = {
75            let mut transaction = self.transactions.transaction().await?;
76            transaction
77                .exec_first(
78                    "SELECT id, name, email, password_hash, created_at, updated_at, deleted_at \
79                     FROM `users` WHERE id = :id AND deleted_at IS NULL",
80                    params! { "id" => raw_id },
81                )
82                .await
83                .with_context(|| format!("falha ao buscar o usuário {id}"))?
84        };
85
86        match entity {
87            Some(entity) => Ok(Some(Box::new(self.hydrate(entity).await?))),
88            None => Ok(None),
89        }
90    }
91
92    /// O filtro de removidos vale também aqui: um e-mail liberado por remoção
93    /// precisa poder ser reusado.
94    async fn find_by_email(&self, email: &str) -> anyhow::Result<Option<Box<dyn User>>> {
95        let entity: Option<UserEntity> = {
96            let mut transaction = self.transactions.transaction().await?;
97            transaction
98                .exec_first(
99                    "SELECT id, name, email, password_hash, created_at, updated_at, deleted_at \
100                     FROM `users` WHERE email = :email AND deleted_at IS NULL",
101                    params! { "email" => email },
102                )
103                .await
104                .context("falha ao buscar usuário por e-mail")?
105        };
106
107        match entity {
108            Some(entity) => Ok(Some(Box::new(self.hydrate(entity).await?))),
109            None => Ok(None),
110        }
111    }
112
113    /// Grava a linha nova, com os dois instantes que o modelo já carimbou.
114    async fn insert(&self, user: &dyn User) -> anyhow::Result<()> {
115        let entity = UserEntity::from_domain(user)?;
116        let mut transaction = self.transactions.transaction().await?;
117
118        transaction
119            .exec_drop(
120                "INSERT INTO `users` (id, name, email, password_hash, created_at, updated_at) \
121                 VALUES (:id, :name, :email, :password_hash, :created_at, :updated_at)",
122                params! {
123                    "id" => entity.raw_id(),
124                    "name" => entity.name(),
125                    "email" => entity.email(),
126                    "password_hash" => entity.password_hash(),
127                    "created_at" => entity.created_at().timestamp_millis(),
128                    "updated_at" => entity.updated_at().timestamp_millis(),
129                },
130            )
131            .await
132            .with_context(|| format!("falha ao gravar o usuário {}", user.id()))?;
133
134        Ok(())
135    }
136
137    /// Atualiza a linha existente.
138    async fn update(&self, user: &dyn User) -> anyhow::Result<()> {
139        let entity = UserEntity::from_domain(user)?;
140        let mut transaction = self.transactions.transaction().await?;
141
142        transaction
143            .exec_drop(
144                "UPDATE `users` SET name = :name, email = :email, \
145                 password_hash = :password_hash, updated_at = :updated_at \
146                 WHERE id = :id AND deleted_at IS NULL",
147                params! {
148                    "name" => entity.name(),
149                    "email" => entity.email(),
150                    "password_hash" => entity.password_hash(),
151                    "updated_at" => entity.updated_at().timestamp_millis(),
152                    "id" => entity.raw_id(),
153                },
154            )
155            .await
156            .with_context(|| format!("falha ao atualizar o usuário {}", user.id()))?;
157
158        Ok(())
159    }
160
161    /// Marca como removido em vez de apagar — o histórico continua auditável.
162    async fn delete(&self, id: &str) -> anyhow::Result<()> {
163        let raw_id = Codec::decode_id(id)?;
164        let mut transaction = self.transactions.transaction().await?;
165
166        transaction
167            .exec_drop(
168                "UPDATE `users` SET deleted_at = :now, updated_at = :now \
169                 WHERE id = :id AND deleted_at IS NULL",
170                params! { "now" => Utc::now().timestamp_millis(), "id" => raw_id },
171            )
172            .await
173            .with_context(|| format!("falha ao remover o usuário {id}"))?;
174
175        Ok(())
176    }
177
178    /// Substitui os vínculos de papel do usuário — apaga e regrava.
179    ///
180    /// O vínculo é entidade fraca, e "mudar" um conjunto de vínculos é
181    /// removê-los e recriá-los. Calcular o diferencial daria o mesmo resultado
182    /// por mais trabalho, e os dois rodam na mesma transação.
183    async fn sync_roles(&self, user_id: &str, role_ids: &[String]) -> anyhow::Result<()> {
184        let raw_user = Codec::decode_id(user_id)?;
185        let raw_roles: Vec<i64> = role_ids
186            .iter()
187            .map(|id| Codec::decode_id(id))
188            .collect::<anyhow::Result<_>>()?;
189
190        let mut transaction = self.transactions.transaction().await?;
191
192        transaction
193            .exec_drop(
194                "DELETE FROM `user_roles` WHERE user_id = :user_id",
195                params! { "user_id" => raw_user },
196            )
197            .await
198            .with_context(|| format!("falha ao limpar os papéis do usuário {user_id}"))?;
199
200        for role_id in raw_roles {
201            transaction
202                .exec_drop(
203                    "INSERT INTO `user_roles` (user_id, role_id) VALUES (:user_id, :role_id)",
204                    params! { "user_id" => raw_user, "role_id" => role_id },
205                )
206                .await
207                .with_context(|| format!("falha ao vincular papel ao usuário {user_id}"))?;
208        }
209
210        Ok(())
211    }
