3use anyhow::Context;
4use portmaster_domain::models::Role;

6use crate::database::interno::mariadb_unit_of_work::MariadbUnitOfWork;
7use crate::entity::codec::Codec;
8use crate::entity::role_entity::RoleEntity;

10use crate::repository::RoleRepository;
11use crate::text::search_key::SearchKey;

13/// Busca por id, já filtrando o soft-delete.
14const FIND_BY_ID: &str = "SELECT id, name, permissions, created_at, updated_at, deleted_at \
15     FROM `roles` WHERE id = ? AND deleted_at IS NULL";
16
17/// A ordem sai pelo id do papel, que é um Snowflake — ordenável por tempo de
18/// criação. É o mais próximo de "ordem de concessão" que o schema guarda, já que
19/// a tabela de ligação não tem coluna de sequência.
20const FIND_BY_USER: &str =
21    "SELECT r.id, r.name, r.permissions, r.created_at, r.updated_at, r.deleted_at \
22     FROM `roles` r \
23     INNER JOIN `user_roles` ur ON ur.role_id = r.id \
24     WHERE ur.user_id = ? AND r.deleted_at IS NULL \
25     ORDER BY r.id";
26
27/// Grava a linha nova.
28const INSERT: &str = "INSERT INTO `roles` (id, name, permissions, search_name) VALUES (?, ?, ?, ?)";
29
30/// Atualiza a linha existente.
31const UPDATE: &str = "UPDATE `roles` SET name = ?, permissions = ?, search_name = ? \
32                      WHERE id = ? AND deleted_at IS NULL";
33
34/// Marca como removida em vez de apagar — o histórico continua auditável.
35const SOFT_DELETE: &str =
36    "UPDATE `roles` SET deleted_at = NOW() WHERE id = ? AND deleted_at IS NULL";
37
38/// O repositório de papéis.
39pub struct RoleMariadbRepository;
40
41impl RoleMariadbRepository {
42    /// Monta o repositório.
43    pub(crate) const fn new() -> Self {
44        Self
45    }
46}
47
48impl RoleRepository for RoleMariadbRepository {
49    async fn find_by_id(&self, id: &str) -> anyhow::Result<Option<Box<dyn Role>>> {
50        let raw_id = Codec::decode_id(id)?;
51        let mut transaction = MariadbUnitOfWork::current().await?;
52
53        let row: Option<RoleRow> = sqlx::query_as(FIND_BY_ID)
54            .bind(raw_id)
55            .fetch_optional(&mut **transaction.as_mut())
56            .await
57            .with_context(|| format!("falha ao buscar o papel {id}"))?;
58
59        row.map(RoleEntity::from_row)
60            .transpose()
61            .map(|entity| entity.map(|e| Box::new(e) as Box<dyn Role>))
62    }
63
64    async fn find_by_user_id(&self, user_id: &str) -> anyhow::Result<Vec<Box<dyn Role>>> {
65        let raw_user = Codec::decode_id(user_id)?;
66        let mut transaction = MariadbUnitOfWork::current().await?;
67
68        let rows: Vec<RoleRow> = sqlx::query_as(FIND_BY_USER)
69            .bind(raw_user)
70            .fetch_all(&mut **transaction.as_mut())
71            .await
72            .with_context(|| format!("falha ao buscar os papéis do usuário {user_id}"))?;
73
74        rows.into_iter()
75            .map(|row| RoleEntity::from_row(row).map(|e| Box::new(e) as Box<dyn Role>))
76            .collect()
77    }
78
79    async fn insert(&self, role: &dyn Role) -> anyhow::Result<()> {
80        let entity = RoleEntity::from_domain(role)?;
81        let mut transaction = MariadbUnitOfWork::current().await?;
82
83        sqlx::query(INSERT)
84            .bind(entity.raw_id())
85            .bind(entity.name())
86            .bind(entity.permissions_json()?)
87            .bind(SearchKey::of(entity.name()))
88            .execute(&mut **transaction.as_mut())
89            .await
90            .with_context(|| format!("falha ao gravar o papel {}", role.id()))?;
91
92        Ok(())
93    }
94
95    async fn update(&self, role: &dyn Role) -> anyhow::Result<()> {
96        let entity = RoleEntity::from_domain(role)?;
97        let mut transaction = MariadbUnitOfWork::current().await?;
98
99        sqlx::query(UPDATE)
100            .bind(entity.name())
101            .bind(entity.permissions_json()?)
102            .bind(SearchKey::of(entity.name()))
103            .bind(entity.raw_id())
104            .execute(&mut **transaction.as_mut())
105            .await
106            .with_context(|| format!("falha ao atualizar o papel {}", role.id()))?;
107
108        Ok(())
109    }
110
111    async fn delete(&self, id: &str) -> anyhow::Result<()> {
112        let raw_id = Codec::decode_id(id)?;
113        let mut transaction = MariadbUnitOfWork::current().await?;
114
115        sqlx::query(SOFT_DELETE)
116            .bind(raw_id)
117            .execute(&mut **transaction.as_mut())
118            .await
119            .with_context(|| format!("falha ao remover o papel {id}"))?;
120
121        Ok(())
122    }
123}
