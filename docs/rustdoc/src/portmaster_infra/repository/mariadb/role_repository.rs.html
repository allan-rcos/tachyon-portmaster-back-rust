15/// Monta o repositório de papéis.
16///
17/// Não guarda estado: a transação vem do escopo da tarefa, não de um campo — o
18/// que permite ao provider reconstruí-lo a cada chamada por custo praticamente
19/// zero.
20pub(super) fn role_repository<T>(
21    transactions: T,
22) -> impl RoleRepository + Sync + Clone + use<T> + 'static
23where
24    T: MySqlTransaction + Send + Sync + Clone + 'static,
25{
26    RoleMariadbRepository { transactions }
27}
28
29/// O repositório de papéis, sobre o `MariaDB`.
30#[derive(Clone)]
31struct RoleMariadbRepository<T> {
32    /// De onde a transação da tarefa vem.
33    transactions: T,
34}
35
36impl<T: MySqlTransaction + Send + Sync> RoleRepository for RoleMariadbRepository<T> {
37    /// Busca por id, já filtrando o soft-delete.
38    async fn find_by_id(&self, id: &str) -> anyhow::Result<Option<Box<dyn Role>>> {
39        let raw_id = Codec::decode_id(id)?;
40        let mut transaction = self.transactions.transaction().await?;
41
42        let entity: Option<RoleEntity> = transaction
43            .exec_first(
44                "SELECT id, name, permissions, created_at, updated_at, deleted_at \
45                 FROM `roles` WHERE id = :id AND deleted_at IS NULL",
46                params! { "id" => raw_id },
47            )
48            .await
49            .with_context(|| format!("falha ao buscar o papel {id}"))?;
50
51        Ok(entity.map(|entity| Box::new(entity) as Box<dyn Role>))
52    }
53
54    /// Os papéis de um usuário, na ordem em que o schema consegue reportá-la.
55    ///
56    /// A ordem sai pelo id do papel, que é um Snowflake — ordenável por tempo de
57    /// criação. É o mais próximo de "ordem de concessão" que o schema guarda, já
58    /// que a tabela de ligação não tem coluna de sequência.
59    async fn find_by_user_id(&self, user_id: &str) -> anyhow::Result<Vec<Box<dyn Role>>> {
60        let raw_user = Codec::decode_id(user_id)?;
61        let mut transaction = self.transactions.transaction().await?;
62
63        let entities: Vec<RoleEntity> = transaction
64            .exec(
65                "SELECT r.id, r.name, r.permissions, r.created_at, r.updated_at, r.deleted_at \
66                 FROM `roles` r \
67                 INNER JOIN `user_roles` ur ON ur.role_id = r.id \
68                 WHERE ur.user_id = :user_id AND r.deleted_at IS NULL \
69                 ORDER BY r.id",
70                params! { "user_id" => raw_user },
71            )
72            .await
73            .with_context(|| format!("falha ao buscar os papéis do usuário {user_id}"))?;
74
75        Ok(entities
76            .into_iter()
77            .map(|entity| Box::new(entity) as Box<dyn Role>)
78            .collect())
79    }
80
81    /// Grava a linha nova, com os dois instantes que o modelo já carimbou.
82    async fn insert(&self, role: &dyn Role) -> anyhow::Result<()> {
83        let entity = RoleEntity::from_domain(role)?;
84        let mut transaction = self.transactions.transaction().await?;
85
86        transaction
87            .exec_drop(
88                "INSERT INTO `roles` (id, name, permissions, search_name, created_at, updated_at) \
89                 VALUES (:id, :name, :permissions, :search_name, :created_at, :updated_at)",
90                params! {
91                    "id" => entity.raw_id(),
92                    "name" => entity.name(),
93                    "permissions" => entity.permissions_json()?,
94                    "search_name" => SearchKey::of(entity.name()),
95                    "created_at" => entity.created_at().timestamp_millis(),
96                    "updated_at" => entity.updated_at().timestamp_millis(),
97                },
98            )
99            .await
100            .with_context(|| format!("falha ao gravar o papel {}", role.id()))?;
101
102        Ok(())
103    }
104
105    /// Atualiza a linha existente.
106    async fn update(&self, role: &dyn Role) -> anyhow::Result<()> {
107        let entity = RoleEntity::from_domain(role)?;
108        let mut transaction = self.transactions.transaction().await?;
109
110        transaction
111            .exec_drop(
112                "UPDATE `roles` SET name = :name, permissions = :permissions, \
113                 search_name = :search_name, updated_at = :updated_at \
114                 WHERE id = :id AND deleted_at IS NULL",
115                params! {
116                    "name" => entity.name(),
117                    "permissions" => entity.permissions_json()?,
118                    "search_name" => SearchKey::of(entity.name()),
119                    "updated_at" => entity.updated_at().timestamp_millis(),
120                    "id" => entity.raw_id(),
121                },
122            )
123            .await
124            .with_context(|| format!("falha ao atualizar o papel {}", role.id()))?;
125
126        Ok(())
127    }
128
129    /// Marca como removido em vez de apagar — o histórico continua auditável.
130    async fn delete(&self, id: &str) -> anyhow::Result<()> {
131        let raw_id = Codec::decode_id(id)?;
132        let mut transaction = self.transactions.transaction().await?;
133
134        transaction
135            .exec_drop(
136                "UPDATE `roles` SET deleted_at = :now, updated_at = :now \
137                 WHERE id = :id AND deleted_at IS NULL",
138                params! { "now" => Utc::now().timestamp_millis(), "id" => raw_id },
139            )
140            .await
141            .with_context(|| format!("falha ao remover o papel {id}"))?;
142
143        Ok(())
144    }
145}
