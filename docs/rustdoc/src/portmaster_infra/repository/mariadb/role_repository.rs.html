12/// Busca por id, já filtrando o soft-delete.
13const FIND_BY_ID: &str = "SELECT id, name, permissions, created_at, updated_at, deleted_at \
14     FROM `roles` WHERE id = ? AND deleted_at IS NULL";
15
16/// A ordem sai pelo id do papel, que é um Snowflake — ordenável por tempo de
17/// criação. É o mais próximo de "ordem de concessão" que o schema guarda, já que
18/// a tabela de ligação não tem coluna de sequência.
19const FIND_BY_USER: &str =
20    "SELECT r.id, r.name, r.permissions, r.created_at, r.updated_at, r.deleted_at \
21     FROM `roles` r \
22     INNER JOIN `user_roles` ur ON ur.role_id = r.id \
23     WHERE ur.user_id = ? AND r.deleted_at IS NULL \
24     ORDER BY r.id";
25
26/// Grava a linha nova.
27const INSERT: &str = "INSERT INTO `roles` (id, name, permissions, search_name) VALUES (?, ?, ?, ?)";
28
29/// Atualiza a linha existente.
30const UPDATE: &str = "UPDATE `roles` SET name = ?, permissions = ?, search_name = ? \
31                      WHERE id = ? AND deleted_at IS NULL";
32
33/// Marca como removida em vez de apagar — o histórico continua auditável.
34const SOFT_DELETE: &str =
35    "UPDATE `roles` SET deleted_at = NOW() WHERE id = ? AND deleted_at IS NULL";

37/// O repositório de papéis.
38#[derive(Clone)]
39pub struct RoleMariadbRepository<T> {
40    /// De onde a transação da tarefa vem.
41    transactions: T,
42}
43
44impl<T> RoleMariadbRepository<T> {
45    /// Monta o repositório.
46    pub(crate) const fn new(transactions: T) -> Self {
47        Self { transactions }
48    }
49}
50
51impl<T: MySqlTransaction + Send + Sync> RoleRepository for RoleMariadbRepository<T> {
52    async fn find_by_id(&self, id: &str) -> anyhow::Result<Option<Box<dyn Role>>> {
53        let raw_id = Codec::decode_id(id)?;
54        let mut transaction = self.transactions.transaction().await?;
55
56        let entity: Option<RoleEntity> = sqlx::query_as(FIND_BY_ID)
57            .bind(raw_id)
58            .fetch_optional(&mut **transaction)
59            .await
60            .with_context(|| format!("falha ao buscar o papel {id}"))?;
61
62        Ok(entity.map(|entity| Box::new(entity) as Box<dyn Role>))
63    }
64
65    async fn find_by_user_id(&self, user_id: &str) -> anyhow::Result<Vec<Box<dyn Role>>> {
66        let raw_user = Codec::decode_id(user_id)?;
67        let mut transaction = self.transactions.transaction().await?;
68
69        let entities: Vec<RoleEntity> = sqlx::query_as(FIND_BY_USER)
70            .bind(raw_user)
71            .fetch_all(&mut **transaction)
72            .await
73            .with_context(|| format!("falha ao buscar os papéis do usuário {user_id}"))?;
74
75        Ok(entities
76            .into_iter()
77            .map(|entity| Box::new(entity) as Box<dyn Role>)
78            .collect())
79    }
80
81    async fn insert(&self, role: &dyn Role) -> anyhow::Result<()> {
82        let entity = RoleEntity::from_domain(role)?;
83        let mut transaction = self.transactions.transaction().await?;
84
85        sqlx::query(INSERT)
86            .bind(entity.raw_id())
87            .bind(entity.name())
88            .bind(entity.permissions_json()?)
89            .bind(SearchKey::of(entity.name()))
90            .execute(&mut **transaction)
91            .await
92            .with_context(|| format!("falha ao gravar o papel {}", role.id()))?;
93
94        Ok(())
95    }
96
97    async fn update(&self, role: &dyn Role) -> anyhow::Result<()> {
98        let entity = RoleEntity::from_domain(role)?;
99        let mut transaction = self.transactions.transaction().await?;
100
101        sqlx::query(UPDATE)
102            .bind(entity.name())
103            .bind(entity.permissions_json()?)
104            .bind(SearchKey::of(entity.name()))
105            .bind(entity.raw_id())
106            .execute(&mut **transaction)
107            .await
108            .with_context(|| format!("falha ao atualizar o papel {}", role.id()))?;
109
110        Ok(())
111    }
112
113    async fn delete(&self, id: &str) -> anyhow::Result<()> {
114        let raw_id = Codec::decode_id(id)?;
115        let mut transaction = self.transactions.transaction().await?;
116
117        sqlx::query(SOFT_DELETE)
118            .bind(raw_id)
119            .execute(&mut **transaction)
120            .await
121            .with_context(|| format!("falha ao remover o papel {id}"))?;
122
123        Ok(())
124    }
125}
