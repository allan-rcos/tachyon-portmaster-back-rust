1//! O contrato de persistência de marker.
2
3use portmaster_domain::domain::Marker;
4
5/// Marcadores booleanos com prazo.
6///
7/// A `infra` nunca sabe o que uma marca significa — só que é um booleano com
8/// validade, num grupo conhecido.
9#[trait_variant::make(Send)]
10pub trait MarkerRepository {
11    /// Grava a marca, aplicando as regras de transição.
12    async fn put(&self, marker: &dyn Marker, ttl_seconds: u64) -> anyhow::Result<()>;
13
14    /// Se a marca está válida.
15    ///
16    /// Marca inexistente, expirada ou desligada respondem igual: `false`. Quem
17    /// pergunta quer saber se pode seguir, não por que não pode.
18    async fn is_valid(&self, group: &str, key: &str) -> anyhow::Result<bool>;
19}
