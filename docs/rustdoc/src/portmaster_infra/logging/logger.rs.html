1//! O logger estruturado que as camadas carregam.
2
3use std::collections::BTreeMap;
4
5/// Um logger nomeado, com campos acumulados.
6///
7/// É barato de clonar e de estender: `with_field` devolve um logger novo em vez
8/// de alterar o corrente, o que permite carimbar o `request_id` num escopo sem
9/// vazá-lo para os demais.
10#[derive(Debug, Clone)]
11pub struct Logger {
12    /// O nome do logger, que vira o alvo da linha.
13    name: String,
14    /// Os campos fixos deste logger, herdados por toda linha que ele emite.
15    ///
16    /// `BTreeMap` e não `HashMap`: a ordem estável faz duas linhas do mesmo
17    /// evento saírem iguais, o que importa para quem faz diff de log.
18    fields: BTreeMap<String, String>,
19}
20
21impl Logger {
22    /// Monta um logger para um componente.
23    pub(crate) fn new(name: &str) -> Self {
24        Self {
25            name: name.to_owned(),
26            fields: BTreeMap::new(),
27        }
28    }
29
30    /// Um logger igual a este, mais um campo.
31    #[must_use]
32    pub fn with_field(&self, key: &str, value: impl Into<String>) -> Self {
33        let mut fields = self.fields.clone();
34        fields.insert(key.to_owned(), value.into());
35
36        Self {
37            name: self.name.clone(),
38            fields,
39        }
40    }
41
42    /// Registra um evento de rotina.
43    pub fn info(&self, message: &str) {
44        tracing::info!(component = %self.name, fields = ?self.fields, "{message}");
45    }
46
47    /// Registra algo suspeito que não impediu a operação.
48    pub fn warn(&self, message: &str) {
49        tracing::warn!(component = %self.name, fields = ?self.fields, "{message}");
50    }
51
52    /// Registra uma falha.
53    pub fn error(&self, message: &str) {
54        tracing::error!(component = %self.name, fields = ?self.fields, "{message}");
55    }
56
57    /// O nome do componente.
58    pub fn name(&self) -> &str {
59        &self.name
60    }
61}

63#[cfg(test)]
64mod tests {
65
66    use crate::logging::interno::tracing_logger_factory::TracingLoggerFactory;
67    use crate::logging::LoggerFactory;
68    use pretty_assertions::assert_eq;
69
70    #[test]
71    fn o_logger_carrega_o_nome_do_componente() {
72        let logger = TracingLoggerFactory::new().create("auth");
73        assert_eq!(logger.name(), "auth");
74    }
75
76    /// É o que permite carimbar o `request_id` num escopo sem que ele vaze para
77    /// as demais requisições que compartilham o mesmo logger base.
78    #[test]
79    fn acrescentar_campo_nao_altera_o_logger_de_origem() {
80        let base = TracingLoggerFactory::new().create("http");
81        let scoped = base.with_field("request_id", "abc123");
82
83        assert!(base.fields.is_empty());
84        assert_eq!(
85            scoped.fields.get("request_id").map(String::as_str),
86            Some("abc123")
87        );
88    }
89
90    #[test]
91    fn campos_se_acumulam() {
92        let logger = TracingLoggerFactory::new()
93            .create("http")
94            .with_field("request_id", "abc")
95            .with_field("user_id", "U1");
96
97        assert_eq!(logger.fields.len(), 2);
98    }
