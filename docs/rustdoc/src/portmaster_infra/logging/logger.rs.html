1//! O contrato do logger que as camadas carregam.
2
3/// Um logger nomeado, com campos acumulados.
4///
5/// É uma trait e não uma struct porque nenhuma camada acima deve saber para
6/// onde o log vai. Hoje vai para o `tracing`; se amanhã for para um coletor, o
7/// `app` e as apresentações não mudam uma linha. É a mesma razão pela qual só a
8/// `infra` emite evento de log — e por que ninguém acima dela chama uma macro
9/// de log direto.
10///
11/// Não há `dyn` aqui: a impl é única e chega pelo
12/// [`LoggerFactory`](crate::logging::LoggerFactory) como `impl Logger`.
13///
14/// ## Os campos do instante são um array, não um logger novo
15///
16/// Cada método recebe `[(&str, &str); N]` com o que só vale para **aquela**
17/// linha. O `N` é constante, então o array vive na pilha e some na
18/// monomorfização; antes cada campo de ocasião custava um logger clonado e uma
19/// entrada num mapa, jogados fora na linha seguinte. Sem campo nenhum, é `[]`.
20pub trait Logger: Clone + Send + Sync + 'static {
21    /// Acrescenta um campo ao contexto da tarefa corrente.
22    ///
23    /// Não devolve logger novo, e não altera este: o campo não pertence ao
24    /// logger, pertence à **tarefa**. Fica guardado no span que o transporte
25    /// abriu, então toda linha emitida dali para baixo o carrega — inclusive as
26    /// que outro componente escrever com outro logger, que nunca soube do
27    /// assunto. Era o que carimbar o campo no logger não conseguia fazer: só
28    /// alcançava quem tivesse aquele logger em mãos.
29    ///
30    /// Fora de um span aberto o campo é descartado em silêncio. Não há onde
31    /// guardá-lo, e recusar a linha por isso seria pior do que perdê-lo.
32    fn with_field(&self, key: &str, value: impl Into<String>);
33
34    /// Registra um evento de rotina.
35    fn info<const N: usize>(&self, message: &str, fields: [(&str, &str); N]);
36
37    /// Registra algo suspeito que não impediu a operação.
38    fn warn<const N: usize>(&self, message: &str, fields: [(&str, &str); N]);
39
40    /// Registra uma falha.
41    fn error<const N: usize>(&self, message: &str, fields: [(&str, &str); N]);
42
43    /// Registra detalhe que só interessa quando se está investigando.
44    fn debug<const N: usize>(&self, message: &str, fields: [(&str, &str); N]);
45
46    /// O nome do componente que este logger identifica.
47    fn name(&self) -> &str;
48}
