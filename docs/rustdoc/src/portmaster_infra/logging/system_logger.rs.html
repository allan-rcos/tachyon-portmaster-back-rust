1//! O logger de sistema, alcançável de onde não há provider.
2
3use crate::logging::intern::tracing_logger::TracingLogger;
4use crate::logging::Logger;
5use std::sync::OnceLock;
6
7/// O nome que identifica as linhas emitidas sem um componente próprio.
8const NAME: &str = "system";
9
10/// A instância única, instalada no boot.
11static INSTANCE: OnceLock<TracingLogger> = OnceLock::new();
12
13/// O logger global do processo.
14///
15/// Existe para os pontos em que **não há** um construtor onde injetar um
16/// logger: o `panic::set_hook`, o middleware que recolhe um pânico, funções
17/// associadas. Fora daí, a regra continua sendo a de sempre — quem tem
18/// construtor recebe o [`Logger`] pela
19/// [`LoggerFactory`](crate::logging::LoggerFactory).
20///
21/// Ele não faz o `tracing` aparecer nas camadas de cima: continua sendo a
22/// `infra` que fala com o `tracing`, e o que sai daqui é a mesma abstração.
23pub struct SystemLogger;
24
25impl SystemLogger {
26    /// Fixa a instância do processo.
27    ///
28    /// Chamar duas vezes não é erro — a segunda é ignorada, porque um boot que
29    /// falha ao instalar o logger não é motivo para derrubar o processo.
30    pub fn install() {
31        let _ = INSTANCE.set(TracingLogger::new(NAME));
32    }
33
34    /// O logger global.
35    ///
36    /// Se ninguém instalou ainda, devolve um equivalente em vez de recusar. O
37    /// momento em que isto acontece — antes de o boot terminar — é justamente
38    /// quando uma linha de log tem mais valor, e perdê-la para uma questão de
39    /// ordem de inicialização seria o pior desfecho possível.
40    pub fn get() -> impl Logger {
41        INSTANCE.get_or_init(|| TracingLogger::new(NAME)).clone()
42    }
43}
