1//! A abertura do pool de conexões.
2//!
3//! Compartilhado por clone — o `Pool` é um ponteiro contado por dentro, então
4//! cada repositório carrega uma referência, não um pool próprio. É recurso de
5//! borda: único, externalizado, e impossível de monomorfizar por consumidor.
6//!
7//! Quem chama é o construtor da
8//! [`MariaDbUnitOfWork`](super::mariadb_unit_of_work::MariaDbUnitOfWork), e não
9//! o provider direto: o pool não tem uso fora do handle que o carrega, e um
10//! `open()` solto convidaria a que tivesse.
11//!
12//! Abrir não conecta. A primeira conexão sai da primeira consulta, ou do `ping`
13//! que o boot faz de propósito para não subir com um banco inalcançável.
14
15use std::path::PathBuf;

17use anyhow::{bail, Context};
18use mysql_async::{Opts, OptsBuilder, Pool, PoolConstraints, PoolOpts, SslOpts};
19use secrecy::ExposeSecret;
20
21use crate::config::pool::POOL_MAX_CONNECTIONS;
22use crate::config::{DatabaseSslMode, InfraSecrets};

24/// Abre o pool sem tocar a rede.
25///
26/// `Pool::new` não conecta: monta o pool e deixa a primeira conexão para a
27/// primeira consulta. É o que mantém esta função **síncrona**, e com ela toda a
28/// cadeia de factories acima — um handshake TCP aqui faria o `async` subir até o
29/// router.
30///
31/// Quem toca a rede é o
32/// [`ping`](super::mariadb_unit_of_work::MariaDbUnitOfWork::ping), passo
33/// explícito do boot: sem ele um erro de credencial só apareceria na primeira
34/// requisição, com o processo já reportado como saudável.
35///
36/// ## O fuso da sessão
37///
38/// O `init` roda em toda conexão nova do pool e fixa a sessão em UTC. Nenhuma
39/// consulta depende disso hoje: não há coluna de data — o tempo é `BIGINT` com
40/// epoch em milissegundos — e nenhuma função de tempo do servidor aparece no
41/// SQL. É rede, não mecanismo: o dia em que alguém escrever um `NOW()` num
42/// `SELECT`, ele responde em UTC como o resto da aplicação.
43///
44/// Vem depois do parse da URI de propósito: uma `?timezone=` na URI o
45/// sobrescreveria sem que nada avisasse.
46pub(super) fn open(secrets: &InfraSecrets) -> anyhow::Result<Pool> {
47    let parsed = Opts::from_url(secrets.database_uri.expose_secret())
48        .context("a URI do banco não está num formato que o driver entenda")?;
49
50    let opts = OptsBuilder::from_opts(parsed)
51        .init(vec!["SET time_zone = '+00:00'"])
52        .ssl_opts(ssl_opts(secrets)?)
53        .pool_opts(pool_opts()?);
54
55    Ok(Pool::new(opts))
56}

58/// Fixa o teto de conexões simultâneas.
59///
60/// O piso é zero, e não o padrão do driver: uma instalação pequena não deve
61/// segurar dez conexões ociosas contra um servidor que talvez nem as ofereça. O
62/// teto é decisão de arquitetura, escolhida por feature de compilação — ver
63/// [`POOL_MAX_CONNECTIONS`].
64fn pool_opts() -> anyhow::Result<PoolOpts> {
65    let max = usize::try_from(POOL_MAX_CONNECTIONS)
66        .context("o teto de conexões do pool não cabe nesta plataforma")?;
67
68    let constraints = PoolConstraints::new(0, max)
69        .with_context(|| format!("{max} não é um teto de conexões utilizável"))?;
70
71    Ok(PoolOpts::default().with_constraints(constraints))
72}

74/// Traduz o modo de TLS dos segredos nas opções do driver.
75///
76/// `verify_ca` sem o bundle da CA é recusado aqui: sem a cadeia não há o que
77/// validar, e cair em silêncio para um modo mais fraco entregaria exatamente a
78/// garantia que se pediu.
79///
80/// `None` desliga o TLS. `Required` cifra sem conferir a ponta — é o que
81/// significa pedir sigilo sem cadeia de confiança — e por isso aceita
82/// certificado inválido de propósito.
83fn ssl_opts(secrets: &InfraSecrets) -> anyhow::Result<Option<SslOpts>> {
84    match secrets.ssl_mode {
85        DatabaseSslMode::Disabled => Ok(None),
86        DatabaseSslMode::Required => Ok(Some(
87            SslOpts::default().with_danger_accept_invalid_certs(true),
88        )),
89        DatabaseSslMode::VerifyCa => {
90            let Some(ca_path) = secrets.ssl_ca_path.as_deref() else {
91                bail!("ssl_mode verify_ca exige o caminho do bundle da CA");
92            };
93
94            Ok(Some(
95                SslOpts::default()
96                    .with_root_certs(vec![PathBuf::from(ca_path).into()])
97                    .with_danger_skip_domain_validation(!secrets.ssl_verify_hostname),
98            ))
99        }
100    }
101}
