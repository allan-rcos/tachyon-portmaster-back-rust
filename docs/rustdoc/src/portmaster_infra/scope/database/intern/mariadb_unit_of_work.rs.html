13/// Monta o handle sobre um pool preguiçoso.
14///
15/// Síncrono: abrir o pool não toca a rede, e o que pode falhar aqui é o parse
16/// da URI e a checagem de TLS, que já eram síncronos. Quem confirma que o banco
17/// responde é o [`MariaDbUnitOfWork::ping`], no boot.
18///
19/// Devolve o tipo concreto porque o provider precisa guardá-lo num `static`, e
20/// um `static` exige um tipo com nome — um `impl Trait` não é um. Nem a função
21/// nem o tipo saem do módulo `scope::database`.
22pub(in crate::scope::database) fn mariadb_unit_of_work(
23    secrets: &InfraSecrets,
24) -> anyhow::Result<MariaDbUnitOfWork> {
25    Ok(MariaDbUnitOfWork {
26        pool: open(secrets)?,
27    })
28}
29
30/// O acesso ao banco que cada repositório carrega.
31///
32/// Sem estado por tarefa: o que ele guarda é o pool, que vale para o processo
33/// inteiro. O estado da tarefa mora no [`MariaDbContext`], e este handle o
34/// alcança pelo mapa do escopo — nunca ao contrário.
35#[derive(Clone)]
36pub(in crate::scope::database) struct MariaDbUnitOfWork {
37    /// De onde a transação sai quando a primeira escrita a pede.
38    pool: MySqlPool,
39}
40
41impl MariaDbUnitOfWork {
42    /// Confirma que o banco responde, e derruba o boot se não responder.
43    ///
44    /// É a primeira conexão de verdade do processo: o pool nasce preguiçoso, e
45    /// até aqui nada tocou a rede. Falhar é o desfecho desejado — subir com um
46    /// banco inalcançável só adia a descoberta para a primeira requisição, com
47    /// o processo já reportado como saudável.
48    pub(in crate::scope::database) async fn ping(&self) -> anyhow::Result<()> {
49        sqlx::query("SELECT 1")
50            .execute(&self.pool)
51            .await
52            .context("o MariaDB não respondeu ao SELECT de sanidade")?;
53
54        Ok(())
55    }
56}
57
58impl MySqlTransaction for MariaDbUnitOfWork {
59    async fn transaction(&self) -> anyhow::Result<TransactionGuard> {
60        let slot = ScopeSlots::current::<MariaDbContext>()?.slot();
61        let mut guard = slot.lock_owned().await;
62
63        if guard.is_none() {
64            *guard = Some(
65                self.pool
66                    .begin()
67                    .await
68                    .context("falha ao abrir transação no MariaDB")?,
69            );
70        }
71
72        OwnedMutexGuard::try_map(guard, Option::as_mut)
73            .map_err(|_| anyhow!("a transação sumiu do escopo entre abrir e emprestar"))
74    }
75}
