1//! O que um contexto por tarefa sabe fazer.
2
3use std::any::Any;
4use std::future::Future;
5use std::pin::Pin;
6
7/// O futuro de fechar um contexto.
8///
9/// Boxeado porque a lista de contextos é heterogênea: o `Arc<dyn ScopeContext>`
10/// que o iterador percorre precisa de um tipo de retorno só, e `async fn` numa
11/// trait dá um futuro opaco diferente por implementação. `pub(super)` para não
12/// competir com o export do arquivo — ele é auxiliar, não contrato.
13pub(super) type Closing<'a> = Pin<Box<dyn Future<Output = anyhow::Result<()>> + Send + 'a>>;
14
15/// Um estado que nasce e morre com a tarefa, e que sabe confirmar-se.
16///
17/// É o que a camada põe no mapa do escopo: o mesmo objeto guarda o estado (a
18/// transação, o buffer de escritas) e sabe fechá-lo. Não há `begin` aqui de
19/// propósito — abrir é a única operação que precisa do recurso caro, e o objeto
20/// que o `install` constrói nasce de um `fn` resolvido pelo linker, sem provider
21/// de onde tirá-lo. Quem abre é o handle injetado nos repositórios.
22///
23/// `Any` é o que permite ao dono recuperar o próprio contexto com o tipo de
24/// volta, depois de ele ter entrado na lista como `dyn`.
25pub(crate) trait ScopeContext: Any + Send + Sync {
26    /// Confirma o que a tarefa acumulou.
27    ///
28    /// Sem nada acumulado é no-op: um caso de uso que só leu não abriu
29    /// transação, e confirmar o nada não é erro.
30    fn commit(&self) -> Closing<'_>;
31
32    /// Descarta o que a tarefa acumulou.
33    ///
34    /// Idempotente, e no-op depois do commit — o escopo o chama na saída sem
35    /// saber se o corpo confirmou.
36    fn rollback(&self) -> Closing<'_>;
37}
