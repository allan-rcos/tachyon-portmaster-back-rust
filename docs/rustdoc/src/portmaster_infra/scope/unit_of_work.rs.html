1//! O contrato da unidade de trabalho.
2
3/// Fecha o que a tarefa acumulou — e nada mais.
4///
5/// Não executa consulta: o que ela oferece é fim e desistência. Manter a
6/// execução fora daqui é o que impede o `app` de rodar SQL.
7///
8/// Não há `begin`. Abrir é a única das três operações que precisa do recurso
9/// caro, e quem tem o pool em mãos é o handle injetado nos repositórios, não o
10/// objeto que o escopo monta a partir da slice. Então abrir virou detalhe de
11/// quem escreve: o primeiro repositório a pedir a transação a abre, e um caso de
12/// uso que só leu da memória nunca abriu nenhuma.
13#[trait_variant::make(Send)]
14pub trait UnitOfWork {
15    /// Confirma o que foi feito.
16    async fn commit(&self) -> anyhow::Result<()>;
17
18    /// Descarta o que foi feito.
19    ///
20    /// Idempotente: chamar sem nada aberto não é erro, porque o caminho de falha
21    /// frequentemente não sabe se chegou a abrir alguma coisa.
22    async fn rollback(&self) -> anyhow::Result<()>;
23}
