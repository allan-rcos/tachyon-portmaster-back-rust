1//! O que a tarefa escreveu em memória e ainda não publicou.
2
3use std::collections::BTreeSet;
4use std::sync::Arc;
5
6use tokio::sync::Mutex;
7
8use crate::scope::intern::scope_slots::ScopeSlots;
9use crate::scope::memory::intern::pending_write::{Operation, PendingWrite};
10use crate::scope::scope_context::{Closing, ScopeContext};
11
12/// As escritas em memória da tarefa corrente.
13///
14/// Existe para que uma escrita em memória feita dentro de um caso de uso siga a
15/// mesma sorte da escrita em banco: se o SQL não confirmar, o cache não publica.
16/// Sem isto, um erro no meio da transação deixaria memória e banco discordando,
17/// e o processo continuaria de pé servindo a discordância.
18pub(crate) struct MokaContext {
19    /// A fila, na ordem em que a tarefa escreveu.
20    pending: Mutex<Vec<PendingWrite>>,
21}
22
23impl MokaContext {
24    /// Registra o contexto da memória na tarefa.
25    pub(super) fn install(slots: &mut ScopeSlots) {
26        slots.put(Self {
27            pending: Mutex::default(),
28        });
29    }
30
31    /// Enfileira uma escrita da tarefa.
32    pub(super) async fn enqueue(&self, write: PendingWrite) {
33        self.pending.lock().await.push(write);
34    }
35
36    /// O que a própria tarefa já escreveu nesta entrada.
37    ///
38    /// `None` significa "a fila não tem nada a dizer, pergunte ao cache";
39    /// `Some(None)`, que a tarefa apagou a entrada. Sem esta consulta, gravar e
40    /// ler na mesma tarefa devolveria o valor anterior — e nenhum repositório do
41    /// `MariaDB` se comporta assim.
42    pub(super) async fn lookup(
43        &self,
44        store: &str,
45        group: &str,
46        key: &str,
47    ) -> Option<Option<Arc<Vec<u8>>>> {
48        let pending = self.pending.lock().await;
49
50        pending
51            .iter()
52            .rev()
53            .filter(|write| write.store == store && write.group == group)
54            .find_map(|write| match &write.operation {
55                Operation::Put {
56                    key: written,
57                    value,
58                    ..
59                } if written == key => Some(Some(Arc::clone(value))),
60                Operation::Put { .. } => None,
61                Operation::ClearGroup => Some(None),
62            })
63    }
64
65    /// Aplica sobre `keys` o que a tarefa acrescentou ou tirou do grupo.
66    pub(super) async fn overlay(&self, store: &str, group: &str, keys: &mut BTreeSet<String>) {
67        let pending = self.pending.lock().await;
68
69        for write in pending
70            .iter()
71            .filter(|write| write.store == store && write.group == group)
72        {
73            match &write.operation {
74                Operation::Put { key, .. } => {
75                    keys.insert(key.clone());
76                }
77                Operation::ClearGroup => keys.clear(),
78            }
79        }
80    }
81}
82
83impl ScopeContext for MokaContext {
84    fn commit(&self) -> Closing<'_> {
85        Box::pin(async {
86            let pending = std::mem::take(&mut *self.pending.lock().await);
87
88            for write in pending {
89                write.apply().await?;
90            }
91
92            Ok(())
93        })
94    }
95
96    /// Descarta o que a tarefa escreveu.
97    ///
98    /// Barato porque nada foi publicado: a fila é a única cópia, e soltá-la é o
99    /// desfazer inteiro.
100    fn rollback(&self) -> Closing<'_> {
101        Box::pin(async {
102            self.pending.lock().await.clear();
103            Ok(())
104        })
105    }
106}
