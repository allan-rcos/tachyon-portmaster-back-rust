1//! Como um repositório em memória grava e lê.
2
3use std::sync::Arc;
4use std::time::Duration;
5
6/// Um armazenamento em memória, sobre bytes.
7///
8/// A chave tem duas partes porque um store guarda mais de um recorte: o
9/// **grupo** é o namespace (a permissão, o grupo de marcador, o recurso cuja
10/// listagem foi cacheada) e a **chave** identifica a entrada dentro dele. É o
11/// que faz `clear_group` ser um recorte exato em vez de uma varredura por
12/// prefixo, que casaria `container:` com `container_summary:`.
13///
14/// Fala bytes, e não objeto de domínio, pela mesma razão que o `MySqlTransaction`
15/// fala linha: quem traduz é o repositório. Aqui a tradução é serialização em
16/// vez de coluna, e é a única diferença entre um repositório destes e um do
17/// `MariaDB`.
18
19#[trait_variant::make(Send)]
20pub(crate) trait MemoryStore {
21    /// O valor da entrada, se ela existir e não tiver expirado.
22    ///
23    /// Dentro de um escopo, o que a própria tarefa escreveu vem antes do que o
24    /// cache tem: sem isso, gravar e ler na mesma tarefa devolveria o valor
25    /// anterior, e nenhum repositório do `MariaDB` se comporta assim.
26    async fn get(&self, group: &str, key: &str) -> anyhow::Result<Option<Arc<Vec<u8>>>>;
27
28    /// Grava a entrada, com o prazo dela.
29    ///
30    /// `ttl` ausente significa "vive enquanto o processo viver" — é o caso do
31    /// catálogo de permissões, que o boot preenche e ninguém mais altera. O
32    /// prazo viaja com o valor porque cada marcador vence no seu próprio tempo,
33    /// não num vencimento global.
34    async fn put(
35        &self,
36        group: &str,
37        key: &str,
38        value: Vec<u8>,
39        ttl: Option<Duration>,
40    ) -> anyhow::Result<()>;
41
42    /// Descarta o grupo inteiro.
43    ///
44    /// Sem varredura por prefixo: o grupo já é o recorte, e comparar prefixo
45    /// casaria `container` com `container_summary`.
46    async fn clear_group(&self, group: &str) -> anyhow::Result<()>;
47
48    /// As chaves vivas de um grupo, em ordem.
49    ///
50    /// Ordenado porque a iteração do Moka não tem ordem definida, e uma listagem
51    /// que muda de ordem a cada chamada é ruim de ler e pior de testar.
52    async fn keys(&self, group: &str) -> anyhow::Result<Vec<String>>;
53}
