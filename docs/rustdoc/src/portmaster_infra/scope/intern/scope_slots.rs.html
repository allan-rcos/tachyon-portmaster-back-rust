1//! O mapa de contextos da tarefa corrente.
2//!
3//! Há **um** task-local no sistema, e é este. É o que mata o aninhamento: por
4//! mais contextos que existam, o escopo faz uma única chamada de
5//! `LocalKey::scope`, e não um nível de indentação por participante.
6//!
7//! > **Lacuna conhecida do `lint-exports`:** o `CURRENT` abaixo nasce de
8//! > `tokio::task_local!`, e itens gerados por macro são invisíveis ao `syn`.
9//! > O arquivo exporta dois itens na prática, não um — o `ScopeSlots` e o
10//! > task-local que só ele usa.
11
12use std::any::{Any, TypeId};
13use std::future::Future;
14use std::sync::Arc;
15
16use anyhow::anyhow;
17
18use crate::scope::scope_context::ScopeContext;
19
20tokio::task_local! {
21    /// Os contextos da tarefa corrente.
22    ///
23    /// Interno à `infra`: nem o `app` nem a apresentação alcançam este
24    /// armazenamento. Se alcançassem, uma camada passaria a depender do formato
25    /// interno da outra, e uma feature escrita só no `api` poderia esquecer de
26    /// alimentá-lo — com o erro aparecendo em produção.
27    pub(crate) static CURRENT: Arc<ScopeSlots>;
28}
29
30/// Os contextos que um escopo instalou, indexados de duas formas.
31///
32/// O mesmo `Arc` entra nas duas listas: por tipo, para o dono recuperar o
33/// contexto dele com o tipo de volta; e em ordem de instalação, para a unidade
34/// de trabalho percorrer todos sem saber de nenhum.
35#[derive(Default)]
36pub(crate) struct ScopeSlots {
37    /// Por tipo, para quem quer o **seu** contexto.
38    by_type: Vec<(TypeId, Arc<dyn Any + Send + Sync>)>,
39    /// Em ordem de instalação, para quem confirma **todos**.
40    all: Vec<Arc<dyn ScopeContext>>,
41}
42
43impl ScopeSlots {
44    /// Registra o contexto de uma camada.
45    pub(crate) fn put<T: ScopeContext>(&mut self, context: T) {
46        let context = Arc::new(context);
47
48        self.by_type.push((TypeId::of::<T>(), context.clone()));
49        self.all.push(context);
50    }
51
52    /// Os contextos instalados, para quem confirma.
53    pub(crate) fn contexts(&self) -> Vec<Arc<dyn ScopeContext>> {
54        self.all.clone()
55    }
56
57    /// Roda `future` com estes contextos instalados na tarefa.
58    pub(crate) async fn install<F: Future>(self, future: F) -> F::Output {
59        CURRENT.scope(Arc::new(self), future).await
60    }
61
62    /// O contexto de `T` na tarefa corrente.
63    ///
64    /// Falha fora de um escopo em vez de devolver algo inútil, e falha de novo
65    /// se o escopo existir sem esta camada — o que só acontece se alguém
66    /// declarar o contexto e esquecer o `#[distributed_slice]`.
67    pub(crate) fn current<T: ScopeContext>() -> anyhow::Result<Arc<T>> {
68        let slots = CURRENT
69            .try_with(Arc::clone)
70            .map_err(|_| anyhow!("nenhum escopo de tarefa ativo"))?;
71
72        slots
73            .by_type
74            .iter()
75            .find(|(id, _)| *id == TypeId::of::<T>())
76            .and_then(|(_, context)| Arc::clone(context).downcast::<T>().ok())
77            .ok_or_else(|| anyhow!("a camada não foi instalada neste escopo"))
78    }
79
80    /// Se há escopo ativo.
81    ///
82    /// Não diz se há transação **aberta** — só se existe onde guardá-la.
83    pub(crate) fn is_active() -> bool {
84        CURRENT.try_with(|_| ()).is_ok()
85    }
86}
