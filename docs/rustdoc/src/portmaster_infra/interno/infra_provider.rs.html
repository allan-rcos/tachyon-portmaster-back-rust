40/// A implementação do provider. Privada: nenhum crate exporta impl.
41pub(crate) struct InfraProviderImpl {
42    /// O pool de conexões, compartilhado por clone — é um `Arc` por dentro.
43    pool: MySqlPool,
44    /// O catálogo de permissões em memória, preenchido no boot.
45    permission_cache: PermissionCache,
46    /// Os grupos de marcador declarados.
47    marker_group_cache: MarkerGroupCache,
48    /// Os marcadores em si, com TTL — é onde o refresh token vive.
49    marker_cache: MarkerCache,
50    /// O cache de leitura, por prefixo de chave.
51    read_cache: ReadCacheStore,
52}
53
54impl InfraProviderImpl {
55    /// Cria os recursos base: o pool já veio pronto, os quatro caches nascem
56    /// aqui e são compartilhados por clone pelo resto da vida do processo.
57    ///
58    /// ## Por que permissão e grupo de marcador têm mapas separados
59    ///
60    /// São vocabulários diferentes. Compartilhar um mapa só faria o grupo
61    /// `refresh-token` aparecer na listagem de permissões — e, pior, ser
62    /// concedido ao papel que o `POST /setup` cria, que recebe tudo que
63    /// estiver registrado. O schema anterior os separava em duas tabelas pelo
64    /// mesmo motivo.
65    ///
66    /// ## Por que o cache de leitura liga os closures de invalidação
67    ///
68    /// A invalidação por prefixo depende deles: sem os closures, uma escrita
69    /// não teria como derrubar todas as listagens afetadas.
70    pub(crate) fn new(pool: MySqlPool) -> Self {
71        Self {
72            pool,
73            permission_cache: PermissionCache::new(CacheLimits::METADATA_CACHE_CAPACITY),
74            marker_group_cache: MarkerGroupCache::new(CacheLimits::METADATA_CACHE_CAPACITY),
75            marker_cache: MarkerCache::new(CacheLimits::MARKER_CACHE_CAPACITY),
76            read_cache: Arc::new(
77                Cache::builder()
78                    .max_capacity(CacheLimits::READ_CACHE_CAPACITY)
79                    .time_to_live(Duration::from_secs(CacheLimits::READ_CACHE_TTL_SECONDS))
80                    .support_invalidation_closures()
81                    .build(),
82            ),
83        }
84    }
85}
