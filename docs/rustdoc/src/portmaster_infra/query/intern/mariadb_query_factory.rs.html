22/// A implementação da fábrica de DQLs.
23#[derive(Clone)]
24pub(crate) struct MariadbQueryFactory;
25
26impl MariadbQueryFactory {
27    /// Monta a fábrica.
28    pub(crate) const fn new() -> Self {
29        Self
30    }
31}
32
33impl QueryFactory for MariadbQueryFactory {
34    fn get_account(
35        &self,
36        user_id: &str,
37    ) -> anyhow::Result<impl SqlDql<View = Option<AccountView>>> {
38        Ok(GetAccountDql::new(Codec::decode_id(user_id)?))
39    }
40
41    fn get_container(
42        &self,
43        id: &str,
44    ) -> anyhow::Result<impl SqlDql<View = Option<ContainerViewItem>>> {
45        Ok(GetContainerDql::new(Codec::decode_id(id)?))
46    }
47
48    fn get_product(&self, id: &str) -> anyhow::Result<impl SqlDql<View = Option<ProductViewItem>>> {
49        Ok(GetProductDql::new(Codec::decode_id(id)?))
50    }
51
52    fn get_role(&self, id: &str) -> anyhow::Result<impl SqlDql<View = Option<RoleViewItem>>> {
53        Ok(GetRoleDql::new(Codec::decode_id(id)?))
54    }
55
56    fn list_containers(
57        &self,
58        params: ContainerListParams,
59    ) -> impl SqlDql<View = ContainerListView> {
60        ListContainersDql::new(params)
61    }
62
63    fn list_container_summaries(
64        &self,
65        params: SummaryListParams,
66    ) -> anyhow::Result<impl SqlDql<View = ContainerSummaryListView>> {
67        let id = params.id.as_deref().map(Codec::decode_id).transpose()?;
68
69        Ok(ListContainerSummariesDql::new(params, id))
70    }
71
72    fn list_products(&self, params: ListParams) -> impl SqlDql<View = ProductListView> {
73        ListProductsDql::new(params)
74    }
75
76    fn list_roles(&self, params: ListParams) -> impl SqlDql<View = RoleListView> {
77        ListRolesDql::new(params)
78    }
79
80    fn list_users(&self, params: UserListParams) -> impl SqlDql<View = UserListView> {
81        ListUsersDql::new(params)
82    }
83
84    fn metrics(&self) -> impl SqlDql<View = MetricsView> {
85        MetricsDql::new()
86    }
87}
