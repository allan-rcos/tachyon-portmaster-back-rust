1//! A execução de consultas sobre o `MariaDB`.
2
3use anyhow::Context;
4
5use crate::query::{QueryRepository, SqlDql};
6use crate::scope::database::mysql_transaction::MySqlTransaction;
7
8/// Monta o executor de consultas sobre o `MariaDB`.
9///
10/// O que ele guarda é o handle do banco; a transação em si vem do escopo da
11/// tarefa, o que permite ao provider reconstruí-lo a cada chamada por custo
12/// nenhum.
13pub(crate) fn mariadb_query_repository<T>(
14    transactions: T,
15) -> impl QueryRepository + Sync + Clone + use<T> + 'static
16where
17    T: MySqlTransaction + Send + Sync + Clone + 'static,
18{
19    MariadbQueryRepository { transactions }
20}
21
22/// A implementação sobre `MariaDB`.
23#[derive(Clone)]
24struct MariadbQueryRepository<T> {
25    /// De onde a transação da tarefa vem.
26    transactions: T,
27}
28
29impl<T: MySqlTransaction + Send + Sync> QueryRepository for MariadbQueryRepository<T> {
30    /// Executa o DQL e entrega as linhas para ele hidratar.
31    ///
32    /// Não há `AssertSqlSafe` no caminho, e é por construção: o `QueryBuilder`
33    /// do sqlx separa o texto do valor no próprio tipo, então não existe ponto
34    /// onde um valor pudesse ser interpolado no SQL. Um construtor nosso daria a
35    /// mesma garantia como afirmação, e ela valeria só enquanto ninguém
36    /// escrevesse a linha errada dentro dele.
37    ///
38    /// O `MariaDB` compara número com texto descartando o índice, então os
39    /// valores continuam ligados pelo tipo que têm — o que agora é o
40    /// `push_bind` de quem monta a consulta, e não uma tradução aqui.
41    async fn run<D: SqlDql>(&self, dql: D) -> anyhow::Result<D::View> {
42        let mut builder = dql.build();
43        let sql = builder.sql().as_str().to_owned();
44        let mut transaction = self.transactions.transaction().await?;
45
46        let rows = builder
47            .build()
48            .fetch_all(&mut **transaction)
49            .await
50            .with_context(|| format!("falha ao executar a consulta de leitura: {sql}"))?;
51
52        dql.read(rows)
53    }
54}
