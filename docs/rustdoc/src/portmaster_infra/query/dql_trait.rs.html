1//! O contrato de uma consulta de leitura.
2
3/// Uma consulta de leitura, independente de backend.
4///
5/// Amarra só o tipo de saída. É o que permite ao [`crate::query::query_repository::QueryRepository`] devolver
6/// `D::View` sem saber o que `View` é.
7pub trait Dql {
8    /// O read model que esta consulta produz.
9    ///
10    /// `Send` porque a View atravessa o `.await` da execução e sai por uma
11    /// fronteira de tarefa — o handler que a pediu pode estar em outra thread.
12    type View: Send;
13
14    /// O que distingue esta consulta de outra do mesmo tipo.
15    ///
16    /// É a identidade da consulta para efeito de cache, e quem a escreve é o
17    /// DQL porque é ele quem sabe o que a consulta é. Antes cada serviço do
18    /// `app` montava a chave à mão, com um prefixo por serviço e a lembrança de
19    /// incluir cada filtro — e um filtro novo que ninguém somasse à chave fazia
20    /// duas consultas diferentes lerem a mesma entrada.
21    ///
22    /// Não sai do SQL: o `QueryBuilder` separa o texto dos valores, e o texto
23    /// sozinho é idêntico entre duas páginas do mesmo `SELECT`.
24    fn cache_key(&self) -> String;
25}
