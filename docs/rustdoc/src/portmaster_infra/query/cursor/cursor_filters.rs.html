1//! O conjunto de filtros sob os quais um cursor vale.
2
3use std::collections::BTreeMap;
4
5use serde::{Deserialize, Serialize};
6
7/// Os filtros sob os quais um cursor foi emitido.
8///
9/// `BTreeMap` por dentro, e não `HashMap`: a serialização precisa ser estável
10/// para que o mesmo conjunto de filtros produza sempre o mesmo JSON. Com ordem
11/// de iteração aleatória, dois cursores idênticos poderiam não se reconhecer.
12///
13/// É um tipo próprio e não um alias porque o construtor pertence a ele — no
14/// molde do `Base62`, o nome do tipo é o namespace da operação. `#[serde(transparent)]`
15/// mantém o formato do token idêntico ao que o alias produzia.
16#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
17#[serde(transparent)]
18pub(crate) struct CursorFilters(BTreeMap<String, String>);
19
20impl CursorFilters {
21    /// Monta o mapa de filtros a partir de pares nomeados.
22    ///
23    /// Os valores viram texto porque o mapa existe só para ser **comparado**
24    /// inteiro, nunca lido de volta como número. Um `None` entra como string
25    /// vazia para que "sem filtro" seja um valor tão explícito quanto qualquer
26    /// outro — se a chave sumisse, dois filtros diferentes poderiam gerar o
27    /// mesmo mapa.
28    pub(crate) fn of<const N: usize>(pairs: [(&str, String); N]) -> Self {
29        Self(
30            pairs
31                .into_iter()
32                .map(|(key, value)| (key.to_owned(), value))
33                .collect(),
34        )
35    }
36}
