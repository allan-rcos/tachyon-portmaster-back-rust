1//! O cursor de paginação keyset.
2
3use crate::query::cursor::CursorFilters;
4use base64::engine::general_purpose::URL_SAFE_NO_PAD;
5use base64::Engine as _;
6use serde::{Deserialize, Serialize};
7
8/// O conteúdo do token, como ele viaja.
9#[derive(Serialize, Deserialize)]
10struct Payload {
11    /// Id da última linha da página servida.
12    id: i64,
13    /// Os filtros sob os quais o token vale.
14    f: CursorFilters,
15}
16
17/// Um cursor keyset.
18#[derive(Debug, Clone, PartialEq, Eq)]
19pub struct Cursor {
20    /// A página seguinte começa estritamente depois deste id.
21    pub(crate) last_id: i64,
22}
23
24impl Cursor {
25    /// Emite o token da próxima página.
26    /// Emite o token da próxima página.
27    ///
28    /// A serialização de um `BTreeMap<String, String>` com um `i64` não tem
29    /// caminho de falha; se um dia tivesse, um token vazio faria a paginação
30    /// recomeçar, que é degradação aceitável para uma consulta de leitura.
31    pub(crate) fn encode(last_id: i64, filters: &CursorFilters) -> String {
32        let payload = Payload {
33            id: last_id,
34            f: filters.clone(),
35        };
36
37        let json = serde_json::to_vec(&payload).unwrap_or_default();
38
39        URL_SAFE_NO_PAD.encode(json)
40    }
41
42    /// Lê um token, ou devolve `None` para recomeçar do princípio.
43    ///
44    /// Toda recusa é o mesmo `None`: token ausente, token vazio, texto que não é
45    /// base64url, JSON que não tem a forma esperada, e token emitido sob outros
46    /// filtros. Quem chama trata todos como "primeira página", e é por isso que
47    /// não existe erro de cursor.
48    /// Lê o token, se ele ainda descreve esta consulta.
49    ///
50    /// Filtros diferentes dos da emissão devolvem `None`: o token descreve uma
51    /// varredura que não é mais a desta requisição, e continuar dali entregaria
52    /// uma página do conjunto anterior.
53    pub(crate) fn decode(token: Option<&str>, current_filters: &CursorFilters) -> Option<Self> {
54        let token = token.filter(|t| !t.is_empty())?;
55
56        let json = URL_SAFE_NO_PAD.decode(token).ok()?;
57        let payload: Payload = serde_json::from_slice(&json).ok()?;
58
59        if &payload.f != current_filters {
60            return None;
61        }
62
63        Some(Self {
64            last_id: payload.id,
65        })
66    }
67
68    /// O id de onde partir, com `0` quando não há cursor válido.
69    ///
70    /// `0` funciona como origem porque todo id é um Snowflake positivo.
71    pub(crate) fn last_id_or_start(token: Option<&str>, current_filters: &CursorFilters) -> i64 {
72        Self::decode(token, current_filters).map_or(0, |cursor| cursor.last_id)
73    }
74
75    /// O token da próxima página — ou `None` quando esta foi a última.
76    ///
77    /// Só emite quando a página veio **cheia**: uma página menor que o limite
78    /// significa que a varredura acabou, e devolver cursor ali levaria o cliente
79    /// a uma requisição a mais só para receber nada.
80    pub(crate) fn next(
81        served: usize,
82        limit: u32,
83        last_id: i64,
84        filters: &CursorFilters,
85    ) -> Option<String> {
86        (served == limit as usize && last_id > 0).then(|| Self::encode(last_id, filters))
87    }
88}
