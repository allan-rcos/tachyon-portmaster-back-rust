9/// A implementação sobre `MariaDB`.
10///
11/// Sem estado: a transação vem do escopo da requisição, o que permite ao
12/// provider reconstruí-la a cada chamada por custo nenhum.
13pub(crate) struct MariadbQueryRepository;
14
15impl MariadbQueryRepository {
16    /// Monta o repositório.
17    pub(crate) const fn new() -> Self {
18        Self
19    }
20}
21
22impl QueryRepository for MariadbQueryRepository {
23    /// Executa o DQL e entrega as linhas para ele hidratar.
24    ///
25    /// ## Por que `AssertSqlSafe` é legítimo aqui
26    ///
27    /// É o que o sqlx 0.9 exige de todo SQL montado em tempo de execução, e a
28    /// exigência é justa: o texto não é mais uma constante que se lê no
29    /// arquivo. A afirmação se sustenta porque o `Select` **nunca interpola
30    /// valor nenhum** — tudo que vem de fora entra como `Bind`, e o único
31    /// trecho de texto variável são placeholders `?` contados a partir do
32    /// tamanho de um `Vec`.
33    ///
34    /// ## Por que os binds são ligados por tipo
35    ///
36    /// Um id enviado como string faz o `MariaDB` comparar número com texto e
37    /// descartar o índice.
38    async fn run<D: SqlDql>(&self, dql: D) -> anyhow::Result<D::View> {
39        let SqlQuery { sql, binds } = dql.build();
40        let mut transaction = MariadbUnitOfWork::current().await?;
41
42        let mut query = sqlx::query(sqlx::AssertSqlSafe(sql.clone()));
43        for bind in binds {
44            query = match bind {
45                Bind::Int(value) => query.bind(value),
46                Bind::Text(value) => query.bind(value),
47            };
48        }
49
50        let rows = query
51            .fetch_all(&mut **transaction.as_mut())
52            .await
53            .with_context(|| format!("falha ao executar a consulta de leitura: {sql}"))?;
54
55        dql.read(rows)
56    }
57}
