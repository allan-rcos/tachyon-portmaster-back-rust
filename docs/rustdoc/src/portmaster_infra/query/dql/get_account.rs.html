1//! A consulta de um usuário com os papéis dele.
2
3use anyhow::Context as _;
4use sqlx::mysql::{MySql, MySqlRow};
5use sqlx::{QueryBuilder, Row as _};
6
7use crate::entity::codec::Codec;
8use crate::query::dql::list_roles::read_item as read_role;
9use crate::query::views::{AccountView, RoleViewItem};
10use crate::query::{Dql, SqlDql};
11
12/// A projeção comum às duas consultas de conta.
13///
14/// Usuário e papel na mesma linha, com o papel prefixado porque `id` e `name`
15/// existem dos dois lados.
16pub(super) const COLUMNS: &str = "u.id AS user_id, u.name AS user_name, u.email AS user_email, \
17     r.id AS role_id, r.name AS role_name, r.permissions AS role_permissions, \
18     (SELECT COUNT(*) FROM user_roles urc WHERE urc.role_id = r.id) AS role_user_count";
19
20/// A ligação até os papéis.
21///
22/// O `deleted_at` do papel mora na condição do `JOIN`, não no `WHERE`. Movido
23/// para o `WHERE`, ele descartaria a linha inteira quando não houvesse papel — o
24/// `LEFT JOIN` viraria `INNER`, e um usuário sem papel nenhum sumiria da
25/// listagem em vez de aparecer com a lista vazia.
26pub(super) const JOIN_ROLES: &str = "LEFT JOIN user_roles ur ON ur.user_id = u.id \
27                                     LEFT JOIN roles r ON r.id = ur.role_id AND r.deleted_at IS NULL";
28
29/// Um usuário pelo id, com os papéis.
30pub fn get_account(id: &str) -> anyhow::Result<impl SqlDql<View = Option<AccountView>>> {
31    Ok(GetAccount {
32        user_id: Codec::decode_id(id)?,
33    })
34}
35
36/// O papel de uma linha, ou `None` no lado vazio do `LEFT JOIN`.
37///
38/// `role_id` nulo é a marca de "este usuário não tem papel": o join não achou
39/// par, e as demais colunas de papel vêm nulas junto.
40pub(super) fn read_role_of(row: &MySqlRow) -> anyhow::Result<Option<RoleViewItem>> {
41    let role_id: Option<i64> = row
42        .try_get("role_id")
43        .context("coluna `role_id` não veio como inteiro opcional")?;
44
45    if role_id.is_none() {
46        return Ok(None);
47    }
48
49    read_role(row, "role_").map(Some)
50}
51
52/// Um usuário pelo id, com os papéis.
53struct GetAccount {
54    /// O alvo, como `BIGINT`.
55    user_id: i64,
56}
57
58impl Dql for GetAccount {
59    type View = Option<AccountView>;
60
61    fn cache_key(&self) -> String {
62        format!("get_account:{}", self.user_id)
63    }
64}
65
66impl SqlDql for GetAccount {
67    fn build(&self) -> QueryBuilder<MySql> {
68        let mut builder = QueryBuilder::new("SELECT ");
69        builder.push(COLUMNS);
70        builder.push(" FROM users u ");
71        builder.push(JOIN_ROLES);
72        builder.push(" WHERE u.id = ");
73        builder.push_bind(self.user_id);
74        builder.push(" AND u.deleted_at IS NULL ORDER BY r.id ASC");
75
76        builder
77    }
78
79    /// Sem linha nenhuma o usuário não existe.
80    ///
81    /// Com linhas, ele existe mesmo que todas tenham `role_id` nulo — é o
82    /// usuário sem papel.
83    fn read(&self, rows: Vec<MySqlRow>) -> anyhow::Result<Self::View> {
84        let Some(first) = rows.first() else {
85            return Ok(None);
86        };
87
88        let id: i64 = first
89            .try_get("user_id")
90            .context("coluna `user_id` não veio como inteiro")?;
91
92        let mut roles = Vec::with_capacity(rows.len());
93        for row in &rows {
94            if let Some(role) = read_role_of(row)? {
95                roles.push(role);
96            }
97        }
98
99        Ok(Some(AccountView {
100            id: Codec::encode_id(id),
101            name: first
102                .try_get("user_name")
103                .context("coluna `user_name` não veio como texto")?,
104            email: first
105                .try_get("user_email")
106                .context("coluna `user_email` não veio como texto")?,
107            roles,
108        }))
109    }
110}
