1//! A consulta de um usuário com os papéis dele.
2
3use crate::query::dql::account_reader::AccountReader;
4use crate::query::row::Row;
5use crate::query::sql::{Bind, Select, SqlQuery};
6use crate::query::views::AccountView;
7use crate::query::{Dql, SqlDql};
8use sqlx::mysql::MySqlRow;
9
10/// A projeção comum às duas consultas.
11///
12/// Usuário e papel na mesma linha, com o papel prefixado porque `id` e `name`
13/// existem dos dois lados.
14const COLUMNS: &str = "u.id AS user_id, u.name AS user_name, u.email AS user_email, \
15                       r.id AS role_id, r.name AS role_name, r.permissions AS role_permissions, \
16                       (SELECT COUNT(*) FROM user_roles urc WHERE urc.role_id = r.id) AS role_user_count";
17
18/// A ligação até os papéis.
19///
20/// O `deleted_at` do papel mora na condição do `JOIN`, não no `WHERE`. Movido
21/// para o `WHERE`, ele descartaria a linha inteira quando não houvesse papel — o
22/// `LEFT JOIN` viraria `INNER`, e um usuário sem papel nenhum sumiria da
23/// listagem em vez de aparecer com a lista vazia.
24const JOIN_ROLES: &str = "LEFT JOIN user_roles ur ON ur.user_id = u.id \
25                          LEFT JOIN roles r ON r.id = ur.role_id AND r.deleted_at IS NULL";
26
27/// Um usuário pelo id, com os papéis.
28pub struct GetAccountDql {
29    /// O alvo, como `BIGINT` — o base62 já foi decodificado pelo caso de uso.
30    user_id: i64,
31}
32
33impl GetAccountDql {
34    /// Monta a consulta.
35    pub(crate) const fn new(user_id: i64) -> Self {
36        Self { user_id }
37    }
38}
39
40impl Dql for GetAccountDql {
41    type View = Option<AccountView>;
42}
43
44impl SqlDql for GetAccountDql {
45    fn build(&self) -> SqlQuery {
46        Select::from("users u")
47            .column(COLUMNS)
48            .join(JOIN_ROLES)
49            .filter("u.id = ?", [Bind::Int(self.user_id)])
50            .filter("u.deleted_at IS NULL", [])
51            .order_by("r.id ASC")
52            .build()
53    }
54
55    /// Sem linha nenhuma o usuário não existe.
56    ///
57    /// Com linhas, ele existe mesmo que todas tenham `role_id` nulo — é o
58    /// usuário sem papel.
59    fn read(&self, rows: Vec<MySqlRow>) -> anyhow::Result<Self::View> {
60        let Some(first) = rows.first() else {
61            return Ok(None);
62        };
63
64        Ok(Some(AccountView {
65            id: Row::id(first, "user_id")?,
66            name: Row::text(first, "user_name")?,
67            email: Row::text(first, "user_email")?,
68            roles: AccountReader::roles_of(&rows)?,
69        }))
70    }
71}
