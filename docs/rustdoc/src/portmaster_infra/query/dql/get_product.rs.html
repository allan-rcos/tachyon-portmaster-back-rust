1//! A consulta de um produto pelo id.
2
3use sqlx::mysql::{MySql, MySqlRow};
4use sqlx::QueryBuilder;
5
6use crate::entity::codec::Codec;
7use crate::query::dql::list_products::read_item;
8use crate::query::views::ProductViewItem;
9use crate::query::{Dql, SqlDql};
10
11/// As colunas que a View de produto precisa.
12const COLUMNS: &str = "p.id, p.name, p.density, p.risk_class";
13
14/// Um produto pelo id.
15///
16/// Falível porque um id em base62 pode simplesmente não ser base62 — uma URL
17/// inventada. Recusar aqui é melhor do que abrir transação e consultar por um
18/// número arbitrário.
19pub fn get_product(id: &str) -> anyhow::Result<impl SqlDql<View = Option<ProductViewItem>>> {
20    Ok(GetProduct {
21        id: Codec::decode_id(id)?,
22    })
23}
24
25/// Um produto pelo id.
26///
27/// Repete o filtro de soft-delete da escrita: sem ele, um produto removido
28/// reapareceria na leitura.
29struct GetProduct {
30    /// O alvo, como `BIGINT`.
31    id: i64,
32}
33
34impl Dql for GetProduct {
35    type View = Option<ProductViewItem>;
36
37    fn cache_key(&self) -> String {
38        format!("get_product:{}", self.id)
39    }
40}
41
42impl SqlDql for GetProduct {
43    fn build(&self) -> QueryBuilder<MySql> {
44        let mut builder = QueryBuilder::new("SELECT ");
45        builder.push(COLUMNS);
46        builder.push(" FROM products p WHERE p.id = ");
47        builder.push_bind(self.id);
48        builder.push(" AND p.deleted_at IS NULL LIMIT 1");
49
50        builder
51    }
52
53    fn read(&self, rows: Vec<MySqlRow>) -> anyhow::Result<Self::View> {
54        rows.first().map(read_item).transpose()
55    }
56}
