1//! A listagem de contêineres com carga e telemetria recente.
2//!
3//! ## As coleções aninhadas viram JSON no próprio banco
4//!
5//! A alternativa — uma consulta por contêiner para a carga e outra para os logs
6//! — custaria 2n idas ao banco por página.
7//!
8//! O timestamp sai como **epoch em ms**, na forma que a View guarda. Sair como
9//! texto de data obrigaria a hidratação a interpretar formato de data vindo de
10//! dentro de um JSON: duas conversões para chegar no mesmo número.
11//!
12//! ## Por que a janela de recentes é sub-consulta escalar
13//!
14//! O `MariaDB` não tem `LATERAL`, então uma tabela derivada não pode enxergar o
15//! `c.id` de fora. Limitar pelo id do n-ésimo mais novo traz as mesmas linhas
16//! com uma correlação que o motor aceita.
17

33/// Quantos registros de telemetria a janela recente traz.
34const RECENT_LOGS: u32 = 10;

36/// A listagem de contêineres com carga e telemetria recente.
37///
38/// `id` restringe a um contêiner só, quando a rota pediu um. Falível pelo mesmo
39/// motivo das consultas por id: um base62 inventado não deve abrir transação.
40pub fn list_container_summaries(
41    cursor: Option<String>,
42    limit: Option<u32>,
43    id: Option<String>,
44) -> anyhow::Result<impl SqlDql<View = ContainerSummaryListView>> {
45    let raw_id = id.as_deref().map(Codec::decode_id).transpose()?;
46
47    Ok(ListContainerSummaries {
48        limit: Paging::effective_limit(limit),
49        id,
50        raw_id,
51        cursor,
52    })
53}

55/// A listagem de contêineres com os satélites deles.
56struct ListContainerSummaries {
57    /// O tamanho da página, já resolvido.
58    limit: u32,
59    /// O id pedido como o cliente o mandou — entra na identidade do cursor.
60    id: Option<String>,
61    /// O mesmo id, decodificado.
62    raw_id: Option<i64>,
63    /// O cursor como o cliente o mandou.
64    cursor: Option<String>,
65}

67impl ListContainerSummaries {
68    /// Os filtros sob os quais um cursor desta consulta vale.
69    fn cursor_filters(&self) -> CursorFilters {
70        CursorFilters::of([
71            ("limit", self.limit.to_string()),
72            ("id", self.id.clone().unwrap_or_default()),
73        ])
74    }
75}

77/// O manifesto agregado pelo banco.
78///
79/// `JSON_ARRAYAGG` devolve `NULL` quando não há linha nenhuma — contêiner vazio,
80/// que é estado normal e não ausência de dado.
81fn read_manifest(json: Option<&str>) -> anyhow::Result<Vec<CargoItemView>> {
82    let entries = read_entries(json, "manifest_json")?;
83    let mut items = Vec::with_capacity(entries.len());
84
85    for entry in &entries {
86        items.push(CargoItemView {
87            product_id: Codec::encode_id(int_of(entry, "product_id")?),
88            product_name: str_of(entry, "product_name")?,
89            quantity: float_of(entry, "quantity")?,
90            weight: float_of(entry, "weight")?,
91        });
92    }
93
94    Ok(items)
95}

97/// A telemetria recente, descartando evento que não reconhece.
98///
99/// Um evento gravado que não corresponde a variante nenhuma é **descartado, não
100/// aproximado**. O campo do fio é um enum: não existe valor que signifique
101/// "aconteceu algo, mas nenhum destes", então escolher uma variante reportaria
102/// um evento que nunca ocorreu. A linha continua em `telemetry_logs` de
103/// qualquer forma.
104fn read_logs(json: Option<&str>) -> anyhow::Result<Vec<TelemetryLogView>> {
105    let entries = read_entries(json, "logs_json")?;
106    let mut logs = Vec::with_capacity(entries.len());
107
108    for entry in &entries {
109        let event = i32::try_from(int_of(entry, "event")?).unwrap_or(-1);
110
111        if TelemetryEvent::from_i32(event).is_none() {
112            continue;
113        }
114
115        logs.push(TelemetryLogView {
116            id: Codec::encode_id(int_of(entry, "id")?),
117            event,
118            description: entry
119                .get("description")
120                .and_then(Value::as_str)
121                .map(str::to_owned),
122            timestamp: int_of(entry, "timestamp")?,
123        });
124    }
125
126    Ok(logs)
127}

129/// As entradas de um array JSON agregado pelo banco.
130fn read_entries(json: Option<&str>, column: &str) -> anyhow::Result<Vec<Value>> {
131    let Some(json) = json else {
132        return Ok(Vec::new());
133    };
134
135    let parsed: Value = serde_json::from_str(json)
136        .with_context(|| format!("coluna `{column}` não é JSON válido"))?;
137
138    Ok(parsed.as_array().cloned().unwrap_or_default())
139}

141/// Um inteiro de dentro do JSON agregado.
142fn int_of(entry: &Value, field: &str) -> anyhow::Result<i64> {
143    entry
144        .get(field)
145        .and_then(Value::as_i64)
146        .with_context(|| format!("campo `{field}` do JSON agregado não é inteiro"))
147}
148
149/// Um real de dentro do JSON agregado.
150fn float_of(entry: &Value, field: &str) -> anyhow::Result<f64> {
151    entry
152        .get(field)
153        .and_then(Value::as_f64)
154        .with_context(|| format!("campo `{field}` do JSON agregado não é real"))
155}
156
157/// Um texto de dentro do JSON agregado.
158fn str_of(entry: &Value, field: &str) -> anyhow::Result<String> {
159    entry
160        .get(field)
161        .and_then(Value::as_str)
162        .map(str::to_owned)
163        .with_context(|| format!("campo `{field}` do JSON agregado não é texto"))
164}

166impl Dql for ListContainerSummaries {
167    type View = ContainerSummaryListView;
168
169    fn cache_key(&self) -> String {
170        format!(
171            "list_container_summaries:{}:{}:{}",
172            self.limit,
173            self.id.as_deref().unwrap_or_default(),
174            self.cursor.as_deref().unwrap_or_default()
175        )
176    }
177}

179impl SqlDql for ListContainerSummaries {
180    fn build(&self) -> QueryBuilder<MySql> {
181        let last_id = Cursor::last_id_or_start(self.cursor.as_deref(), &self.cursor_filters());
182
183        let mut builder = QueryBuilder::new("SELECT ");
184        builder.push(COLUMNS);
185
186        builder.push(
187            ", (SELECT JSON_ARRAYAGG(JSON_OBJECT( \
188             'product_id', ci.product_id, 'product_name', p.name, \
189             'quantity', ci.quantity, 'weight', ci.weight)) \
190             FROM container_items ci \
191             INNER JOIN products p ON p.id = ci.product_id \
192             WHERE ci.container_id = c.id) AS manifest_json",
193        );
194
195        builder.push(format!(
196            ", (SELECT JSON_ARRAYAGG(JSON_OBJECT( \
197             'id', t.id, 'event', t.event, 'description', t.description, \
198             'timestamp', CAST(UNIX_TIMESTAMP(t.timestamp) * 1000 AS SIGNED))) \
199             FROM telemetry_logs t \
200             WHERE t.container_id = c.id \
201             AND t.id >= COALESCE((SELECT t2.id FROM telemetry_logs t2 \
202             WHERE t2.container_id = c.id ORDER BY t2.id DESC LIMIT 1 OFFSET {}), 0)) AS logs_json",
203            RECENT_LOGS - 1
204        ));
205
206        builder.push(", (SELECT COUNT(*) FROM containers WHERE deleted_at IS NULL");
207        if let Some(id) = self.raw_id {
208            builder.push(" AND id = ");
209            builder.push_bind(id);
210        }
211        builder.push(") AS _total");
212
213        builder.push(" FROM containers c WHERE c.id > ");
214        builder.push_bind(last_id);
215        builder.push(" AND c.deleted_at IS NULL");
216
217        if let Some(id) = self.raw_id {
218            builder.push(" AND c.id = ");
219            builder.push_bind(id);
220        }
221
222        builder.push(" ORDER BY c.id ASC LIMIT ");
223        builder.push_bind(i64::from(self.limit));
224
225        builder
226    }
227
228    fn read(&self, rows: Vec<MySqlRow>) -> anyhow::Result<Self::View> {
229        let mut items = Vec::with_capacity(self.limit as usize);
230        let mut total = 0;
231        let mut last_id = 0;
232
233        for row in &rows {
234            let manifest_json: Option<String> = row
235                .try_get("manifest_json")
236                .context("coluna `manifest_json` não veio como texto opcional")?;
237            let logs_json: Option<String> = row
238                .try_get("logs_json")
239                .context("coluna `logs_json` não veio como texto opcional")?;
240
241            items.push(ContainerSummaryViewItem {
242                container: read_item(row)?,
243                manifest: read_manifest(manifest_json.as_deref())?,
244                recent_logs: read_logs(logs_json.as_deref())?,
245            });
246
247            last_id = row
248                .try_get("id")
249                .context("coluna `id` não veio como inteiro")?;
250            total = row
251                .try_get("_total")
252                .context("coluna `_total` não veio como inteiro")?;
253        }
254
255        Ok(ContainerSummaryListView {
256            next_cursor: Cursor::next(items.len(), self.limit, last_id, &self.cursor_filters()),
257            total,
258            items,
259        })
260    }
