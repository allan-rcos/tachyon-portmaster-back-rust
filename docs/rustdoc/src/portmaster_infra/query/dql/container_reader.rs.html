1//! A leitura de um contêiner e dos seus satélites a partir de uma linha.
2
3use crate::entity::codec::Codec;
4use crate::query::row::Row;
5use crate::query::views::{CargoItemView, ContainerViewItem, TelemetryLogView};
6use anyhow::Context;
7use portmaster_domain::enums::{ContainerStatus, TelemetryEvent};
8use serde_json::Value;
9use sqlx::mysql::MySqlRow;
10
11/// Lê contêiner, carga e telemetria de uma linha de consulta.
12pub(crate) struct ContainerReader;
13
14impl ContainerReader {
15    /// Uma linha de `containers` como a View a quer.
16    pub(crate) fn item(row: &MySqlRow) -> anyhow::Result<ContainerViewItem> {
17        Ok(ContainerViewItem {
18            id: Row::id(row, "id")?,
19            code: Row::text(row, "code")?,
20            current_weight: Row::real(row, "current_weight")?,
21            max_capacity: Row::real(row, "max_capacity")?,
22            status: Row::enum_index(row, "status", ContainerStatus::from_i32, "ContainerStatus")?,
23        })
24    }
25
26    /// O manifesto agregado pelo banco.
27    ///
28    /// `JSON_ARRAYAGG` devolve `NULL` quando não há linha nenhuma — contêiner vazio,
29    /// que é estado normal e não ausência de dado.
30    pub(crate) fn manifest_of(json: Option<&str>) -> anyhow::Result<Vec<CargoItemView>> {
31        let entries = Self::entries_of(json, "manifest_json")?;
32        let mut items = Vec::with_capacity(entries.len());
33
34        for entry in &entries {
35            items.push(CargoItemView {
36                product_id: Codec::encode_id(Self::int_of(entry, "product_id")?),
37                product_name: Self::str_of(entry, "product_name")?,
38                quantity: Self::float_of(entry, "quantity")?,
39                weight: Self::float_of(entry, "weight")?,
40            });
41        }
42
43        Ok(items)
44    }
45
46    /// A telemetria recente agregada pelo banco.
47    /// Hidrata a telemetria recente, descartando evento que não reconhece.
48    ///
49    /// Um evento gravado que não corresponde a variante nenhuma é **descartado,
50    /// não aproximado**. O campo do fio é um enum: não existe valor que
51    /// signifique "aconteceu algo, mas nenhum destes", então escolher uma
52    /// variante reportaria um evento que nunca ocorreu. A linha continua em
53    /// `telemetry_logs` de qualquer forma.
54    pub(crate) fn logs_of(json: Option<&str>) -> anyhow::Result<Vec<TelemetryLogView>> {
55        let entries = Self::entries_of(json, "logs_json")?;
56        let mut logs = Vec::with_capacity(entries.len());
57
58        for entry in &entries {
59            let event = i32::try_from(Self::int_of(entry, "event")?).unwrap_or(-1);
60
61            if TelemetryEvent::from_i32(event).is_none() {
62                continue;
63            }
64
65            logs.push(TelemetryLogView {
66                id: Codec::encode_id(Self::int_of(entry, "id")?),
67                event,
68                description: entry
69                    .get("description")
70                    .and_then(Value::as_str)
71                    .map(str::to_owned),
72                timestamp: Self::int_of(entry, "timestamp")?,
73            });
74        }
75
76        Ok(logs)
77    }
78
79    /// As entradas de um array JSON agregado pelo banco.
80    pub(crate) fn entries_of(json: Option<&str>, column: &str) -> anyhow::Result<Vec<Value>> {
81        let Some(json) = json else {
82            return Ok(Vec::new());
83        };
84
85        let parsed: Value = serde_json::from_str(json)
86            .with_context(|| format!("coluna `{column}` não é JSON válido"))?;
87
88        Ok(parsed.as_array().cloned().unwrap_or_default())
89    }
90
91    /// Um inteiro de dentro do JSON agregado.
92    pub(crate) fn int_of(entry: &Value, field: &str) -> anyhow::Result<i64> {
93        entry
94            .get(field)
95            .and_then(Value::as_i64)
96            .with_context(|| format!("campo `{field}` do JSON agregado não é inteiro"))
97    }
98
99    /// Um real de dentro do JSON agregado.
100    pub(crate) fn float_of(entry: &Value, field: &str) -> anyhow::Result<f64> {
101        entry
102            .get(field)
103            .and_then(Value::as_f64)
104            .with_context(|| format!("campo `{field}` do JSON agregado não é real"))
105    }
106
107    /// Um texto de dentro do JSON agregado.
108    pub(crate) fn str_of(entry: &Value, field: &str) -> anyhow::Result<String> {
109        entry
110            .get(field)
111            .and_then(Value::as_str)
112            .map(str::to_owned)
113            .with_context(|| format!("campo `{field}` do JSON agregado não é texto"))
114    }
115}
