1//! A consulta de um produto pelo id.
2
3use crate::query::dql::product_reader::ProductReader;
4use crate::query::sql::{Bind, Select, SqlQuery};
5use crate::query::views::ProductViewItem;
6use crate::query::{Dql, SqlDql};
7use sqlx::mysql::MySqlRow;
8
9/// As colunas que a View de produto precisa.
10///
11/// Nomeadas em vez de `*`: a projeção é o contrato da hidratação, e um `SELECT *`
12/// faria uma coluna nova entrar na consulta sem que ninguém a pedisse.
13const COLUMNS: &str = "p.id, p.name, p.density, p.risk_class";
14
15/// Um produto pelo id.
16///
17/// Repete o filtro de soft-delete da escrita: sem ele, um produto removido
18/// reapareceria na leitura.
19pub struct GetProductDql {
20    /// O alvo, como `BIGINT` — o base62 já foi decodificado pelo caso de uso.
21    id: i64,
22}
23
24impl GetProductDql {
25    /// Monta a consulta.
26    pub(crate) const fn new(id: i64) -> Self {
27        Self { id }
28    }
29}
30
31impl Dql for GetProductDql {
32    type View = Option<ProductViewItem>;
33}
34
35impl SqlDql for GetProductDql {
36    fn build(&self) -> SqlQuery {
37        Select::from("products p")
38            .column(COLUMNS)
39            .filter("p.id = ?", [Bind::Int(self.id)])
40            .filter("p.deleted_at IS NULL", [])
41            .limit(1)
42            .build()
43    }
44
45    fn read(&self, rows: Vec<MySqlRow>) -> anyhow::Result<Self::View> {
46        rows.first().map(ProductReader::item).transpose()
47    }
48}
