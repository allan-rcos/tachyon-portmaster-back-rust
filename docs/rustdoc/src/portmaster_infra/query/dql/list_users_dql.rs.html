12/// A projeção comum às duas consultas.
13///
14/// Usuário e papel na mesma linha, com o papel prefixado porque `id` e `name`
15/// existem dos dois lados.
16const COLUMNS: &str = "u.id AS user_id, u.name AS user_name, u.email AS user_email, \
17                       r.id AS role_id, r.name AS role_name, r.permissions AS role_permissions, \
18                       (SELECT COUNT(*) FROM user_roles urc WHERE urc.role_id = r.id) AS role_user_count";
19
20/// A ligação até os papéis.
21///
22/// O `deleted_at` do papel mora na condição do `JOIN`, não no `WHERE`. Movido
23/// para o `WHERE`, ele descartaria a linha inteira quando não houvesse papel — o
24/// `LEFT JOIN` viraria `INNER`, e um usuário sem papel nenhum sumiria da
25/// listagem em vez de aparecer com a lista vazia.
26const JOIN_ROLES: &str = "LEFT JOIN user_roles ur ON ur.user_id = u.id \
27                          LEFT JOIN roles r ON r.id = ur.role_id AND r.deleted_at IS NULL";
28
29/// A listagem de usuários.
30pub struct ListUsersDql {
31    /// Página e limite — esta listagem pagina por página, não por cursor.
32    params: UserListParams,
33}
34
35impl ListUsersDql {
36    /// Monta a consulta.
37    pub(crate) const fn new(params: UserListParams) -> Self {
38        Self { params }
39    }
40
41    /// O tamanho da página.
42    fn limit(&self) -> u32 {
43        Paging::effective_limit(self.params.limit)
44    }
45
46    /// Quantos usuários pular. Página ausente ou zero é a primeira.
47    ///
48    /// O `page` vem da query string, então é um `u32` arbitrário que o cliente
49    /// escolhe. Multiplicá-lo pelo limite estoura o `u32` a partir de ~86
50    /// milhões de páginas — em release isso daria a volta e devolveria a página
51    /// errada em silêncio. Saturar é o comportamento certo: pedir uma página
52    /// além do fim é uma lista vazia, não um resultado sorteado.
53    fn offset(&self) -> u32 {
54        let page = self.params.page.filter(|value| *value > 0).unwrap_or(1);
55
56        page.saturating_sub(1).saturating_mul(self.limit())
57    }
58}
59
60impl Dql for ListUsersDql {
61    type View = UserListView;
62}
63
64impl SqlDql for ListUsersDql {
65    /// A página sai numa tabela derivada, e não de um LIMIT no SELECT de fora.
66    ///
67    /// O `LEFT JOIN` com papéis multiplica as linhas por usuário, então um
68    /// LIMIT externo cortaria no meio de um usuário: o vigésimo apareceria com
69    /// parte dos papéis e nenhum indício de que faltou.
70    fn build(&self) -> SqlQuery {
71        let page = Select::from("users")
72            .column("id, name, email")
73            .filter("deleted_at IS NULL", [])
74            .order_by("id ASC")
75            .limit(self.limit())
76            .offset(self.offset())
77            .to_sql();
78
79        Select::from(format!("({page}) AS u"))
80            .column(COLUMNS)
81            .join(JOIN_ROLES)
82            .order_by("u.id ASC, r.id ASC")
83            .build()
84    }
85
86    /// Agrupa o fan-out de papéis de volta em um usuário por item.
87    ///
88    /// As linhas chegam agrupadas por usuário (`ORDER BY u.id`), então comparar
89    /// com a última basta para saber se começou outro — sem mapa auxiliar e sem
90    /// perder a ordem da consulta.
91    fn read(&self, rows: Vec<MySqlRow>) -> anyhow::Result<Self::View> {
92        let mut items: Vec<AccountView> = Vec::with_capacity(self.limit() as usize);
93
94        for row in &rows {
95            let id = Row::id(row, "user_id")?;
96
97            if items.last().map(|last| last.id.as_str()) != Some(id.as_str()) {
98                items.push(AccountView {
99                    id,
100                    name: Row::text(row, "user_name")?,
101                    email: Row::text(row, "user_email")?,
102                    roles: Vec::new(),
103                });
104            }
105
106            if let Some(role) = AccountReader::role_of(row)? {
107                // `last` existe: ou já estava, ou acabou de ser inserido acima.
108                if let Some(user) = items.last_mut() {
109                    user.roles.push(role);
110                }
111            }
112        }
113
114        Ok(UserListView { items })
115    }
116}

118#[cfg(test)]
119mod tests {
120    use super::*;
121    use crate::query::dql::get_account_dql::GetAccountDql;
122    use crate::query::sql::Bind;
123    use pretty_assertions::assert_eq;
124
125    #[test]
126    fn a_conta_traz_os_papeis_por_left_join() {
127        let query = GetAccountDql::new(123).build();
128
129        assert!(
130            query
131                .sql
132                .contains("LEFT JOIN roles r ON r.id = ur.role_id AND r.deleted_at IS NULL"),
133            "o filtro de papel apagado tem que ficar no JOIN: {}",
134            query.sql
135        );
136        assert_eq!(query.binds, vec![Bind::Int(123)]);
137    }
138
139    /// É o que impede o fan-out de papéis de cortar a página no meio de um
140    /// usuário.
141    #[test]
142    fn a_pagina_de_usuarios_sai_de_uma_tabela_derivada() {
143        let query = ListUsersDql::new(UserListParams {
144            page: Some(3),
145            limit: Some(10),
146        })
147        .build();
148
149        assert_eq!(
150            query.sql,
151            "SELECT u.id AS user_id, u.name AS user_name, u.email AS user_email, \
152             r.id AS role_id, r.name AS role_name, r.permissions AS role_permissions, \
153             (SELECT COUNT(*) FROM user_roles urc WHERE urc.role_id = r.id) AS role_user_count \
154             FROM (SELECT id, name, email FROM users WHERE deleted_at IS NULL \
155             ORDER BY id ASC LIMIT 10 OFFSET 20) AS u \
156             LEFT JOIN user_roles ur ON ur.user_id = u.id \
157             LEFT JOIN roles r ON r.id = ur.role_id AND r.deleted_at IS NULL \
158             ORDER BY u.id ASC, r.id ASC"
159        );
160        assert!(query.binds.is_empty());
161    }
162
163    #[test]
164    fn a_primeira_pagina_nao_pula_nada() {
165        for page in [None, Some(0), Some(1)] {
166            let dql = ListUsersDql::new(UserListParams { page, limit: None });
167
168            assert_eq!(dql.offset(), 0, "página {page:?} não deveria pular linhas");
169        }
170    }
