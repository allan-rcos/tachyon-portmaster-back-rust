21impl Dql for MetricsDql {

23}
24
25impl SqlDql for MetricsDql {
26    /// Oito sub-consultas escalares num `SELECT` sem `FROM`: uma ida ao banco
27    /// devolve a linha inteira do painel.
28    ///
29    /// Oito consultas separadas dariam o mesmo número por oito vezes o custo
30    /// de rede. Os índices de status são bindados, e não interpolados, mesmo
31    /// sendo constantes nossas. Interpolar número em SQL é o hábito que um dia
32    /// encontra um valor que não é constante.
33    fn build(&self) -> SqlQuery {
34        let occupancy = |status: ContainerStatus, alias: &str| {
35            (
36                format!("(SELECT COUNT(*) FROM containers WHERE deleted_at IS NULL AND status = ?) AS {alias}"),
37                vec![Bind::Int(status.as_i32().into())],
38            )
39        };
40
41        let (empty, empty_bind) = occupancy(ContainerStatus::Empty, "occ_empty");
42        let (loading, loading_bind) = occupancy(ContainerStatus::Loading, "occ_loading");
43        let (sealed, sealed_bind) = occupancy(ContainerStatus::Sealed, "occ_sealed");
44        let (in_transit, in_transit_bind) = occupancy(ContainerStatus::InTransit, "occ_in_transit");
45
46        Select::from("DUAL")
47            .column("(SELECT COUNT(*) FROM containers WHERE deleted_at IS NULL) AS total_containers")
48            .column_bound(
49                "(SELECT COUNT(*) FROM containers WHERE deleted_at IS NULL AND status <> ?) AS active_containers",
50                [Bind::Int(ContainerStatus::Empty.as_i32().into())],
51            )
52            .column(
53                "(SELECT COALESCE(SUM(current_weight), 0) FROM containers WHERE deleted_at IS NULL) AS yard_load",
54            )
55            .column("(SELECT COUNT(*) FROM products WHERE deleted_at IS NULL) AS registered_products")
56            .column_bound(empty, empty_bind)
57            .column_bound(loading, loading_bind)
58            .column_bound(sealed, sealed_bind)
59            .column_bound(in_transit, in_transit_bind)
60            .build()
61    }
62
63    /// Um `SELECT` de agregações sempre devolve exatamente uma linha.
64    ///
65    /// Sem nenhuma, o painel zerado é a leitura honesta — não há o que
66    /// reportar.
67    fn read(&self, rows: Vec<MySqlRow>) -> anyhow::Result<Self::View> {
68        let Some(row) = rows.first() else {
69            return Ok(MetricsView::default());
70        };
71
72        Ok(MetricsView {
73            active_containers: Row::number(row, "active_containers")?,
74            total_containers: Row::number(row, "total_containers")?,
75            yard_load: Row::real(row, "yard_load")?,
76            registered_products: Row::number(row, "registered_products")?,
77            occupancy: OccupancyView {
78                empty: Row::number(row, "occ_empty")?,
79                loading: Row::number(row, "occ_loading")?,
80                sealed: Row::number(row, "occ_sealed")?,
81                in_transit: Row::number(row, "occ_in_transit")?,
82            },
83        })
84    }
85}

87#[cfg(test)]
88mod tests {
89    use super::*;
90    use pretty_assertions::assert_eq;
91
92    #[test]
93    fn o_painel_sai_de_uma_ida_so_ao_banco() {
94        let query = MetricsDql::new().build();
95
96        assert_eq!(
97            query.sql.matches("AS ").count(),
98            8,
99            "as oito agregações deveriam sair na mesma linha: {}",
100            query.sql
101        );
102    }
103
104    #[test]
105    fn os_status_sao_bindados_e_na_ordem_das_colunas() {
106        let query = MetricsDql::new().build();
107
108        assert_eq!(
109            query.binds,
110            vec![
111                // active_containers (<> Empty), depois a ocupação em ordem.
112                Bind::Int(0),
113                Bind::Int(0),
114                Bind::Int(1),
115                Bind::Int(2),
116                Bind::Int(3),
117            ]
118        );
119    }
120
121    #[test]
122    fn sem_linha_o_painel_sai_zerado() {
123        assert_eq!(
124            MetricsDql::new().read(Vec::new()).unwrap(),
125            MetricsView::default()
126        );
127    }
