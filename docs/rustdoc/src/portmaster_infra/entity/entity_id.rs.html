3use std::sync::OnceLock;
4
5use anyhow::{anyhow, Context as _};
6
7use crate::entity::codec::Codec;

9/// O id de uma linha: `BIGINT` na coluna, base62 no fio.
10///
11/// Um campo só onde antes havia dois. Cada entity guardava `id: String` e
12/// `raw_id: i64` lado a lado, e os dois precisavam concordar — mantidos em
13/// sincronia à mão no `from_row` e no `from_domain`, sem nada que impedisse um
14/// de ficar para trás.
15///
16/// ## O base62 é derivado, e só quando alguém o pede
17///
18/// A coluna guarda o `BIGINT`; o base62 existe porque o trait de domínio expõe
19/// `fn id(&self) -> &str`. Como aquela assinatura devolve **referência**, não dá
20/// para codificar na hora e devolver o resultado — daí o [`OnceLock`], que
21/// codifica na primeira leitura e guarda o resultado para as seguintes.
22///
23/// Quem lê uma linha só para pegar a FK, ou para conferir que ela existe, nunca
24/// paga a codificação.
25#[derive(Debug, Clone)]
26pub(crate) struct EntityId {
27    /// O id como a coluna o guarda.
28    raw: i64,
29    /// O base62 correspondente, vazio até alguém pedir.
30    encoded: OnceLock<String>,
31}
32
33impl EntityId {
34    /// O id como o banco o guarda.
35    pub(crate) const fn raw(&self) -> i64 {
36        self.raw
37    }
38
39    /// O id como o domínio o expõe.
40    ///
41    /// A primeira chamada codifica; as seguintes devolvem o que ficou guardado.
42    pub(crate) fn as_str(&self) -> &str {
43        self.encoded.get_or_init(|| Codec::encode_id(self.raw))
44    }
45}
46
47impl TryFrom<i64> for EntityId {
48    type Error = anyhow::Error;
49
50    /// Adota o id que veio da coluna.
51    ///
52    /// Recusa negativo em vez de codificá-lo: o Snowflake não acende o bit de
53    /// sinal, então um id negativo veio de uma linha que o schema não deveria
54    /// admitir. Transformá-lo em `0` afirmaria uma identidade que a linha não
55    /// tem, que é a mesma coisa que escolher variante de enum por aproximação.
56    fn try_from(raw: i64) -> Result<Self, Self::Error> {
57        if raw.is_negative() {
58            return Err(anyhow!("id negativo na coluna: {raw}"));
59        }
60
61        Ok(Self {
62            raw,
63            encoded: OnceLock::new(),
64        })
65    }
66}
67
68impl TryFrom<&str> for EntityId {
69    type Error = anyhow::Error;
70
71    /// Adota o id que veio do domínio, que já está em base62.
72    ///
73    /// Popula os dois lados de uma vez: quem chega por aqui veio de um
74    /// `from_domain`, já tem a string em mãos, e guardá-la evita recodificar o
75    /// que acabou de ser decodificado.
76    fn try_from(id: &str) -> Result<Self, Self::Error> {
77        let raw = Codec::decode_id(id).with_context(|| format!("id de domínio inválido: {id}"))?;
78        let encoded = OnceLock::new();
79
80        let _ = encoded.set(id.to_owned());
81
82        Ok(Self { raw, encoded })
83    }
84}
