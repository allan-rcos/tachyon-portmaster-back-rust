3use chrono::{DateTime, Utc};
4use mysql_async::prelude::FromRow;
5use portmaster_domain::domain::Container;
6use portmaster_domain::enums::ContainerStatus;
7
8use crate::entity::decode::Decode;
9use crate::entity::entity_id::EntityId;
10
11/// A entity, que é também a linha de `containers`.
12#[derive(Clone, FromRow)]
13pub struct ContainerEntity {
14    /// A identidade, nas duas formas.
15    #[mysql(
16        deserialize_with = "Decode::entity_id",
17        serialize_with = "Decode::entity_id_value"
18    )]
19    id: EntityId,
20    /// O código do contêiner.
21    code: String,
22    /// Peso embarcado, em quilos.
23    current_weight: f64,
24    /// Teto de peso.
25    max_capacity: f64,
26    /// O status, já como enum de domínio — a coluna guarda o índice.
27    ///
28    /// O [`Decode`] valida o índice na leitura: um valor que não corresponde a
29    /// variante nenhuma é uma linha que o schema não deveria admitir, e escolher
30    /// uma variante por aproximação afirmaria um estado que o banco não guardou.
31    #[mysql(
32        deserialize_with = "Decode::container_status",
33        serialize_with = "Decode::container_status_value"
34    )]
35    status: ContainerStatus,
36    /// Quando a linha nasceu, em UTC.
37    #[mysql(deserialize_with = "Decode::utc", serialize_with = "Decode::utc_value")]
38    created_at: DateTime<Utc>,
39    /// Quando a linha mudou pela última vez, em UTC.
40    #[mysql(deserialize_with = "Decode::utc", serialize_with = "Decode::utc_value")]
41    updated_at: DateTime<Utc>,
42    /// Quando foi removida, ou `None` se ativa — o soft-delete.
43    #[mysql(deserialize_with = "Decode::utc_opt")]
44    deleted_at: Option<DateTime<Utc>>,
45}
46
47impl ContainerEntity {
48    /// Recria a entity a partir de qualquer [`Container`], para gravá-la.
49    pub(crate) fn from_domain(source: &dyn Container) -> anyhow::Result<Self> {
50        Ok(Self {
51            id: EntityId::try_from(source.id())?,
52            code: source.code().to_owned(),
53            current_weight: source.current_weight(),
54            max_capacity: source.max_capacity(),
55            status: source.status(),
56            created_at: source.created_at(),
57            updated_at: source.updated_at(),
58            deleted_at: source.deleted_at(),
59        })
60    }
61
62    /// O id como o banco o guarda.
63    pub(crate) const fn raw_id(&self) -> i64 {
64        self.id.raw()
65    }
66}
