1//! As conversões de coluna que o derive não faz sozinho.
2//!
3//! O `FromRow` derivado converte uma coluna quando o tipo do campo sabe nascer
4//! de um `Value`. Quatro dos tipos que as entities guardam não sabem — o
5//! [`EntityId`], os dois enums de domínio e o `DateTime<Utc>` — e para esses o
6//! derive aceita um par de funções por campo: uma que lê e uma que devolve o
7//! valor à linha.
8//!
9//! Todas as quatro conversões partem de um inteiro, porque é isso que as colunas
10//! guardam: id e epoch em `BIGINT`, índice de enum em `TINYINT`.
11//!
12//! ## Por que a devolução existe
13//!
14//! Quando um campo falha, o derive precisa recompor a linha inteira para
15//! entregá-la no erro — inclusive os campos que já tinham sido lidos. Sem a
16//! função de volta ele tentaria um `Into<Value>` que estes tipos não têm, e o
17//! par ficaria pela metade.
18//!
19//! Só quem **precede** outro campo é recomposto, então `deleted_at`, último em
20//! todas as entities, lê sem devolver. Um campo novo depois dele passa a exigir
21//! a volta, e a falta dela é erro de compilação — não silêncio.
22
23use chrono::{DateTime, Utc};
24use mysql_async::{from_value_opt, FromValueError, Value};
25use portmaster_domain::enums::{ContainerStatus, RiskClass};
26
27use crate::entity::entity_id::EntityId;
28
29/// Traduz entre a coluna e o tipo que a entity guarda.
30///
31/// Namespace no molde do [`Codec`](super::codec::Codec): cada par de funções é
32/// a mesma fronteira vista dos dois lados, e nenhuma delas faz sentido fora de
33/// um atributo de campo.
34pub(crate) struct Decode;
35
36impl Decode {
37    /// Adota o id que veio da coluna.
38    ///
39    /// A recusa devolve o `Value` original e não a mensagem do [`EntityId`]: o
40    /// derive só carrega o valor, e é ele que aparece na linha do
41    /// `FromRowError`. Quem a lê ainda sabe qual consulta falhou, porque o
42    /// repositório embrulha o erro com o id que pediu.
43    pub fn entity_id(value: Value) -> Result<EntityId, FromValueError> {
44        let raw: i64 = from_value_opt(value.clone())?;
45
46        EntityId::try_from(raw).map_err(|_| FromValueError(value))
47    }
48
49    /// Devolve o id à linha, como a coluna o guarda.
50    pub fn entity_id_value(id: EntityId) -> Value {
51        Value::Int(id.into_raw())
52    }
53
54    /// Lê a coluna de tempo como o instante que o epoch nela nomeia.
55    ///
56    /// A coluna é `BIGINT` com epoch em milissegundos — um número com um
57    /// significado só, que nenhuma variável de sessão reinterpreta. Um valor que
58    /// não corresponde a instante nenhum é recusado em vez de saturado: data
59    /// aproximada é data errada dita com confiança.
60    pub fn utc(value: Value) -> Result<DateTime<Utc>, FromValueError> {
61        let millis: i64 = from_value_opt(value.clone())?;
62
63        DateTime::from_timestamp_millis(millis).ok_or(FromValueError(value))
64    }
65
66    /// Devolve o instante à linha, como o epoch que a coluna guarda.
67    pub const fn utc_value(at: DateTime<Utc>) -> Value {
68        Value::Int(at.timestamp_millis())
69    }
70
71    /// O mesmo que [`utc`](Self::utc), para a coluna que aceita nulo.
72    ///
73    /// `NULL` é ausência legítima aqui — é o soft-delete não aplicado — e não
74    /// falha de conversão.
75    pub fn utc_opt(value: Value) -> Result<Option<DateTime<Utc>>, FromValueError> {
76        if value == Value::NULL {
77            return Ok(None);
78        }
79
80        Self::utc(value).map(Some)
81    }
82
83    /// Lê o índice gravado como a classe de risco que ele nomeia.
84    ///
85    /// Um índice fora da faixa é recusado em vez de aproximado: escolher a
86    /// variante mais próxima afirmaria uma classe de risco que o banco não
87    /// guardou, e classe de risco errada é carga perigosa reportada como
88    /// inofensiva.
89    pub fn risk_class(value: Value) -> Result<RiskClass, FromValueError> {
90        let index = Self::index(&value)?;
91
92        RiskClass::try_from(index).map_err(|_| FromValueError(value))
93    }
94
95    /// Devolve a classe de risco à linha, como índice.
96    pub fn risk_class_value(class: RiskClass) -> Value {
97        Value::Int(i64::from(class.as_i32()))
98    }
99
100    /// Lê o índice gravado como o status que ele nomeia.
101    ///
102    /// Recusa pelo mesmo motivo da classe de risco: um status aproximado
103    /// afirmaria um estado que a linha não guarda.
104    pub fn container_status(value: Value) -> Result<ContainerStatus, FromValueError> {
105        let index = Self::index(&value)?;
106
107        ContainerStatus::try_from(index).map_err(|_| FromValueError(value))
108    }
109
110    /// Devolve o status à linha, como índice.
111    pub fn container_status_value(status: ContainerStatus) -> Value {
112        Value::Int(i64::from(status.as_i32()))
113    }
114
115    /// O índice de enum que a coluna guarda.
116    ///
117    /// A coluna é `TINYINT` e o driver a entrega como `i64`; os enums de
118    /// domínio falam `i32`. O estreitamento é falível de propósito — um valor
119    /// que não cabe em `i32` não corresponde a variante nenhuma de qualquer
120    /// forma.
121    fn index(value: &Value) -> Result<i32, FromValueError> {
122        let raw: i64 = from_value_opt(value.clone())?;
123
124        i32::try_from(raw).map_err(|_| FromValueError(value.clone()))
125    }
126}
