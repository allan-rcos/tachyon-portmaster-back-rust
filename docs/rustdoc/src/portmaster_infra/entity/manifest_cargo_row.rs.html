1//! A linha crua de `manifest_cargo`, como o `sqlx` a lê.
2
3use chrono::{DateTime, Utc};
4
5/// Uma linha de `container_items`.
6///
7/// Entidade fraca: só `created_at`, sem `updated_at` nem `deleted_at`. Não é
8/// atualizada nem sofre soft-delete — mudar uma linha é removê-la e recriá-la,
9/// e removê-la é `DELETE` de verdade.
10#[derive(sqlx::FromRow)]
11pub(crate) struct ManifestCargoRow {
12    /// FK do contêiner, como `BIGINT`.
13    pub(crate) container_id: i64,
14    /// FK do produto, como `BIGINT`.
15    pub(crate) product_id: i64,
16    /// Coluna `quantity`.
17    pub(crate) quantity: f64,
18    /// Coluna `weight`, já calculada na escrita.
19    pub(crate) weight: f64,
20    /// Quando a linha entrou, em UTC.
21    pub(crate) created_at: DateTime<Utc>,
22}
