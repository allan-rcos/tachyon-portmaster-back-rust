1//! A tradução entre o que a coluna guarda e o que o domínio expõe.
2//!
3//! Esta é a única camada que vê o `i64` do Snowflake e o índice numérico de um
4//! enum. Ao ler uma linha, deriva o base62 e a variante que o trait expõe; ao
5//! gravar, decodifica de volta. Se o id físico mudar um dia, só este arquivo
6//! muda.
7
8use anyhow::{anyhow, Context};
9use portmaster_domain::id::Base62;
10
11/// Converte entre a representação da coluna e a do domínio.
12///
13/// Namespace, no molde do `Base62`: as três funções são a mesma fronteira vista
14/// de três ângulos, e nenhuma faz sentido fora dela.
15pub(crate) struct Codec;
16
17impl Codec {
18    /// Traduz um id base62 no `BIGINT` da coluna.
19    ///
20    /// Um id inválido aqui não é corrupção interna: é uma URL inventada, e precisa
21    /// falhar como entrada malformada em vez de virar uma consulta por um id
22    /// arbitrário.
23    pub fn decode_id(id: &str) -> anyhow::Result<i64> {
24        Base62::decode(id).with_context(|| format!("id fora do formato base62: {id}"))
25    }
26
27    /// Traduz um `BIGINT` no id base62 que o trait expõe.
28    pub fn encode_id(id: i64) -> String {
29        Base62::encode(id)
30    }
31
32    /// Traduz o índice numérico de uma coluna no enum de domínio.
33    ///
34    /// Um valor que não corresponde a variante nenhuma é uma linha que o schema não
35    /// deveria admitir — falhar alto aqui é melhor do que escolher uma variante por
36    /// aproximação e seguir com o dado errado.
37    pub fn decode_enum<T>(
38        value: i32,
39        from: impl Fn(i32) -> Option<T>,
40        name: &str,
41    ) -> anyhow::Result<T> {
42        from(value)
43            .ok_or_else(|| anyhow!("valor {value} não corresponde a nenhuma variante de {name}"))
44    }
45}
