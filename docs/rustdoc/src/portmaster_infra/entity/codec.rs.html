1//! A tradução entre o id da coluna e o id do fio.
2//!
3//! Esta é a única camada que vê o `i64` do Snowflake. Ao ler uma linha, deriva o
4//! base62 que o trait expõe; ao gravar, decodifica de volta. Se o id físico
5//! mudar um dia, só este arquivo muda.
6//!
7//! A validação de índice de enum saiu daqui: ela é do ponto de conversão, e
8//! cada `FromRow` e cada leitura de DQL a faz com o `ok_or_else` de quem sabe
9//! que enum está lendo.
10
11use anyhow::Context;
12
13/// Converte entre o id da coluna e o id do fio.
14///
15/// Namespace no molde do `Base62`: as duas funções são a mesma fronteira vista
16/// dos dois lados, e nenhuma faz sentido fora dela.
17pub(crate) struct Codec;
18
19impl Codec {
20    /// Traduz um id base62 no `BIGINT` da coluna.
21    ///
22    /// Um id inválido aqui não é corrupção interna: é uma URL inventada, e precisa
23    /// falhar como entrada malformada em vez de virar uma consulta por um id
24    /// arbitrário. O `try_from` cobre o caso extra que a decodificação não cobre:
25    /// uma string curta o bastante para caber num `u128` e grande demais para
26    /// caber na coluna.
27    pub fn decode_id(id: &str) -> anyhow::Result<i64> {
28        let decoded =
29            base62::decode(id).with_context(|| format!("id fora do formato base62: {id}"))?;
30
31        i64::try_from(decoded).with_context(|| format!("id base62 fora da faixa da coluna: {id}"))
32    }
33
34    /// Traduz um `BIGINT` no id base62 que o trait expõe.
35    ///
36    /// Id negativo não existe: o Snowflake não acende o bit de sinal. Se um
37    /// aparecer, ele veio de uma linha que o schema não deveria admitir, e
38    /// codificá-lo como zero é preferível a um pânico de conversão no meio de
39    /// uma leitura.
40    pub fn encode_id(id: i64) -> String {
41        base62::encode(u64::try_from(id).unwrap_or_default())
42    }
43}
