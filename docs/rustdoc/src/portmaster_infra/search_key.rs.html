1//! Normalização de texto para busca.
2//!
3//! A coluna `search_*` existe porque o banco pode trabalhar a favor: em vez de
4//! forçar um `LIKE` sobre a string original — que erraria por acento, caixa ou
5//! espaço duplicado —, guarda-se uma versão normalizada ao lado, indexada, e a
6//! busca roda sobre ela.
7//!
8//! O objeto de domínio segue fiel ao que o usuário digitou; quem ganha uma
9//! coluna auxiliar é a tabela, e só a `infra` sabe que ela existe.
10
11use unicode_normalization::char::is_combining_mark;
12use unicode_normalization::UnicodeNormalization as _;
13
14/// Normaliza um texto para a coluna de busca.
15///
16/// Continua sendo uma função só, e não a expressão repetida em cada ponto de
17/// uso, porque os dois lados **têm** de produzir o mesmo byte: quem grava a
18/// coluna e quem monta o `LIKE` que a consulta. Duas cópias que divergissem não
19/// dariam erro nenhum — só devolveriam busca vazia para registro que existe.
20pub(crate) struct SearchKey;
21
22impl SearchKey {
23    /// Reduz um texto à chave usada pelo filtro `LIKE`.
24    ///
25    /// Tira acento, baixa a caixa e colapsa espaço. A decomposição é a do
26    /// Unicode (NFKD), não uma tabela de letras latinas escrita à mão: `Á` vira
27    /// `A` seguido de um acento combinante, e é o acento que se descarta.
28    ///
29    /// O que **não** tem decomposição é preservado, e é por isso que a
30    /// normalização é essa e não uma transliteração para ASCII: `Ячмень` fica
31    /// em cirílico, minúsculo. Reduzi-lo a `yachmen` tornaria o registro
32    /// inencontrável para quem o digitou como ele é.
33    pub(crate) fn of(value: &str) -> String {
34        let folded: String = value
35            .nfkd()
36            .filter(|character| !is_combining_mark(*character))
37            .flat_map(char::to_lowercase)
38            .collect();
39
40        folded.split_whitespace().collect::<Vec<_>>().join(" ")
41    }
42}
