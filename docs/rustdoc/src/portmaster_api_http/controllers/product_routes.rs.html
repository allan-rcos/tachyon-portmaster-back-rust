3use crate::controllers::product_controller::ProductController;
4use crate::router::route::Route;

6/// A tabela de produto.
7pub(crate) fn routes<C: ProductController>(controller: C) -> Vec<Route> {
8    let list = controller.clone();
9    let create = controller.clone();
10    let read = controller.clone();
11    let update = controller.clone();
12
13    vec![
14        Route::get("/products", move |params| list.clone().list(params)),
15        Route::post("/products", move |body| create.clone().create(body)),
16        Route::get("/products/{id}", move |id| read.clone().get(id)),
17        Route::put("/products/{id}", move |id, body| {
18            update.clone().update(id, body)
19        }),
20        Route::delete("/products/{id}", move |id| controller.clone().delete(id)),
21    ]
22}
