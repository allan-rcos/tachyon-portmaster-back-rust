1//! As rotas de sessão.
2//!
3//! As quatro são públicas no sentido de não exigirem access token — quem chega
4//! em `/auth/refresh` está justamente com um que expirou. O que elas exigem,
5//! cada uma à sua maneira, está dentro do controller.
6
7use crate::controllers::auth_controller::AuthController;
8use crate::router::route::Route;
9
10/// A tabela de sessão.
11pub(crate) fn routes<C: AuthController>(controller: C) -> Vec<Route> {
12    let setup = controller.clone();
13    let login = controller.clone();
14    let refresh = controller.clone();
15
16    vec![
17        Route::post("/setup", move |body| setup.clone().setup(body)),
18        Route::post("/auth/login", move |body| login.clone().login(body)),
19        Route::post("/auth/refresh", move || refresh.clone().refresh()),
20        Route::post("/auth/logout", move || controller.clone().logout()),
21    ]
22}
