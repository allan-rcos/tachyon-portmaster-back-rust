18/// Os handlers de papel, genéricos sobre o caso de uso.
19#[derive(Clone)]
20pub(crate) struct RoleControllerImpl<R> {
21    /// O caso de uso de papel.
22    roles: R,
23}
24
25impl<R: RoleUseCase> RoleControllerImpl<R> {
26    /// Monta o controller.
27    pub(crate) const fn new(roles: R) -> Self {
28        Self { roles }
29    }
30
31    /// O papel pelo lado de leitura, já na forma do fio.
32    async fn read(&self, context: UserContext, id: String) -> Result<RoleXResponse, ApiError> {
33        let view = self
34            .roles
35            .get(GetRoleQuery { context, id })
36            .await
37            .map_err(to_api)?;
38
39        Ok(RoleXResponse::of(view))
40    }
41}
42
43impl<R: RoleUseCase + Clone + Send + Sync + 'static> RoleController for RoleControllerImpl<R> {
44    async fn list(
45        &self,
46        context: UserContext,
47        params: PageParams,
48    ) -> Result<RoleListXResponse, ApiError> {
49        let view = self
50            .roles
51            .list(ListRolesQuery {
52                context,
53                cursor: params.cursor,
54                limit: params.limit,
55                search: params.search,
56            })
57            .await
58            .map_err(to_api)?;
59
60        Ok(RoleListXResponse::of(view))
61    }
62
63    /// Cria o papel e o relê pelo lado de leitura.
64    ///
65    /// A releitura é o que faz a resposta trazer `user_count` — o objeto de
66    /// domínio recém-criado não o conhece, e ele faz parte do que a mensagem
67    /// publica.
68    async fn create(
69        &self,
70        context: UserContext,
71        request: RoleCreateXRequest,
72    ) -> Result<RoleXResponse, ApiError> {
73        let role = self
74            .roles
75            .create(CreateRoleCommand {
76                context: context.clone(),
77                name: request.name.unwrap_or_default(),
78                permissions: request.permissions.unwrap_or_default(),
79            })
80            .await
81            .map_err(to_api)?;
82
83        self.read(context, role.id().to_owned()).await
84    }
85
86    async fn update_permissions(
87        &self,
88        context: UserContext,
89        id: String,
90        request: RolePermissionsUpdateXRequest,
91    ) -> Result<RoleXResponse, ApiError> {
92        let role = self
93            .roles
94            .update_permissions(UpdateRolePermissionsCommand {
95                context: context.clone(),
96                id,
97                permissions: request.permissions.unwrap_or_default(),
98            })
99            .await
100            .map_err(to_api)?;
101
102        self.read(context, role.id().to_owned()).await
103    }
104}
105
106/// Traduz a recusa do serviço de papéis no status que o cliente recebe.
107fn to_api(error: RoleError) -> ApiError {
108    match error {
109        RoleError::Missing(id) => {
110            ApiError::new(StatusCode::NOT_FOUND, format!("Role {id} was not found."))
111        }
112        RoleError::App(shared) => ApiError::of_app(shared),
113    }
114}
