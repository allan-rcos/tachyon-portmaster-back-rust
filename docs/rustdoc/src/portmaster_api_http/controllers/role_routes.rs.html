6/// A tabela de papel.
7pub(crate) fn routes<C: RoleController>(controller: C) -> Vec<Route> {
8    let list = controller.clone();
9    let create = controller.clone();
10
11    vec![
12        Route::get("/roles", move |params| list.clone().list(params)),
13        Route::post("/roles", move |body| create.clone().create(body)),
14        Route::put("/roles/{id}/permissions", move |id, body| {
15            controller.clone().update_permissions(id, body)
16        }),
17    ]
18}
