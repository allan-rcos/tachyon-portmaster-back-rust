1//! O controller do painel. Não sai do módulo.
2
3use portmaster_app::context::UserContext;
4use portmaster_app::queries::metrics::GetMetricsQuery;
5use portmaster_app::services::MetricsUseCase;
6
7use crate::controllers::metrics_controller::MetricsController;
8use crate::error::api_error::ApiError;
9use crate::wire::vo::metrics::metrics_x_response::MetricsXResponse;
10
11/// Os handlers do painel, genéricos sobre o caso de uso.
12#[derive(Clone)]
13pub(crate) struct MetricsControllerImpl<M> {
14    /// O caso de uso do painel.
15    metrics: M,
16}
17
18impl<M: MetricsUseCase> MetricsControllerImpl<M> {
19    /// Monta o controller.
20    pub(crate) const fn new(metrics: M) -> Self {
21        Self { metrics }
22    }
23}
24
25impl<M: MetricsUseCase + Clone + Send + Sync + 'static> MetricsController
26    for MetricsControllerImpl<M>
27{
28    async fn get(&self, context: UserContext) -> Result<MetricsXResponse, ApiError> {
29        let view = self
30            .metrics
31            .get(GetMetricsQuery { context })
32            .await
33            .map_err(ApiError::of_app)?;
34
35        Ok(MetricsXResponse::of(view))
36    }
37}
