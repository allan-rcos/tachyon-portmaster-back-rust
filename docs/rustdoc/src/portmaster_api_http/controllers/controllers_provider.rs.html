33/// O nome do logger que o controller de sessão recebe.
34const AUTH_CHANNEL: &str = "auth";
35
36/// O nome do ambiente, para o `/info`.
37///
38/// É o único dado de [`ApiConfig`] que um controller carrega, e ele precisa
39/// estar aqui e não por argumento: quem chama [`ControllersProvider::server`] é
40/// a tabela de rotas, que não tem — nem deveria ter — a configuração em mãos.
41///
42/// `RwLock` porque é configuração, e configuração se troca: instalar de novo
43/// muda o que o `/info` responde da próxima requisição em diante.
44static ENVIRONMENT: RwLock<Option<String>> = RwLock::new(None);
45
46/// Os controllers, já costurados com os services que consomem.
47///
48/// Construídos uma vez no boot e **clonados por requisição**, ao contrário dos
49/// services, que são reconstruídos a cada chamada. É a diferença que justifica
50/// o `self` por valor nos handlers: um controller é um punhado de handles, e
51/// clonar é o que o axum faz com todo handler.
52pub(crate) struct ControllersProvider;
53
54impl ControllersProvider {
55    /// Instala o nome do ambiente que o `/info` responde.
56    ///
57    /// É a única coisa que um controller carrega da configuração desta camada,
58    /// e é por isso que ele entra sozinho: trocar o ambiente não tem por que
59    /// obrigar a passar o segredo do token junto. Quem instala o segredo é o
60    /// [`TokenProvider`], que é quem o consome.
61    pub(crate) fn install_environment(environment: String) {
62        *ENVIRONMENT.write().unwrap_or_else(PoisonError::into_inner) = Some(environment);
63    }
64
65    /// A conta do próprio usuário.
66    pub(crate) fn account() -> anyhow::Result<impl AccountController + use<> + 'static> {
67        Ok(account_controller(
68            AppProvider::account_service()?,
69            SessionContext,
70        ))
71    }
72
73    /// Sessão: login, setup, refresh e logout.
74    pub(crate) fn auth() -> anyhow::Result<impl AuthController + use<> + 'static> {
75        Ok(auth_controller(
76            AppProvider::session_service()?,
77            AppProvider::mark_service(),
78            AppProvider::random_id_generator(),
79            TokenProvider::token_service()?,
80            CookieContext,
81            AppProvider::logger_factory().create(AUTH_CHANNEL),
82            SessionPolicy::REFRESH_TTL.as_secs(),
83        ))
84    }
85
86    /// Contêineres.
87    pub(crate) fn container() -> anyhow::Result<impl ContainerController + use<> + 'static> {
88        Ok(container_controller(
89            AppProvider::container_service()?,
90            SessionContext,
91        ))
92    }
93
94    /// Carga e telemetria.
95    pub(crate) fn manifest() -> anyhow::Result<impl ManifestController + use<> + 'static> {
96        Ok(manifest_controller(
97            AppProvider::manifest_service()?,
98            SessionContext,
99        ))
100    }
101
102    /// Metadados de sistema.
103    pub(crate) fn metadata() -> impl MetadataController + use<> + 'static {
104        metadata_controller(AppProvider::metadata_service(), SessionContext)
105    }
106
107    /// O painel do pátio.
108    pub(crate) fn metrics() -> anyhow::Result<impl MetricsController + use<> + 'static> {
109        Ok(metrics_controller(
110            AppProvider::metrics_service()?,
111            SessionContext,
112        ))
113    }

116    pub(crate) fn product() -> anyhow::Result<impl ProductController + use<> + 'static> {
117        Ok(product_controller(
118            AppProvider::product_service()?,
119            SessionContext,
120        ))
121    }
122
123    /// Papéis.
124    pub(crate) fn role() -> anyhow::Result<impl RoleController + use<> + 'static> {
125        Ok(role_controller(
126            AppProvider::role_service()?,
127            SessionContext,
128        ))
129    }
130
131    /// O estado do próprio processo.
132    ///
133    /// Sem o nome do ambiente instalado, vale o padrão de [`ApiConfig`] — o
134    /// mesmo que valeria se a variável não estivesse no ambiente. É só o que o
135    /// `/info` mostra, e um `/info` que não responde seria pior do que um que
136    /// responde o padrão.
137    pub(crate) fn server() -> impl ServerController + use<> + 'static {
138        server_controller(
139            ENVIRONMENT
140                .read()
141                .unwrap_or_else(PoisonError::into_inner)
142                .clone()
143                .unwrap_or_else(|| ApiConfig::default().environment),
144        )
145    }
146
147    /// Usuários.
148    pub(crate) fn user() -> anyhow::Result<impl UserController + use<> + 'static> {
149        Ok(user_controller(
150            AppProvider::user_service()?,
151            SessionContext,
152        ))
153    }
