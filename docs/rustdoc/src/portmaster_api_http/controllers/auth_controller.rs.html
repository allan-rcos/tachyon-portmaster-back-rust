4use crate::wire::api_response::ApiResponse;

6use crate::wire::vo::auth::login_x_request::LoginXRequest;

10/// Os handlers de sessão.
11///
12/// A sessão **é** um par de cookies, e quem decide o que entra neles é quem
13/// emite o token — este controller. Isso não aparece na assinatura: devolver
14/// `Vec<Cookie<'static>>` poria o tipo interno do crate `cookie` num contrato e
15/// obrigaria cada rota a dobrar aquele vetor sobre a resposta. Ele escreve pela
16/// [`CookiePort`](crate::middleware::cookie_port::CookiePort), e o middleware
17/// carimba o que foi escrito.
18///
19/// Pela mesma razão, `refresh` e `logout` não recebem mais os cabeçalhos: eles
20/// pediam a `HeaderMap` inteira porque só a impl de cookie sabia sob que nome
21/// procurar. A porta sabe, então o argumento não precisa existir.
22#[trait_variant::make(Send)]
23pub(crate) trait AuthController: Clone + Sync + 'static {
24    /// Declara, no boot, o grupo de marcador em que as sessões de refresh vivem.
25    ///
26    /// É aqui e não no `app` porque este controller é o único que marca naquele
27    /// grupo, e é ele quem sabe o nome dele — a camada de aplicação é agnóstica
28    /// de sessão. Precisa rodar antes da primeira requisição: o repositório de
29    /// marcadores recusa marcar num grupo que não conhece.
30    async fn register_marker_group(&self) -> Result<(), ApiError>;
31
32    /// `POST /setup`
33    ///
34    /// Abre uma vez na vida de um deploy: cria o primeiro usuário e já o loga.
35    async fn setup(self, request: Body<SetupXRequest>) -> ApiResponse<LoginXResponse>;
36
37    /// `POST /auth/login`
38    async fn login(self, request: Body<LoginXRequest>) -> ApiResponse<LoginXResponse>;
39
40    /// `POST /auth/refresh`
41    ///
42    /// Não devolve corpo: o par novo viaja nos cookies, e a resposta é um `204`.
43    async fn refresh(self) -> ApiResponse;
44
45    /// `POST /auth/logout`
46    ///
47    /// Sair é sempre possível — a revogação do refresh é esforço, não condição,
48    /// e o que pode falhar aqui é só escrever os cookies que apagam a sessão.
49    async fn logout(self) -> ApiResponse;
50}
