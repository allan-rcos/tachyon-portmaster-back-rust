1//! O erro como o cliente o vê.
2//!
3//! Esta é a **única** camada que conhece status HTTP. O `domain` devolve erro
4//! tipado, o `app` o une num [`AppError`] e o agrupa por natureza, e a tradução
5//! para número acontece aqui — em [`ApiError::of_app`], num lugar só. Espalhar o
6//! código pelo sistema foi o que o PHP fazia (o `LeafContext` carregava o status
7//! desde o domínio), e o preço era que a mesma violação de regra tinha um status
8//! gravado nela mesmo quando a saída não era HTTP.
9//!
10//! ## O corpo de erro é negociado como qualquer outro
11//!
12//! Um erro vira [`ProblemX`], que é um VO de resposta como qualquer outro, e sai
13//! pela [`EncodePort`](crate::middleware::encode_port::EncodePort) da
14//! requisição — em JSON ou em `FlatBuffers`, conforme o `Accept`. Antes o corpo
15//! era `application/problem+json` fixo, o que num sistema cujo cliente de
16//! produção fala `FlatBuffers` significava que todo caminho de erro entregava
17//! algo que ele não sabia ler.
18//!
19//! Quando não há requisição de onde negociar — um erro que nasce antes de
20//! qualquer cabeçalho ser lido — vale JSON, que é o padrão do contexto fora do
21//! escopo.
22

33/// Um erro pronto para virar resposta.
34///
35/// Um status e uma explicação, e nada mais. Ele carregava cookies, porque uma
36/// recusa às vezes precisa mexer na sessão — um refresh token morto tem que sair
37/// do navegador junto com o `401`, senão o cliente o reapresenta a cada
38/// tentativa e nunca chega ao login. Isso continua acontecendo, só que pela
39/// [`CookiePort`](crate::middleware::cookie_port::CookiePort): quem descobriu
40/// que o token morreu escreve o cookie ali, e o layer o carimba na resposta
41/// qualquer que ela seja.
42#[derive(Debug)]
43pub struct ApiError {
44    /// O status HTTP da resposta.
45    status: StatusCode,
46    /// O que aconteceu, em texto, para o corpo do problema.
47    detail: String,
48}
49
50impl ApiError {
51    /// Um erro com status e explicação.
52    pub(crate) fn new(status: StatusCode, detail: impl Into<String>) -> Self {
53        Self {
54            status,
55            detail: detail.into(),
56        }
57    }
58
59    /// Rota protegida sem sessão.
60    ///
61    /// O **401 é o único status que nasce nesta camada**: é a ausência de
62    /// sessão, e só quem lê o token sabe disso. Permissão (403), validação (422)
63    /// e conflito (409) vêm todos do `app`.
64    pub(crate) fn unauthenticated() -> Self {
65        Self::new(
66            StatusCode::UNAUTHORIZED,
67            "Authentication is required to access this resource.",
68        )
69    }
70
71    /// Corpo que não deu para ler no formato anunciado.
72    ///
73    /// **400, e não o 404 que o PHP passou a devolver.** O 404 é afirmado pela
74    /// suíte Go em cinco pontos para recurso genuinamente ausente; colidir os
75    /// dois significaria que nem um painel nem uma política de retry conseguem
76    /// separar "essa rota não existe" de "seus bytes são lixo".
77    ///
78    /// Cobre também o corpo **vazio**: ausência é um caso de ilegibilidade, não
79    /// um caso à parte.
80    pub(crate) fn unreadable_body(detail: impl Into<String>) -> Self {
81        Self::new(StatusCode::BAD_REQUEST, detail)
82    }
83
84    /// Resposta que existe mas não coube no fio.
85    ///
86    /// 502 e não 500: distingue "a resposta foi montada e falhou ao serializar"
87    /// do 500 que o middleware `Recover` produz quando algo entrou em pânico. Os
88    /// dois são defeito nosso, mas levam a investigações diferentes — e a suíte
89    /// Go não afirma 502 em ponto nenhum, então o código está livre.
90    pub(crate) fn unrenderable(detail: impl Into<String>) -> Self {
91        Self::new(StatusCode::BAD_GATEWAY, detail)
92    }
93
94    /// O status, para quem precisa decidir sobre ele antes de responder.
95    ///
96    /// Só os testes perguntam: no caminho de produção o erro vira resposta
97    /// direto, sem ninguém inspecioná-lo pelo meio.
98    #[cfg(test)]
99    pub(crate) const fn status(&self) -> StatusCode {
100        self.status
101    }
102
103    /// Desmonta o erro no status e no corpo de problema.
104    pub(crate) fn into_parts(self) -> (StatusCode, ProblemX) {
105        let problem = ProblemX {
106            kind: "about:blank",
107            title: self.status.canonical_reason().unwrap_or("Error").to_owned(),
108            status: i32::from(self.status.as_u16()),
109            detail: self.detail,
110        };
111
112        (self.status, problem)
113    }
114
115    /// Traduz o erro **comum** do `app` para o status que o cliente recebe.
116    ///
117    /// São as três recusas que qualquer caso de uso pode devolver, e a tradução
118    /// delas mora num lugar só. As recusas próprias de um serviço — "produto não
119    /// encontrado", "esse contêiner não está selado" — são casadas pelo
120    /// controller que chamou aquele caso de uso, que é quem sabe o que cada uma
121    /// significa na rota dele.
122    ///
123    /// ## Validação vem em lote
124    ///
125    /// O domínio acumula **todos** os campos em vez de parar no primeiro.
126    /// Juntá-los num `detail` só é o que o `ProblemDetails` do schema comporta:
127    /// ele não tem `details[]`, e o `.fbs` é contrato com o cliente, não algo a
128    /// alterar por conveniência.
129    ///
130    /// ## O slug da permissão negada não vai no corpo
131    ///
132    /// Dizer ao cliente qual permissão faltou descreve para ele o mapa de
133    /// autorização do sistema. Quem precisa do detalhe é o operador, e para esse
134    /// ele já está no log.
135    ///
136    /// ## Falha de infra não descreve a topologia
137    ///
138    /// O que a infra reportou fica no log com a cadeia inteira; o cliente recebe
139    /// só que houve falha. O motivo real ("a conexão com o banco foi recusada")
140    /// não é acionável para ele.
141    pub fn of_app(error: AppError) -> Self {
142        let status = match error.kind() {
143            AppErrorKind::Validation => StatusCode::UNPROCESSABLE_ENTITY,
144            AppErrorKind::Authorization => StatusCode::FORBIDDEN,
145            AppErrorKind::Internal => StatusCode::INTERNAL_SERVER_ERROR,
146        };
147
148        let detail = match error {
149            AppError::Validation(fields) => describe_fields(&fields),
150
151            AppError::PermissionDenied { permission } => {
152                SystemLogger::get().info(
153                    "acesso negado por falta de permissão",
154                    [("permission", permission)],
155                );
156
157                "You do not have permission to perform this action.".to_owned()
158            }
159
160            AppError::Infra(cause) => {
161                SystemLogger::get().error(
162                    "falha de infraestrutura",
163                    [("error", &format!("{cause:?}"))],
164                );
165
166                "An unexpected error occurred.".to_owned()
167            }
168        };
169
170        Self::new(status, detail)
171    }
172}

174impl IntoResponse for ApiError {
175    /// Codifica o problema no formato que a requisição negociou.
176    ///
177    /// Este caminho é o da recusa que vira resposta **sozinha** — a de um
178    /// extractor, ou a de um middleware. O erro não carrega mais um encoder de
179    /// reserva anexado à mão: o formato está no escopo da requisição, e a
180    /// [`EncodePort`](crate::middleware::encode_port::EncodePort) o alcança de
181    /// onde quer que este erro tenha nascido.
182    ///
183    /// É o único ponto do sistema que constrói o adaptador em vez de recebê-lo
184    /// injetado, e não há alternativa: `IntoResponse::into_response` não recebe
185    /// argumento nenhum. O adaptador é um ZST, então construí-lo não é nada.
186    fn into_response(self) -> Response {
187        let (status, problem) = self.into_parts();
188
189        EncodeContext.respond(status, &problem)
190    }
191}

193/// Resume os campos recusados numa linha.
194fn describe_fields(fields: &[FieldError]) -> String {
195    if fields.is_empty() {
196        return "dados inválidos".to_owned();
197    }
198
199    fields
200        .iter()
201        .map(ToString::to_string)
202        .collect::<Vec<_>>()
203        .join("; ")
204}
