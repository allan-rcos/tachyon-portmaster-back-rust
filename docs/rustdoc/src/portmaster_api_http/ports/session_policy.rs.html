1//! Quanto a sessão vale, e sob que política os cookies dela viajam.
2
3use std::time::Duration;
4
5use cookie::SameSite;
6
7/// A política de sessão, decidida em tempo de compilação.
8///
9/// Nada disto é segredo nem identidade de deploy — é **arquitetura**, e é por
10/// isso que saiu do ambiente. Um `if` em produção sobre quanto tempo uma sessão
11/// dura é um bug esperando o dia de errar, e o que se ganhava em troca era a
12/// possibilidade de dois deploys da mesma versão discordarem sobre o que a API
13/// promete.
14///
15/// ## Uma duração só para o token e para o cookie
16///
17/// [`Self::ACCESS_TTL`] é ao mesmo tempo o `exp` da claim e o `Max-Age` do
18/// cookie que a carrega, e isso não é economia: eram duas variáveis de ambiente
19/// (`APP_JWT_TTL` e o max-age do cookie) que **podiam divergir**, e divergindo
20/// produziam um cookie que o navegador guarda depois de o token dentro dele já
21/// não valer — ou o contrário, um token bom que o navegador jogou fora.
22///
23/// ## `Secure` segue o perfil de compilação
24///
25/// Release liga; debug não. O ambiente de desenvolvimento roda em HTTP puro, e
26/// um cookie `Secure` simplesmente não seria enviado — a sessão nunca
27/// funcionaria localmente. A imagem de contêiner é compilada em release, então
28/// um build local que precise servir HTTP puro passa
29/// `RUSTFLAGS="-C debug-assertions=on"`, que é o que o `Dockerfile` faz sob o
30/// `ARG RUST_DEBUG_ASSERTIONS` e o que a suíte de integração usa.
31///
32/// Struct-namespace e não um `mod` de consts soltas: o módulo já é o arquivo, e
33/// agrupá-las num tipo é o que mantém um export só por arquivo.
34pub(crate) struct SessionPolicy;
35
36impl SessionPolicy {
37    /// Quanto vale um access token — e o cookie que o carrega.
38    pub(crate) const ACCESS_TTL: Duration = Duration::from_secs(3600);
39
40    /// Quanto vale um refresh token — e o cookie que o carrega.
41    ///
42    /// Quatorze dias, como o PHP.
43    pub(crate) const REFRESH_TTL: Duration = Duration::from_secs(1_209_600);
44
45    /// Se os cookies de sessão exigem HTTPS.
46    pub(crate) const SECURE: bool = !cfg!(debug_assertions);
47
48    /// A política `SameSite` dos cookies de sessão.
49    ///
50    /// `Strict` porque nenhuma navegação de terceiro precisa chegar autenticada:
51    /// o front é uma aplicação só, e o que `Lax` liberaria é justamente o
52    /// vetor de CSRF que não temos motivo para abrir.
53    pub(crate) const SAME_SITE: SameSite = SameSite::Strict;
54}
