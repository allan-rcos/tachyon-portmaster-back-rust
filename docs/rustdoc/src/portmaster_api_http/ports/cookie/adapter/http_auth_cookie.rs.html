10/// Emite e lê os cookies de sessão.
11///
12/// Construídos pelo crate `cookie`, e não por `format!`. O que se ganha não é
13/// concisão: é `SameSite` virar um enum em vez de uma string que ninguém valida,
14/// `Max-Age` virar uma duração tipada, e a leitura do cabeçalho `Cookie` deixar
15/// de ser um parser nosso — que tratava `a=1; b=2` bem e o resto da gramática
16/// não.
17///
18/// Não guarda nada. Nome, validade, `Secure` e `SameSite` são decididos em
19/// compilação pela [`SessionPolicy`] e pelo [`CookieName`]; antes eram seis
20/// campos copiados da configuração, e seis variáveis de ambiente que podiam
21/// discordar entre si.
22#[derive(Debug, Clone, Copy, Default)]
23pub(crate) struct HttpAuthCookie;
24
25impl HttpAuthCookie {
26    /// Monta o emissor de cookies.
27    pub(crate) const fn new() -> Self {
28        Self
29    }
30
31    /// Monta o cookie com a política que vale para os dois.
32    fn build(name: CookieName, value: &str, max_age: std::time::Duration) -> Cookie<'static> {
33        Cookie::build((name.as_str(), value.to_owned()))
34            .path("/")
35            .max_age(as_cookie_duration(max_age))
36            .http_only(true)
37            .secure(SessionPolicy::SECURE)
38            .same_site(SessionPolicy::SAME_SITE)
39            .build()
40    }
41
42    /// O cookie que apaga um nome.
43    ///
44    /// Reaproveita o [`Self::build`] com valor vazio e validade zero: apagar é
45    /// emitir o mesmo cookie já vencido, e escrever isso à parte abriria a chance
46    /// de o cookie de logout sair com `Path` ou `SameSite` diferente do de login
47    /// — caso em que o navegador guarda os dois e a sessão não morre.
48    fn clear(name: CookieName) -> Cookie<'static> {
49        Self::build(name, "", std::time::Duration::ZERO)
50    }
51
52    /// Procura um cookie pelo nome no cabeçalho `Cookie`.
53    ///
54    /// Um cookie presente e vazio conta como ausente: é o que um `Max-Age=0`
55    /// deixa para trás em alguns clientes, e tratá-lo como valor faria o
56    /// logout parecer não ter funcionado.
57    fn read(headers: &HeaderMap, name: CookieName) -> Option<String> {
58        headers
59            .get_all(header::COOKIE)
60            .iter()
61            .filter_map(|value| value.to_str().ok())
62            .flat_map(Cookie::split_parse)
63            .filter_map(Result::ok)
64            .find(|cookie| cookie.name() == name.as_str())
65            .map(|cookie| cookie.value().to_owned())
66            .filter(|value| !value.is_empty())
67    }
68}
69
70impl AuthCookie for HttpAuthCookie {
71    fn issue_access(&self, token: &str) -> Cookie<'static> {
72        Self::build(CookieName::Access, token, SessionPolicy::ACCESS_TTL)
73    }
74
75    fn issue_refresh(&self, token: &str) -> Cookie<'static> {
76        Self::build(CookieName::Refresh, token, SessionPolicy::REFRESH_TTL)
77    }
78
79    fn clear_access(&self) -> Cookie<'static> {
80        Self::clear(CookieName::Access)
81    }
82
83    fn clear_refresh(&self) -> Cookie<'static> {
84        Self::clear(CookieName::Refresh)
85    }
86
87    fn read_access(&self, headers: &HeaderMap) -> Option<String> {
88        Self::read(headers, CookieName::Access)
89    }
90
91    fn read_refresh(&self, headers: &HeaderMap) -> Option<String> {
92        Self::read(headers, CookieName::Refresh)
93    }
94}
95
96/// A duração em segundos, saturada na faixa do crate `cookie`.
97fn as_cookie_duration(ttl: std::time::Duration) -> cookie::time::Duration {
98    cookie::time::Duration::seconds(i64::try_from(ttl.as_secs()).unwrap_or(i64::MAX))
99}

101#[cfg(test)]
102mod tests {
103    use super::*;
104    use axum::http::HeaderValue;
105    use pretty_assertions::assert_eq;
106
107    /// `HttpOnly` é o que separa um XSS que incomoda de um que rouba a sessão.
108    #[test]
109    fn o_cookie_de_sessao_e_http_only() {
110        let cookie = HttpAuthCookie::new().issue_access("jwt.de.mentira");
111
112        assert_eq!(cookie.http_only(), Some(true));
113        assert_eq!(cookie.secure(), Some(SessionPolicy::SECURE));
114        assert_eq!(cookie.same_site(), Some(SessionPolicy::SAME_SITE));
115    }
116
117    /// Em desenvolvimento o servidor é HTTP puro, e um cookie `Secure` nunca
118    /// chegaria — a sessão simplesmente não funcionaria localmente.
119    #[test]
120    fn secure_segue_o_perfil_de_compilacao() {
121        assert_eq!(
122            HttpAuthCookie::new().issue_access("t").secure(),
123            Some(!cfg!(debug_assertions))
124        );
125    }
126
127    #[test]
128    fn limpar_expira_o_cookie_na_hora() {
129        let cookie = HttpAuthCookie::new().clear_refresh();
130
131        assert_eq!(cookie.value(), "");
132        assert_eq!(cookie.max_age(), Some(cookie::time::Duration::ZERO));
133    }
134
135    /// Apagar tem que produzir o mesmo cookie que emitir, menos o valor: se o
136    /// `Path` ou o `SameSite` divergirem, o navegador guarda os dois.
137    #[test]
138    fn o_cookie_que_apaga_casa_com_o_que_emite() {
139        let issued = HttpAuthCookie::new().issue_access("t");
140        let cleared = HttpAuthCookie::new().clear_access();
141
142        assert_eq!(cleared.name(), issued.name());
143        assert_eq!(cleared.path(), issued.path());
144        assert_eq!(cleared.same_site(), issued.same_site());
145        assert_eq!(cleared.secure(), issued.secure());
146    }
147
148    #[test]
149    fn o_cookie_apresentado_e_encontrado_entre_outros() {
150        let mut headers = HeaderMap::new();
151        headers.insert(
152            header::COOKIE,
153            HeaderValue::from_static("theme=dark; auth_token=abc123; other=1"),
154        );
155
156        assert_eq!(
157            HttpAuthCookie::new().read_access(&headers),
158            Some("abc123".to_owned())
159        );
160    }
161
162    /// É o que um `Max-Age=0` deixa para trás em alguns clientes; tratá-lo como
163    /// valor faria o logout parecer não ter funcionado.
164    #[test]
165    fn cookie_vazio_conta_como_ausente() {
166        let mut headers = HeaderMap::new();
167        headers.insert(header::COOKIE, HeaderValue::from_static("auth_token="));
168
169        assert_eq!(HttpAuthCookie::new().read_access(&headers), None);
170    }
