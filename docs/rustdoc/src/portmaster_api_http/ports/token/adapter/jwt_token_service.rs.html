1//! O token de sessão como JWT. Não sai do módulo.
2//!
3//! ## O principal viaja como `FlatBuffers` dentro da claim
4//!
5//! O JWT não carrega claims soltas por usuário. O principal inteiro — id, nome,
6//! e-mail, papéis e permissões — é serializado na tabela `TokenUser` de
7//! `token.fbs`, codificado em base64url e posto numa claim só. Duas razões: o
8//! token fica menor que o JSON equivalente, e o payload deixa de ser legível
9//! para quem espia o cookie por curiosidade.
10//!
11//! Não é sigilo — base64 não esconde nada de quem queira ler. O que a assinatura
12//! garante é **integridade**: o conteúdo não pode ser alterado sem o segredo.
13

15use base64::Engine as _;
16use chrono::Utc;
17use jsonwebtoken::{Algorithm, DecodingKey, EncodingKey, Header, Validation};
18use portmaster_app::context::{RoleContext, UserContext};
19use portmaster_app::domain::User;

21use serde::{Deserialize, Serialize};

29/// As claims do access token.
30#[derive(Debug, Serialize, Deserialize)]
31struct Claims {
32    /// Id do usuário, em base62.
33    sub: String,
34    /// Quem emitiu.
35    iss: String,
36    /// Quando expira, em epoch de segundos.
37    exp: u64,
38    /// Quando foi emitido.
39    iat: u64,
40    /// O principal inteiro, em `FlatBuffers` e base64url.
41    user_data: String,
42}
43
44/// Emite e confere tokens assinados em HS256.
45#[derive(Clone)]
46pub(crate) struct JwtTokenService {
47    /// A chave de assinatura.
48    encoding: EncodingKey,
49    /// A chave de verificação.
50    decoding: DecodingKey,
51    /// Quem emitiu, gravado e conferido na claim `iss`.
52    issuer: String,
53}
54
55impl JwtTokenService {
56    /// Monta o serviço a partir da configuração.
57    pub(crate) fn new(config: &JwtConfig) -> Self {
58        let secret = config.secret.expose_secret().as_bytes();
59
60        Self {
61            encoding: EncodingKey::from_secret(secret),
62            decoding: DecodingKey::from_secret(secret),
63            issuer: config.issuer.clone(),
64        }
65    }
66
67    /// O agora, em epoch de segundos.
68    ///
69    /// `Utc::now()` direto, e não um relógio injetado: quem confere o `exp` é o
70    /// `jsonwebtoken`, contra o relógio de parede real, e não há onde injetar
71    /// nada ali. Um relógio trocável na emissão e fixo na conferência não tornava
72    /// a expiração testável — só dava essa impressão.
73    fn epoch_seconds() -> u64 {
74        u64::try_from(Utc::now().timestamp()).unwrap_or_default()
75    }
76
77    /// Serializa o principal na tabela `TokenUser`, em base64url.
78    ///
79    /// Método privado e não linha solta no [`Self::issue`]: são vinte linhas de
80    /// coreografia de builder cujo ponto — o que entra no token e o que fica de
81    /// fora — merece o nome e a explicação acima.
82    fn pack_principal(user: &dyn User) -> String {
83        let token_user = fbs::token::TokenUser {
84            id: Some(user.id().to_owned()),
85            name: Some(user.name().to_owned()),
86            email: Some(user.email().to_owned()),
87            roles: Some(
88                user.roles()
89                    .iter()
90                    .map(|role| fbs::token::TokenRole {
91                        id: Some(role.id().to_owned()),
92                        name: Some(role.name().to_owned()),
93                        permissions: Some(role.permissions().to_vec()),
94                    })
95                    .collect(),
96            ),
97        };
98
99        let mut builder = planus::Builder::new();
100
101        URL_SAFE_NO_PAD.encode(builder.finish(&token_user, None))
102    }
103
104    /// Desfaz o [`Self::pack_principal`].
105    ///
106    /// Tolera todo campo ausente menos os ids: um papel sem id não identifica
107    /// nada, e um principal sem id não é ninguém. Nome e permissões vazios
108    /// degradam a sessão sem invalidá-la.
109    fn unpack_principal(packed: &str) -> Option<UserContext> {
110        use planus::ReadAsRoot as _;
111
112        let bytes = URL_SAFE_NO_PAD.decode(packed).ok()?;
113        let token_user = fbs::token::TokenUserRef::read_as_root(&bytes).ok()?;
114
115        let roles = token_user
116            .roles()
117            .ok()
118            .flatten()
119            .map(|roles| {
120                roles
121                    .iter()
122                    .filter_map(|role| {
123                        let role = role.ok()?;
124
125                        Some(RoleContext {
126                            id: role.id().ok().flatten()?.to_owned(),
127                            name: role.name().ok().flatten().unwrap_or_default().to_owned(),
128                            permissions: role
129                                .permissions()
130                                .ok()
131                                .flatten()
132                                .map(|slugs| {
133                                    slugs
134                                        .iter()
135                                        .filter_map(|slug| Some(slug.ok()?.to_owned()))
136                                        .collect()
137                                })
138                                .unwrap_or_default(),
139                        })
140                    })
141                    .collect()
142            })
143            .unwrap_or_default();
144
145        Some(UserContext {
146            id: token_user.id().ok().flatten()?.to_owned(),
147            name: token_user
148                .name()
149                .ok()
150                .flatten()
151                .unwrap_or_default()
152                .to_owned(),
153            email: token_user
154                .email()
155                .ok()
156                .flatten()
157                .unwrap_or_default()
158                .to_owned(),
159            roles,
160        })
161    }
162}
163
164impl TokenService for JwtTokenService {
165    fn issue(&self, user: &dyn User) -> Result<String, ApiError> {
166        let now = Self::epoch_seconds();
167
168        #[allow(
169            clippy::arithmetic_side_effects,
170            reason = "epoch em segundos mais um TTL de minutos: estourar u64 são ~500 bilhões de anos"
171        )]
172        let claims = Claims {
173            sub: user.id().to_owned(),
174            iss: self.issuer.clone(),
175            exp: now + SessionPolicy::ACCESS_TTL.as_secs(),
176            iat: now,
177            user_data: Self::pack_principal(user),
178        };
179
180        jsonwebtoken::encode(&Header::new(Algorithm::HS256), &claims, &self.encoding).map_err(|e| {
181            ApiError::new(axum::http::StatusCode::INTERNAL_SERVER_ERROR, e.to_string())
182        })
183    }
184
185    /// Confere a assinatura e devolve o principal que o token carrega.
186    ///
187    /// ## Sem tolerância de relógio
188    ///
189    /// O padrão da lib são 60 segundos, pensados para quando quem assina e quem
190    /// confere são máquinas diferentes — aqui são o mesmo processo, com o mesmo
191    /// relógio. Mantê-la só faria todo token valer um minuto a mais do que o
192    /// `exp` que ele mesmo declara.
193    fn verify(&self, token: &str) -> Option<UserContext> {
194        let mut validation = Validation::new(Algorithm::HS256);
195        validation.set_issuer(&[&self.issuer]);
196
197        validation.leeway = 0;
198
199        let claims = jsonwebtoken::decode::<Claims>(token, &self.decoding, &validation)
200            .ok()?
201            .claims;
202
203        Self::unpack_principal(&claims.user_data)
204    }
205}
