12/// O mínimo que o HS256 aceita como segredo.
13///
14/// Um segredo menor que o digest não acrescenta entropia à assinatura: dá para
15/// enumerar o espaço de chaves mais rápido do que forjar o hash.
16const MIN_JWT_SECRET_BYTES: usize = 32;
17
18/// Tudo que o processo precisa para subir.
19#[derive(Debug, Clone)]
20pub struct Secrets {
21    /// O que é desta camada.
22    pub api: ApiConfig,
23    /// O que é das camadas de baixo.
24    pub app: AppSecrets,
25}
26
27impl Secrets {
28    /// Lê o ambiente e monta a configuração de todas as camadas.
29    ///
30    /// Falta de segredo obrigatório derruba o boot. É deliberado: um sistema que
31    /// sobe com o JWT num valor padrão aceita tokens que qualquer um pode
32    /// forjar, e descobrir isso em produção é caro demais comparado a não subir.
33    pub fn load() -> anyhow::Result<Self> {
34        let jwt_secret = required(Env::JWT_SECRET)?;
35
36        anyhow::ensure!(
37            jwt_secret.len() >= MIN_JWT_SECRET_BYTES,
38            "{} precisa de ao menos {MIN_JWT_SECRET_BYTES} bytes: um segredo menor que o \
39             digest do HS256 não acrescenta entropia à assinatura",
40            Env::JWT_SECRET
41        );
42
43        Ok(Self {
44            api: ApiConfig {
45                host: optional(Env::HOST).unwrap_or_else(|| "127.0.0.1".to_owned()),
46                port: parsed(Env::PORT, 8000)?,
47                environment: optional(Env::ENVIRONMENT).unwrap_or_else(|| "development".to_owned()),
48                request_timeout: Duration::from_secs(parsed(Env::REQUEST_TIMEOUT, 30)?),
49                cors_origins: optional(Env::CORS_ORIGINS)
50                    .map(|list| {
51                        list.split(',')
52                            .map(str::trim)
53                            .filter(|origin| !origin.is_empty())
54                            .map(str::to_owned)
55                            .collect()
56                    })
57                    .unwrap_or_default(),
58                jwt: JwtConfig {
59                    secret: SecretString::from(jwt_secret),
60                    ttl: Duration::from_secs(parsed(Env::JWT_TTL, 3600)?),
61                    issuer: optional(Env::JWT_ISSUER)
62                        .unwrap_or_else(|| "tachyon/portmaster".to_owned()),
63                    cookie_name: optional(Env::JWT_COOKIE_NAME)
64                        .unwrap_or_else(|| "auth_token".to_owned()),
65                    cookie_secure: flag(Env::JWT_COOKIE_SECURE, false),
66                    cookie_same_site: optional(Env::JWT_COOKIE_SAME_SITE)
67                        .unwrap_or_else(|| "Strict".to_owned()),
68                    refresh_cookie_name: optional(Env::REFRESH_COOKIE_NAME)
69                        .unwrap_or_else(|| "refresh_token".to_owned()),
70                    // Quatorze dias, como o PHP.
71                    refresh_ttl: Duration::from_secs(parsed(Env::REFRESH_TTL, 1_209_600)?),
72                },
73            },
74            app: AppSecrets {
75                domain: DomainSecrets {
76                    cluster_id: parsed(Env::CLUSTER_ID, 0)?,
77                    server_id: parsed(Env::SERVER_ID, 0)?,
78                },
79                infra: InfraSecrets {
80                    database_uri: SecretString::from(database_uri()?),
81                    ssl_mode: ssl_mode()?,
82                    ssl_ca_path: optional(Env::DB_SSL_CA),
83                    ssl_verify_hostname: flag(Env::DB_SSL_VERIFY_CN, true),
84                },
85            },
86        })
87    }
88}

90/// Monta a URI de conexão a partir das partes.
91///
92/// Partes soltas e não uma URI inteira porque é assim que o `docker-compose` e o
93/// PHP já as expõem — trocar por uma variável única obrigaria quem faz deploy a
94/// reescrever a configuração para ganhar nada.
95fn database_uri() -> anyhow::Result<String> {
96    let user = required(Env::DB_USER)?;
97    let password = required(Env::DB_PASSWORD)?;
98    let host = optional(Env::DB_HOST).unwrap_or_else(|| "127.0.0.1".to_owned());
99    let port: u16 = parsed(Env::DB_PORT, 3306)?;
100    let name = required(Env::DB_NAME)?;
101
102    Ok(format!("mysql://{user}:{password}@{host}:{port}/{name}"))
103}
104
105/// Traduz o modo de TLS pedido.
106fn ssl_mode() -> anyhow::Result<DatabaseSslMode> {
107    let Some(raw) = optional(Env::DB_SSL_MODE) else {
108        return Ok(DatabaseSslMode::Disabled);
109    };
110
111    match raw.to_ascii_lowercase().as_str() {
112        "disabled" | "disable" | "off" | "" => Ok(DatabaseSslMode::Disabled),
113        "required" | "require" | "on" => Ok(DatabaseSslMode::Required),
114        "verify_ca" | "verify-ca" => Ok(DatabaseSslMode::VerifyCa),
115        other => anyhow::bail!(
116            "{} não reconhece `{other}`: use disabled, required ou verify_ca",
117            Env::DB_SSL_MODE
118        ),
119    }
120}

122/// Uma variável obrigatória.
123fn required(name: &str) -> anyhow::Result<String> {
124    optional(name).with_context(|| format!("a variável {name} é obrigatória e não está definida"))
125}
126
127/// Uma variável opcional; vazia conta como ausente.
128fn optional(name: &str) -> Option<String> {
129    std::env::var(name)
130        .ok()
131        .filter(|value| !value.trim().is_empty())
132}
133
134/// Uma variável numérica, com padrão.
135///
136/// Valor presente mas ilegível é **erro**, não queda no padrão: quem escreveu
137/// `APP_PORT=oito mil` quis dizer alguma coisa, e subir na 8000 esconderia o
138/// engano até alguém notar que o serviço está na porta errada.
139fn parsed<T: std::str::FromStr>(name: &str, default: T) -> anyhow::Result<T>
140where
141    T::Err: std::fmt::Display,
142{
143    match optional(name) {
144        None => Ok(default),
145        Some(raw) => raw
146            .parse()
147            .map_err(|e| anyhow::anyhow!("{name} não é um número válido: {raw} ({e})")),
148    }
149}
150
151/// Uma variável booleana, com padrão.
152fn flag(name: &str, default: bool) -> bool {
153    match optional(name) {
154        None => default,
155        Some(raw) => matches!(
156            raw.to_ascii_lowercase().as_str(),
157            "1" | "true" | "yes" | "on"
158        ),
159    }
160}

162#[cfg(test)]
163mod tests {

165    use pretty_assertions::assert_eq;

167    /// O mesmo modo aparece escrito de formas diferentes em compose, shell e
168    /// documentação; recusar por causa de um hífen seria hostilidade.
169    #[test]
170    fn o_modo_de_tls_aceita_as_grafias_usuais() {
171        for grafia in ["verify_ca", "verify-ca", "VERIFY_CA"] {
172            temporarily(Env::DB_SSL_MODE, Some(grafia), || {
173                assert_eq!(ssl_mode().unwrap(), DatabaseSslMode::VerifyCa);
174            });
175        }
176    }
177
178    #[test]
179    fn um_modo_de_tls_desconhecido_derruba_o_boot() {
180        // Cair no padrão aqui conectaria em claro um deploy que pediu TLS.
181        temporarily(Env::DB_SSL_MODE, Some("mais_ou_menos"), || {
182            assert!(ssl_mode().is_err());
183        });
184    }
185
186    #[test]
187    fn variavel_vazia_conta_como_ausente() {
188        // Um compose que declara a chave sem valor não deve derrotar o padrão.
189        temporarily(Env::HOST, Some("   "), || {
190            assert_eq!(optional(Env::HOST), None);
191        });
192    }
193
194    #[test]
195    fn numero_ilegivel_e_erro_e_nao_padrao() {
196        temporarily(Env::PORT, Some("oito mil"), || {
197            assert!(parsed::<u16>(Env::PORT, 8000).is_err());
198        });
199    }
200
201    #[test]
202    fn o_booleano_so_e_verdadeiro_nas_grafias_afirmativas() {
203        for afirmativo in ["1", "true", "TRUE", "yes", "on"] {
204            temporarily(Env::JWT_COOKIE_SECURE, Some(afirmativo), || {
205                assert!(flag(Env::JWT_COOKIE_SECURE, false));
206            });
207        }
208
209        for negativo in ["0", "false", "no", "qualquer coisa"] {
210            temporarily(Env::JWT_COOKIE_SECURE, Some(negativo), || {
211                assert!(!flag(Env::JWT_COOKIE_SECURE, true));
212            });
213        }
214    }
215
216    /// Serializa os testes que mexem no ambiente.
217    ///
218    /// O ambiente é global ao **processo**, e o cargo roda os testes em threads
219    /// paralelas: sem este lock, dois testes que escrevem a mesma variável se
220    /// atropelam e o que falha é sorteado a cada execução.
221    static ENV: std::sync::Mutex<()> = std::sync::Mutex::new(());
222
223    ///
224    /// `unwrap_or_else` em vez de `unwrap`: se um teste anterior entrou em
225    /// pânico segurando o lock, envenená-lo faria os outros falharem por
226    /// arrasto, escondendo qual foi o problema original.
227    /// Roda `body` com a variável definida, e a restaura depois.
228    fn temporarily(name: &str, value: Option<&str>, body: impl FnOnce()) {
229        let _guard = ENV
230            .lock()
231            .unwrap_or_else(std::sync::PoisonError::into_inner);
232
233        let previous = std::env::var(name).ok();
234
235        match value {
236            Some(value) => std::env::set_var(name, value),
237            None => std::env::remove_var(name),
238        }
239
240        body();
241
242        match previous {
243            Some(value) => std::env::set_var(name, value),
244            None => std::env::remove_var(name),
245        }
246    }
