1//! O acumulador que a chain preenche, um grupo por elo.
2
3use portmaster_app::{AppSecrets, DomainSecrets, InfraSecrets};
4
5use crate::config::api_config::ApiConfig;
6use crate::config::jwt_config::JwtConfig;
7use crate::config::secrets::Secrets;
8
9/// O estado meio construído da configuração, enquanto a chain roda.
10///
11/// Deliberadamente **não** é a [`Secrets`]: aquela é imutável e tem que estar
12/// completa. Esta é a metade do caminho, e existe só entre o primeiro elo e o
13/// [`Self::into_secrets`] — depois disso nem o draft nem a chain existem em
14/// memória.
15///
16/// Cada slot pertence a exatamente um elo. É o que dispensa ordem: os elos são
17/// independentes, ninguém lê o slot de ninguém, e acrescentar um grupo é
18/// acrescentar um campo aqui e um arquivo em
19/// [`chain`](crate::config::chain).
20#[derive(Debug, Default)]
21pub(crate) struct BootDraft {
22    /// Onde o servidor escuta, preenchido pelo `ApiChain`.
23    pub(crate) api: Option<ApiConfig>,
24    /// Token e cookies, preenchido pelo `JwtChain`.
25    pub(crate) jwt: Option<JwtConfig>,
26    /// Identidade de deploy, preenchida pelo `DomainChain`.
27    pub(crate) domain: Option<DomainSecrets>,
28    /// Conexão com o banco, preenchida pelo `DatabaseChain`.
29    pub(crate) infra: Option<InfraSecrets>,
30}
31
32impl BootDraft {
33    /// Congela o rascunho na configuração definitiva.
34    ///
35    /// Um slot vazio aqui não é variável de ambiente faltando — quem lida com
36    /// isso é o [`EnvSource`](crate::config::env_source::EnvSource), e todo elo
37    /// preenche o seu slot com padrões quando não acha nada. É elo que não
38    /// rodou, ou seja, um `#[distributed_slice]` que ninguém declarou.
39    pub(crate) fn into_secrets(self) -> anyhow::Result<Secrets> {
40        Ok(Secrets {
41            api: self.api.ok_or_else(|| missing("api"))?,
42            jwt: self.jwt.ok_or_else(|| missing("jwt"))?,
43            app: AppSecrets {
44                domain: self.domain.ok_or_else(|| missing("domain"))?,
45                infra: self.infra.ok_or_else(|| missing("infra"))?,
46            },
47        })
48    }
49}
50
51/// A recusa por um grupo que elo nenhum preencheu.
52fn missing(group: &str) -> anyhow::Error {
53    anyhow::anyhow!(
54        "o grupo de configuração \"{group}\" nunca foi preenchido: falta o elo dele em CONFIG_LINKS"
55    )
56}
