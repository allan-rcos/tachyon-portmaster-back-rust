3use linkme::distributed_slice;

5use crate::config::api_config::ApiConfig;

12/// Lê as variáveis do servidor HTTP para uma [`ApiConfig`].
13///
14/// A forma de um elo, escrita por extenso aqui e seguida por todos os outros:
15/// pegue o `default()` do VO e leia cada variável usando o campo dele como
16/// fallback. Os padrões ficam no VO e em nenhum outro lugar — este arquivo não
17/// repete um sequer, e acrescentar um campo com padrão custa a linha que o lê.
18pub(crate) struct ApiChain;
19
20impl ApiChain {
21    /// Preenche o slot `api` do rascunho.
22    fn read(env: &mut EnvSource, draft: &mut BootDraft) {
23        let defaults = ApiConfig::default();
24
25        draft.api = Some(ApiConfig {
26            host: env.string(Env::HOST, &defaults.host),
27            port: env.number(Env::PORT, defaults.port),
28            environment: env.string(Env::ENVIRONMENT, &defaults.environment),
29            request_timeout: env.duration(Env::REQUEST_TIMEOUT, defaults.request_timeout),
30            cors_origins: env.list(Env::CORS_ORIGINS, &defaults.cors_origins),
31        });
32    }
33}
34
35/// O servidor HTTP é um grupo da chain deste binário.
36#[allow(
37    unsafe_code,
38    reason = "o #[distributed_slice] expande para um static com link_section; o desvio fica no registro, e não sobe para o lib.rs"
39)]
40#[distributed_slice(CONFIG_LINKS)]
41static API: ConfigLink = ConfigLink {
42    read: ApiChain::read,
43};
