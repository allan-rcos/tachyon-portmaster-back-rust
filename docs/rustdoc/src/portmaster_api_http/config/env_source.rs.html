1//! O leitor tipado do ambiente, e a lista do que estava errado.
2
3use std::collections::HashMap;
4use std::str::FromStr;
5use std::time::Duration;
6
7/// Lê variáveis de ambiente e acumula o que não deu para ler.
8///
9/// Nenhum método falha na hora. Um valor ilegível vira uma linha na lista e a
10/// leitura segue com o padrão — é o que faz um boot recusado nomear **todas** as
11/// variáveis erradas de uma vez, em vez de obrigar quem faz deploy a corrigir
12/// uma, subir de novo, e descobrir a seguinte.
13///
14/// O ambiente é lido **uma vez**, em [`Self::of_process`], e o que os elos
15/// consultam é a cópia. O que se ganha não é desempenho: é que a leitura deixa
16/// de depender de um estado global do processo, e um teste monta a fonte com
17/// [`Self::of_pairs`] em vez de escrever em `std::env` e serializar-se contra os
18/// outros testes.
19///
20/// É o `EnvSource` do PHP, com uma diferença: lá o elo do JWT lançava exceção no
21/// segredo curto, saindo da chain no meio. Aqui ele também acumula, por
22/// [`Self::refuse`]. Um segredo fraco continua derrubando o boot — só derruba
23/// depois de os outros elos terem contado o que também está errado, que é a
24/// única coisa que a exceção precoce impedia.
25pub(crate) struct EnvSource {
26    /// O ambiente como estava quando o processo subiu.
27    vars: HashMap<String, String>,
28    /// O que não deu para ler, na ordem em que apareceu.
29    errors: Vec<String>,
30}
31
32impl EnvSource {
33    /// O leitor sobre o ambiente deste processo.
34    pub(crate) fn of_process() -> Self {
35        Self::of_pairs(std::env::vars())
36    }
37
38    /// O leitor sobre um ambiente dado.
39    ///
40    /// Variável presente e vazia é descartada aqui, na entrada: um compose que
41    /// declara a chave sem valor não deve derrotar o padrão, porque a intenção
42    /// de quem escreveu `APP_HOST:` é não escolher host nenhum.
43    pub(crate) fn of_pairs(pairs: impl IntoIterator<Item = (String, String)>) -> Self {
44        Self {
45            vars: pairs
46                .into_iter()
47                .filter(|(_, value)| !value.trim().is_empty())
48                .collect(),
49            errors: Vec::new(),
50        }
51    }
52
53    /// Uma variável de texto, com padrão.
54    pub(crate) fn string(&self, name: &str, default: &str) -> String {
55        self.raw(name).unwrap_or(default).to_owned()
56    }
57
58    /// Uma variável de texto obrigatória.
59    ///
60    /// Ausente, registra a queixa e devolve string vazia. O vazio nunca chega a
61    /// ser usado: quem lê a lista de erros recusa o boot antes de o draft virar
62    /// configuração.
63    pub(crate) fn required(&mut self, name: &str) -> String {
64        let Some(value) = self.raw(name).map(str::to_owned) else {
65            self.refuse(format!(
66                "a variável {name} é obrigatória e não está definida"
67            ));
68
69            return String::new();
70        };
71
72        value
73    }
74
75    /// Uma variável numérica, com padrão.
76    ///
77    /// Valor presente mas ilegível é **queixa**, não queda no padrão: quem
78    /// escreveu `APP_PORT=oito mil` quis dizer alguma coisa, e subir na 8000
79    /// esconderia o engano até alguém notar que o serviço está na porta errada.
80    pub(crate) fn number<T: FromStr>(&mut self, name: &str, default: T) -> T
81    where
82        T::Err: std::fmt::Display,
83    {
84        let Some(raw) = self.raw(name).map(str::to_owned) else {
85            return default;
86        };
87
88        match raw.parse() {
89            Ok(value) => value,
90            Err(error) => {
91                self.refuse(format!("{name} não é um número válido: {raw} ({error})"));
92
93                default
94            }
95        }
96    }
97
98    /// Uma variável de segundos, com padrão.
99    pub(crate) fn duration(&mut self, name: &str, default: Duration) -> Duration {
100        Duration::from_secs(self.number(name, default.as_secs()))
101    }
102
103    /// Uma variável booleana, com padrão.
104    ///
105    /// Grafia irreconhecível conta como falso em vez de virar queixa: as
106    /// afirmativas cobrem tudo que um compose ou um shell produzem, e qualquer
107    /// outra coisa é a intenção de desligar.
108    pub(crate) fn flag(&self, name: &str, default: bool) -> bool {
109        match self.raw(name) {
110            None => default,
111            Some(raw) => matches!(
112                raw.to_ascii_lowercase().as_str(),
113                "1" | "true" | "yes" | "on"
114            ),
115        }
116    }
117
118    /// Uma variável de lista separada por vírgula, com padrão.
119    pub(crate) fn list(&self, name: &str, default: &[String]) -> Vec<String> {
120        let Some(raw) = self.raw(name) else {
121            return default.to_vec();
122        };
123
124        raw.split(',')
125            .map(str::trim)
126            .filter(|item| !item.is_empty())
127            .map(str::to_owned)
128            .collect()
129    }
130
131    /// Registra uma queixa que o elo descobriu sozinho.
132    ///
133    /// É o que um elo usa quando o problema não é a leitura e sim o valor — um
134    /// segredo curto demais, um modo de TLS que não existe. A leitura continua,
135    /// e o boot é recusado no fim com esta linha junto das outras.
136    pub(crate) fn refuse(&mut self, message: impl Into<String>) {
137        self.errors.push(message.into());
138    }
139
140    /// O veredito, depois de a chain inteira ter rodado.
141    pub(crate) fn into_result(self) -> anyhow::Result<()> {
142        anyhow::ensure!(
143            self.errors.is_empty(),
144            "variáveis de ambiente inválidas:\n  - {}",
145            self.errors.join("\n  - ")
146        );
147
148        Ok(())
149    }
150
151    /// O valor cru de uma variável.
152    fn raw(&self, name: &str) -> Option<&str> {
153        self.vars.get(name).map(String::as_str)
154    }
155}
