1//! O erro como o cliente o vê.
2//!
3//! Esta é a **única** camada que conhece status HTTP. O `domain` devolve erro
4//! tipado, o `app` o une num [`AppError`], e a tradução para número acontece
5//! aqui — em [`ApiError::of_app`], num lugar só. Espalhar o código pelo
6//! sistema foi o que o PHP fazia (o `LeafContext` carregava o status desde o
7//! domínio), e o preço era que a mesma violação de regra tinha um status gravado
8//! nela mesmo quando a saída não era HTTP.
9//!
10//! ## O corpo é sempre `application/problem+json`
11//!
12//! Mesmo quando a requisição pediu `FlatBuffers`. É o que o PHP fazia, e há uma
13//! razão para manter: um erro pode acontecer **antes** de a negociação ter sido
14//! resolvida — corpo malformado, `Accept` ilegível, pânico — e nesses casos não
15//! existe formato negociado para responder. Ter um formato de erro único e
16//! sempre disponível evita a pergunta "em que formato eu digo que não entendi o
17//! formato?".
18
19use axum::http::{header, StatusCode};
20use axum::response::{IntoResponse, Response};
21use portmaster_app::domain::FieldError;
22use portmaster_app::error::AppError;
23use serde::Serialize;
24
25/// O tipo de mídia do corpo de erro (RFC 7807).
26const PROBLEM_JSON: &str = "application/problem+json";
27
28/// Um erro pronto para virar resposta.
29///
30/// Carrega cookies porque uma recusa às vezes **precisa** mexer na sessão: um
31/// refresh token morto tem que sair do navegador junto com o 401, senão o
32/// cliente o reapresenta a cada tentativa e nunca chega ao login. É o único caso,
33/// e é da mesma natureza do status — parte da resposta, não do erro.
34#[derive(Debug)]
35pub struct ApiError {
36    /// O status HTTP da resposta.
37    status: StatusCode,
38    /// O que aconteceu, em texto, para o corpo do problema.
39    detail: String,
40    /// Os `Set-Cookie` a acrescentar na resposta, um cabeçalho por entrada.
41    cookies: Vec<String>,
42}
43
44/// O corpo, na forma da RFC 7807 — espelha `ProblemDetails` de `common.fbs`.
45#[derive(Debug, Serialize)]
46struct ProblemDetails<'a> {
47    /// URI do tipo de problema; `about:blank` quando não há uma página.
48    r#type: &'a str,
49    /// O nome canônico do status.
50    title: &'a str,
51    /// O status, repetido no corpo para quem só lê o payload.
52    status: u16,
53    /// O que aconteceu, em texto.
54    detail: &'a str,
55}
56
57impl ApiError {
58    /// Um erro com status e explicação.
59    pub(crate) fn new(status: StatusCode, detail: impl Into<String>) -> Self {
60        Self {
61            status,
62            detail: detail.into(),
63            cookies: Vec::new(),
64        }
65    }
66
67    /// Acrescenta um `Set-Cookie` à recusa.
68    pub(crate) fn with_cookie(mut self, cookie: String) -> Self {
69        self.cookies.push(cookie);
70        self
71    }
72
73    /// Rota protegida sem sessão.
74    ///
75    /// O **401 é o único status que nasce nesta camada**: é a ausência de
76    /// sessão, e só quem lê o token sabe disso. Permissão (403), validação (422)
77    /// e conflito (409) vêm todos do `app`.
78    pub(crate) fn unauthenticated() -> Self {
79        Self::new(
80            StatusCode::UNAUTHORIZED,
81            "Authentication is required to access this resource.",
82        )
83    }
84
85    /// Corpo que não deu para ler no formato anunciado.
86    ///
87    /// **400, e não o 404 que o PHP passou a devolver.** O 404 é afirmado pela
88    /// suíte Go em cinco pontos para recurso genuinamente ausente; colidir os
89    /// dois significaria que nem um painel nem uma política de retry conseguem
90    /// separar "essa rota não existe" de "seus bytes são lixo".
91    ///
92    /// Cobre também o corpo **vazio**: ausência é um caso de ilegibilidade, não
93    /// um caso à parte.
94    pub(crate) fn unreadable_body(detail: impl Into<String>) -> Self {
95        Self::new(StatusCode::BAD_REQUEST, detail)
96    }
97
98    /// Resposta que existe mas não coube no fio.
99    ///
100    /// 502 e não 500: distingue "a resposta foi montada e falhou ao serializar"
101    /// do 500 que o middleware `Recover` produz quando algo entrou em pânico. Os
102    /// dois são defeito nosso, mas levam a investigações diferentes — e a suíte
103    /// Go não afirma 502 em ponto nenhum, então o código está livre.
104    pub(crate) fn unrenderable(detail: impl Into<String>) -> Self {
105        Self::new(StatusCode::BAD_GATEWAY, detail)
106    }
107
108    /// O status, para quem precisa decidir sobre ele antes de responder.
109    ///
110    /// Só os testes perguntam: no caminho de produção o erro vira resposta
111    /// direto, sem ninguém inspecioná-lo pelo meio.
112    #[cfg(test)]
113    pub(crate) const fn status(&self) -> StatusCode {
114        self.status
115    }
116
117    /// Traduz o erro do `app` para o status que o cliente recebe.
118    ///
119    /// É o **único** ponto de tradução do sistema. Um `match` exaustivo garante
120    /// que uma variante nova de [`AppError`] não passe despercebida como 500.
121    /// ## Validação vem em lote
122    ///
123    /// O domínio acumula **todos** os campos em vez de parar no primeiro.
124    /// Juntá-los num `detail` só é o que o `ProblemDetails` do schema comporta:
125    /// ele não tem `details[]`, e o `.fbs` é contrato com o cliente, não algo a
126    /// alterar por conveniência.
127    ///
128    /// ## O slug da permissão negada não vai no corpo
129    ///
130    /// Dizer ao cliente qual permissão faltou descreve para ele o mapa de
131    /// autorização do sistema. Quem precisa do detalhe é o operador, e para esse
132    /// ele já está no log.
133    ///
134    /// ## Regra de negócio violada é conflito, não erro de formato
135    ///
136    /// O pedido estava bem escrito; é o contêiner que não podia ser selado
137    /// agora. Credencial recusada responde igual para e-mail desconhecido e
138    /// senha errada — o `app` já garante isso.
139    ///
140    /// ## Falha de infra não descreve a topologia
141    ///
142    /// O que a infra reportou fica no log com a cadeia inteira; o cliente recebe
143    /// só que houve falha. O motivo real ("a conexão com o banco foi recusada")
144    /// não é acionável para ele.
145    pub fn of_app(error: AppError) -> Self {
146        match error {
147            AppError::Validation(fields) => {
148                Self::new(StatusCode::UNPROCESSABLE_ENTITY, describe_fields(&fields))
149            }
150
151            AppError::Unauthenticated => Self::new(
152                StatusCode::UNAUTHORIZED,
153                AppError::Unauthenticated.to_string(),
154            ),
155
156            AppError::Forbidden { permission } => {
157                tracing::info!(permission, "acesso negado por falta de permissão");
158                Self::new(
159                    StatusCode::FORBIDDEN,
160                    "You do not have permission to perform this action.",
161                )
162            }
163
164            AppError::NotFound { resource, id } => Self::new(
165                StatusCode::NOT_FOUND,
166                format!("{resource} não encontrado: {id}"),
167            ),
168
169            AppError::Conflict(message) => Self::new(StatusCode::CONFLICT, message),
170
171            AppError::Container(e) => Self::new(StatusCode::CONFLICT, e.to_string()),
172            AppError::Manifest(e) => Self::new(StatusCode::CONFLICT, e.to_string()),
173            AppError::Marker(e) => Self::new(StatusCode::CONFLICT, e.to_string()),
174            AppError::Metadata(e) => Self::new(StatusCode::CONFLICT, e.to_string()),
175
176            AppError::Auth(e) => Self::new(StatusCode::UNAUTHORIZED, e.to_string()),
177
178            AppError::Infra(e) => {
179                tracing::error!(error = ?e, "falha de infraestrutura");
180                Self::new(
181                    StatusCode::INTERNAL_SERVER_ERROR,
182                    "An unexpected error occurred.",
183                )
184            }
185        }
186    }
187}

189impl IntoResponse for ApiError {
190    /// Escreve o corpo `application/problem+json` e os cookies.
191    ///
192    /// A serialização de uma struct de quatro campos escalares não falha; se um
193    /// dia falhasse, um corpo vazio com o status certo ainda é uma resposta
194    /// melhor do que um pânico dentro do handler de erro.
195    fn into_response(self) -> Response {
196        let problem = ProblemDetails {
197            r#type: "about:blank",
198            title: title_of(self.status),
199            status: self.status.as_u16(),
200            detail: &self.detail,
201        };
202
203        let body = serde_json::to_vec(&problem).unwrap_or_default();
204
205        let mut response =
206            (self.status, [(header::CONTENT_TYPE, PROBLEM_JSON)], body).into_response();
207
208        for cookie in self.cookies {
209            if let Ok(value) = cookie.parse() {
210                response.headers_mut().append(header::SET_COOKIE, value);
211            }
212        }
213
214        response
215    }
216}
217
218/// Resume os campos recusados numa linha.
219fn describe_fields(fields: &[FieldError]) -> String {
220    if fields.is_empty() {
221        return "dados inválidos".to_owned();
222    }
223
224    fields
225        .iter()
226        .map(ToString::to_string)
227        .collect::<Vec<_>>()
228        .join("; ")
229}
230
231/// O nome canônico de um status.
232fn title_of(status: StatusCode) -> &'static str {
233    status.canonical_reason().unwrap_or("Error")
234}

236#[cfg(test)]
237mod tests {
238    use super::*;
239    use pretty_assertions::assert_eq;
240
241    /// O slug vai para o log.
242    ///
243    /// No corpo, ele descreveria ao cliente o mapa de autorização do sistema.
244    #[test]
245    fn a_permissao_negada_nao_vaza_o_slug_no_corpo() {
246        let error = ApiError::of_app(AppError::Forbidden {
247            permission: "container:seal".into(),
248        });
249
250        assert_eq!(error.status(), StatusCode::FORBIDDEN);
251        assert!(
252            !error.detail.contains("container:seal"),
253            "o corpo não deveria nomear a permissão: {}",
254            error.detail
255        );
256    }
257
258    #[test]
259    fn a_falha_de_infra_nao_vaza_o_motivo() {
260        let error = ApiError::of_app(AppError::Infra(anyhow::anyhow!(
261            "Connection refused (os error 111) para db:3306"
262        )));
263
264        assert_eq!(error.status(), StatusCode::INTERNAL_SERVER_ERROR);
265        assert!(
266            !error.detail.contains("db:3306"),
267            "topologia vazou: {}",
268            error.detail
269        );
270    }
271
272    /// O domínio acumula; perder campos aqui obrigaria o cliente a descobrir
273    /// um problema por requisição.
274    #[test]
275    fn a_validacao_lista_todos_os_campos() {
276        let error = ApiError::of_app(AppError::Validation(vec![
277            FieldError::new("email", "malformado"),
278            FieldError::new("password", "curta demais"),
279        ]));
280
281        assert_eq!(error.status(), StatusCode::UNPROCESSABLE_ENTITY);
282        assert!(error.detail.contains("email"));
283        assert!(error.detail.contains("password"));
284    }
285
286    #[test]
287    fn a_regra_de_negocio_violada_e_conflito() {
288        // O pedido estava bem escrito; o que não podia era o estado do contêiner.
289        use portmaster_app::domain::ContainerStatus;
290        let _ = ContainerStatus::Empty;
291
292        let error = ApiError::of_app(AppError::Conflict(
293            "Only a sealed container can be dispatched.".into(),
294        ));
295
296        assert_eq!(error.status(), StatusCode::CONFLICT);
297    }
298
299    #[test]
300    fn o_corpo_de_erro_e_problem_json() {
301        let response = ApiError::unauthenticated().into_response();
302
303        assert_eq!(response.status(), StatusCode::UNAUTHORIZED);
304        assert_eq!(
305            response
306                .headers()
307                .get(header::CONTENT_TYPE)
308                .and_then(|value| value.to_str().ok()),
309            Some(PROBLEM_JSON)
310        );
311    }
