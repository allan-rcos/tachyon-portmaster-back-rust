12/// O que um controller produziu, pronto para virar resposta.
13///
14/// Genérica sobre o VO. Não há `Box<dyn>` aqui, e não há como haver: o corpo é
15/// um tipo concreto que a rota conhece em tempo de compilação, e a codificação é
16/// monomorfizada junto com ele.
17///
18/// ## Por que ela carrega um `Result`
19///
20/// Porque o erro precisa sair pelo mesmo formato que o acerto. Se o sucesso
21/// passasse por aqui e a falha por outro caminho, existiriam dois lugares
22/// escrevendo corpo — e um deles acabaria escrevendo num formato que o cliente
23/// não pediu. Envolvendo o `Result`, negociar é uma coisa só que acontece uma
24/// vez, no [`IntoResponse`] abaixo.
25///
26/// ## O parâmetro tem padrão
27///
28/// Porque uma resposta sem corpo não tem VO nenhum, e `ApiResponse::no_content()`
29/// precisa resolver sozinho. `ProblemX` é o que ela usaria de qualquer forma se
30/// o `Result` desse errado — o corpo de erro é o mesmo em toda resposta.
31///
32/// ## E por que o corpo é `Option`
33///
34/// Porque `204` é uma resposta como as outras. Era um tipo à parte — o
35/// `NoContent` —, o que dava duas formas de responder. Um controller que às
36/// vezes tem corpo e às vezes não tinha de escolher entre elas no meio do
37/// método.
38pub(crate) struct ApiResponse<X: ResponseX = ProblemX> {
39    /// O status do acerto — o da falha vem do próprio erro.
40    status: StatusCode,
41    /// O que responder, ou por que não dá; `Ok(None)` é o `204`.
42    body: Result<Option<X>, ApiError>,
43}
44
45impl<X: ResponseX> ApiResponse<X> {
46    /// Um `200` com o que o controller devolveu.
47    pub(crate) fn ok(body: Result<X, ApiError>) -> Self {
48        Self::with_status(StatusCode::OK, body)
49    }
50
51    /// Um `201` para o recurso recém-criado.
52    pub(crate) fn created(body: Result<X, ApiError>) -> Self {
53        Self::with_status(StatusCode::CREATED, body)
54    }
55
56    /// A resposta com um status escolhido.
57    pub(crate) fn with_status(status: StatusCode, body: Result<X, ApiError>) -> Self {
58        Self {
59            status,
60            body: body.map(Some),
61        }
62    }
63}
64
65impl ApiResponse<ProblemX> {
66    /// Um `204` para a operação cujo resultado é o próprio estado ter mudado.
67    ///
68    /// É o que o PHP devolvia em refresh, logout e nas operações sem retorno.
69    /// Recebe um `Result` como as outras duas, e pela mesma razão: a falha
70    /// precisa sair pelo mesmo caminho do acerto, com o status que o erro
71    /// carrega.
72    ///
73    /// Fixada em `ProblemX` porque uma resposta sem corpo não tem VO nenhum, e o
74    /// parâmetro precisa de algum: o `ProblemX` é o que ela usaria de qualquer
75    /// forma se o `Result` desse errado. Fixá-la aqui é também o que a faz
76    /// resolver sem turbofish.
77    pub(crate) fn no_content(body: Result<(), ApiError>) -> Self {
78        Self {
79            status: StatusCode::NO_CONTENT,
80            body: body.map(|()| None),
81        }
82    }
83}
84
85impl<X: ResponseX> IntoResponse for ApiResponse<X> {
86    /// Codifica o corpo — ou o problema — no formato negociado.
87    ///
88    /// Cookie não aparece aqui. Quem os carimba é o layer de cookie, depois do
89    /// handler e para toda resposta, o que é o que faz uma recusa poder mexer na
90    /// sessão sem que este tipo saiba disso.
91    fn into_response(self) -> Response {
92        match self.body {
93            Ok(Some(body)) => EncodeContext.respond(self.status, &body),
94            Ok(None) => self.status.into_response(),
95            Err(error) => {
96                let (status, problem) = error.into_parts();
97
98                EncodeContext.respond(status, &problem)
99            }
100        }
101    }
102}
