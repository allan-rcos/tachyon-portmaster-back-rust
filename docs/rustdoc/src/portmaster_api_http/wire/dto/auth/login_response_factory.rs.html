1//! O que `POST /auth/login` e `POST /setup` respondem.
2
3use crate::error::api_error::ApiError;
4use crate::wire::dto::auth::user_response_factory::UserResponseFactory;
5use crate::wire::factory::response_factory::ResponseFactory;
6use crate::wire::tables as fbs;
7
8/// Monta a resposta de uma sessão aberta.
9///
10/// O aninhamento é o que o planus já sabe fazer: `table()` devolve a tabela
11/// *owned* com o `User` dentro, e o impl gerado de `WriteAsOffset` escreve o
12/// filho antes do pai — que é, linha por linha, o `createUser` + `createLoginResponse`
13/// que o PHP escrevia à mão.
14pub(crate) struct LoginResponseFactory {
15    /// O access token emitido.
16    token: String,
17    /// Como o token viaja — `cookie`, e não `Bearer`.
18    token_type: &'static str,
19    /// O dono da sessão, na forma enxuta que o login publica.
20    user: UserResponseFactory,
21}
22
23impl LoginResponseFactory {
24    /// Monta a factory com o token emitido e o dono dele.
25    pub(crate) const fn new(
26        token: String,
27        token_type: &'static str,
28        user: UserResponseFactory,
29    ) -> Self {
30        Self {
31            token,
32            token_type,
33            user,
34        }
35    }
36}
37
38impl ResponseFactory for LoginResponseFactory {
39    type Table = fbs::auth::LoginResponse;
40
41    /// Monta a tabela, com a do `User` aninhada dentro.
42    ///
43    /// A factory filha produz a **tabela**, e o planus a escreve no lugar certo
44    /// — é ele quem faz a coreografia de builder que o PHP escrevia à mão.
45    fn table(&self) -> Result<Self::Table, ApiError> {
46        Ok(fbs::auth::LoginResponse {
47            token: Some(self.token.clone()),
48            token_type: Some(self.token_type.to_owned()),
49            user: Some(Box::new(self.user.table()?)),
50        })
51    }
52}

108    /// A suíte Go manda `Accept: application/x-flatbuffers` em toda requisição
109    /// — ela não cobre JSON em ponto nenhum.
110    ///
111    /// Este `assert_eq!` da string exata é o **único** contrato verificável do
112    /// caminho JSON, e a ordem dos campos faz parte dele: é a ordem de
113    /// declaração do `.fbs`.
114    #[test]
115    fn o_json_bate_com_o_que_o_swagger_documenta() {
116        let json = JsonEncodeStrategy
117            .encode(&factory())
118            .expect("a resposta precisa serializar");
119
120        assert_eq!(
121            String::from_utf8(json).expect("o JSON é UTF-8"),
122            r#"{"token":"jwt.de.mentira","token_type":"cookie","user":{"id":"aZ3","name":"Ana","email":"ana@portmaster.local"}}"#
123        );
124    }

126    /// A tabela do wire não tem onde pôr `password_hash`, e é isso que garante

140    /// É o ponto do desenho: uma factory, duas strategies, nenhuma das duas
141    /// sabendo o que a outra faz.
142    #[test]
143    fn a_mesma_factory_sai_nos_dois_formatos() {

151    /// O `Renderable` devolve `Offset<()>`, e o `finish` escreve por ele.
152    ///
153    /// `Offset<T>::ALIGNMENT` é 4 para todo `T`, então apagar o tipo não muda
154    /// um byte — este teste é o que sustenta essa afirmação.
155    #[test]
156    fn o_aninhamento_produz_os_mesmos_bytes_pelo_caminho_apagado() {
157        let erased = FlatBuffersEncodeStrategy
158            .encode(&factory())
159            .expect("a resposta precisa serializar");
160
161        let typed = {
162            let mut builder = planus::Builder::new();
163            let table = factory().table().expect("a tabela precisa montar");
164            builder.finish(&table, None).to_vec()
165        };
166
167        assert_eq!(erased, typed);
168    }
