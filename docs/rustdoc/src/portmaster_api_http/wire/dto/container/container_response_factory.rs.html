1//! Um contêiner.
2
3use crate::error::api_error::ApiError;
4use crate::wire::convert::Convert;
5use crate::wire::factory::response_factory::ResponseFactory;
6use crate::wire::tables as fbs;
7use portmaster_app::views::ContainerViewItem;
8
9/// Monta a tabela de um contêiner.
10pub(crate) struct ContainerResponseFactory {
11    /// A View de origem, que a `table()` traduz.
12    source: ContainerViewItem,
13}
14
15impl ContainerResponseFactory {
16    /// Monta a factory sobre a View, que é o que a leitura devolve.
17    pub(crate) const fn of(source: ContainerViewItem) -> Self {
18        Self { source }
19    }
20
21    /// Monta a factory sobre o objeto de domínio, que é o que a escrita devolve.
22    pub(crate) fn of_domain(container: &dyn portmaster_app::domain::Container) -> Self {
23        Self {
24            source: ContainerViewItem {
25                id: container.id().to_owned(),
26                code: container.code().to_owned(),
27                current_weight: container.current_weight(),
28                max_capacity: container.max_capacity(),
29                status: container.status().as_i32(),
30            },
31        }
32    }
33}
34
35impl ResponseFactory for ContainerResponseFactory {
36    type Table = fbs::container::ContainerResponse;
37
38    fn table(&self) -> Result<Self::Table, ApiError> {
39        Ok(fbs::container::ContainerResponse {
40            id: Some(self.source.id.clone()),
41            code: Some(self.source.code.clone()),
42            current_weight: self.source.current_weight,
43            max_capacity: self.source.max_capacity,
44            status: Convert::container_status(self.source.status),
45        })
46    }
47}
