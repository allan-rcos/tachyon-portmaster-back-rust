17/// Converte a lista de papéis que perfil e usuário-admin publicam igual.
18///
19/// As duas mensagens carregam o mesmo recorte de papel, e a conversão é a mesma
20/// — duplicá-la nos dois arquivos criaria a chance de uma divergir da outra.
21pub(crate) fn roles_of(
22    roles: &[RoleViewItem],
23) -> Result<Vec<fbs::account::RoleResponse>, ApiError> {
24    roles
25        .iter()
26        .map(|role| RoleResponseFactory::of(role.clone()).table())
27        .collect()
