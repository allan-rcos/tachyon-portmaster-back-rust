1//! O corpo da requisição, já na mensagem que a factory descreve.
2
3use axum::body::Bytes;
4use axum::extract::{FromRequest, Request};
5use axum::http::StatusCode;
6
7use crate::error::api_error::ApiError;
8use crate::wire::factory::request_factory::RequestFactory;
9use crate::wire::media_type::MediaType;
10use crate::wire::strategy::decode_strategy::DecodeStrategy;
11use crate::wire::strategy::flatbuffers_decode_strategy::FlatBuffersDecodeStrategy;
12use crate::wire::strategy::json_decode_strategy::JsonDecodeStrategy;
13use crate::wire::wire::Wire;
14
15/// Teto do corpo de uma requisição.
16///
17/// Nenhum endpoint desta API recebe payload grande — o maior é uma lista de ids
18/// de papel. O limite existe para que um corpo enorme seja recusado antes de ser
19/// lido inteiro na memória, e não depois.
20const MAX_BODY_BYTES: usize = 1024 * 1024;
21
22/// O corpo da requisição, lido pela factory `F`.
23///
24/// O parâmetro é a **factory**, não a mensagem: é ela que sabe ler os dois
25/// formatos, e pedi-la por tipo é o que mantém a rota dizendo qual mensagem
26/// espera — `Body<LoginRequestFactory>` — sem instanciar nada.
27pub(crate) struct Body<F: RequestFactory>(pub(crate) F::Message);
28
29impl<S, F> FromRequest<S> for Body<F>
30where
31    F: RequestFactory,
32    S: Send + Sync,
33{
34    type Rejection = ApiError;
35
36    /// Lê o corpo no formato que a requisição anunciou.
37    ///
38    /// O `match` de formato fica aqui, e não num `dyn`: `DecodeStrategy::decode`
39    /// é genérico sobre a factory e por isso a trait não é object-safe — o que é
40    /// conveniente, porque mantém o caminho de requisição estático.
41    async fn from_request(request: Request, state: &S) -> Result<Self, Self::Rejection> {
42        let (mut parts, body) = request.into_parts();
43        let wire = Wire::from_request_parts(&mut parts, state).await?;
44        let request = Request::from_parts(parts, body);
45
46        let bytes = Bytes::from_request(request, state)
47            .await
48            .map_err(|e| ApiError::unreadable_body(format!("não foi possível ler o corpo: {e}")))?;
49
50        if bytes.len() > MAX_BODY_BYTES {
51            return Err(ApiError::new(
52                StatusCode::PAYLOAD_TOO_LARGE,
53                "corpo grande demais",
54            ));
55        }
56
57        let message = match wire.request() {
58            MediaType::Json => JsonDecodeStrategy.decode::<F>(&bytes),
59            MediaType::FlatBuffers => FlatBuffersDecodeStrategy.decode::<F>(&bytes),
60        }?;
61
62        Ok(Self(message))
63    }
64}
