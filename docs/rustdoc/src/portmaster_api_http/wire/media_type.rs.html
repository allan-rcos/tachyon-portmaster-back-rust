1//! Qual dos dois formatos está em jogo.
2
3/// `application/json`.
4const JSON: &str = "application/json";
5
6/// O tipo nativo desta API.
7const FLATBUFFERS: &str = "application/x-flatbuffers";
8
9/// O que um cliente genérico manda quando só sabe que o corpo não é texto.
10const OCTET_STREAM: &str = "application/octet-stream";
11
12/// O curinga que um navegador manda em toda navegação.
13const ANY: &str = "*/*";
14
15/// Qual dos dois formatos está em jogo.
16///
17/// Os dois construtores **falham** quando o cabeçalho pede um tipo que não
18/// sabemos produzir nem ler. Antes eles caíam num padrão em silêncio, e o
19/// resultado era um cliente que pediu XML recebendo JSON com `200` — o pior dos
20/// dois mundos, porque nem recebe o que pediu nem descobre que não vai receber.
21/// Quem traduz a falha em status é o middleware: `406` na saída, `415` na
22/// entrada.
23#[derive(Debug, Clone, Copy, PartialEq, Eq)]
24pub(crate) enum MediaType {
25    /// Texto, para depuração e clientes genéricos.
26    Json,
27    /// Binário, o formato nativo.
28    FlatBuffers,
29}
30
31impl MediaType {
32    /// O formato do **corpo que chegou**, pelo `Content-Type`.
33    ///
34    /// Cabeçalho ausente cai em `FlatBuffers`: quem manda corpo sem anunciar o
35    /// tipo é um cliente nosso falando o formato nativo. Cabeçalho presente e
36    /// desconhecido é erro — ele anunciou alguma coisa, e não foi nenhuma das
37    /// duas que lemos.
38    pub(crate) fn of_request(content_type: Option<&str>) -> anyhow::Result<Self> {
39        let Some(content_type) = content_type.map(str::to_ascii_lowercase) else {
40            return Ok(Self::FlatBuffers);
41        };
42
43        Self::of_header(&content_type).ok_or_else(|| {
44            anyhow::anyhow!(
45                "não sabemos ler um corpo em `{content_type}`: use {JSON} ou {FLATBUFFERS}"
46            )
47        })
48    }
49
50    /// O formato da **resposta**, pelo `Accept`.
51    ///
52    /// Cabeçalho ausente cai em JSON, e um `*/*` em qualquer posição também —
53    /// entre os dois, o legível serve melhor a quem não pediu nada em
54    /// particular, e é o que um navegador manda. Só um tipo concreto que não
55    /// sabemos escrever é erro; tratar o curinga como erro daria `406` para todo
56    /// navegador que abrisse a API.
57    pub(crate) fn of_response(accept: Option<&str>) -> anyhow::Result<Self> {
58        let Some(accept) = accept.map(str::to_ascii_lowercase) else {
59            return Ok(Self::Json);
60        };
61
62        if accept.contains(ANY) {
63            return Ok(Self::Json);
64        }
65
66        Self::of_header(&accept).ok_or_else(|| {
67            anyhow::anyhow!(
68                "não sabemos escrever uma resposta em `{accept}`: use {JSON} ou {FLATBUFFERS}"
69            )
70        })
71    }
72
73    /// O valor de `Content-Type` a devolver.
74    pub(crate) const fn header_value(self) -> &'static str {
75        match self {
76            Self::Json => JSON,
77            Self::FlatBuffers => FLATBUFFERS,
78        }
79    }
80
81    /// O formato que um cabeçalho já em minúsculas nomeia.
82    ///
83    /// Busca por substring, e não por igualdade, porque os dois cabeçalhos vêm
84    /// com parâmetros e listas: `application/json; charset=utf-8` de um lado,
85    /// `text/html,application/json;q=0.9` do outro.
86    fn of_header(header: &str) -> Option<Self> {
87        if header.contains("json") {
88            return Some(Self::Json);
89        }
90
91        if header.contains("flatbuffers") || header.contains(OCTET_STREAM) {
92            return Some(Self::FlatBuffers);
93        }
94
95        None
96    }
97}

99#[cfg(test)]
100mod tests {
101    use super::*;
102    use pretty_assertions::assert_eq;
103
104    #[test]
105    fn sem_content_type_o_corpo_e_binario() {
106        assert_eq!(
107            MediaType::of_request(None).expect("ausente é o formato nativo"),
108            MediaType::FlatBuffers
109        );
110        assert_eq!(
111            MediaType::of_request(Some(FLATBUFFERS)).expect("é o formato nativo"),
112            MediaType::FlatBuffers
113        );
114        assert_eq!(
115            MediaType::of_request(Some("application/json; charset=utf-8")).expect("é json"),
116            MediaType::Json
117        );
118    }
119
120    /// Ele anunciou alguma coisa, e não foi nenhuma das duas que lemos.
121    #[test]
122    fn um_content_type_desconhecido_e_recusado() {
123        assert!(MediaType::of_request(Some("application/xml")).is_err());
124    }
125
126    #[test]
127    fn sem_accept_a_resposta_e_legivel() {
128        assert_eq!(
129            MediaType::of_response(None).expect("ausente é json"),
130            MediaType::Json
131        );
132        assert_eq!(
133            MediaType::of_response(Some("*/*")).expect("curinga é json"),
134            MediaType::Json
135        );
136        assert_eq!(
137            MediaType::of_response(Some(FLATBUFFERS)).expect("é o formato nativo"),
138            MediaType::FlatBuffers
139        );
140        assert_eq!(
141            MediaType::of_response(Some(OCTET_STREAM)).expect("binário genérico"),
142            MediaType::FlatBuffers
143        );
144    }
145
146    #[test]
147    fn o_accept_com_qualidade_e_lista_ainda_e_entendido() {
148        // Um navegador manda algo como `text/html,application/json;q=0.9,*/*`.
149        assert_eq!(
150            MediaType::of_response(Some("text/html,application/json;q=0.9,*/*;q=0.8"))
151                .expect("a lista nomeia json"),
152            MediaType::Json
153        );
154    }
155
156    /// O curinga tem que ganhar do tipo que não sabemos escrever, senão todo
157    /// navegador que abrisse a API levaria 406.
158    #[test]
159    fn o_curinga_salva_o_accept_que_so_nomeia_o_que_nao_escrevemos() {
160        assert_eq!(
161            MediaType::of_response(Some("text/html,application/xhtml+xml,*/*;q=0.8"))
162                .expect("o curinga está lá"),
163            MediaType::Json
164        );
165    }
166
167    #[test]
168    fn um_accept_concreto_e_desconhecido_e_recusado() {
169        assert!(MediaType::of_response(Some("application/xml")).is_err());
170        assert!(MediaType::of_response(Some("text/html")).is_err());
171    }
172}
