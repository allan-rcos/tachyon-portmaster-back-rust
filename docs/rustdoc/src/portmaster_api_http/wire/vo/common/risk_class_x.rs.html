6/// A classe de risco de um produto, na numeração da ONU.
7///
8/// O conjunto é fechado e publicado em `common.fbs`: cliente e servidor
9/// compartilham o mesmo vocabulário. Este enum é a forma dele que não
10/// depende de formato — os dois DTOs saem daqui, e nenhum deles é este.
11#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
12pub(crate) enum RiskClassX {
13    /// Explosivos.
14    Class1Explosives,
15    /// Gases.
16    Class2Gases,
17    /// Líquidos inflamáveis.
18    Class3FlammableLiquids,
19    /// Sólidos inflamáveis.
20    Class4FlammableSolids,
21    /// Substâncias oxidantes.
22    Class5OxidizingSubstances,
23    /// Substâncias tóxicas.
24    Class6ToxicSubstances,
25    /// Materiais radioativos.
26    Class7RadioactiveMaterials,
27    /// Substâncias corrosivas.
28    Class8CorrosiveSubstances,
29    /// Diversos.
30    Class9Miscellaneous,
31    /// Sem classificação.
32    #[default]
33    None,
34}
35
36impl RiskClassX {
37    /// O valor a partir do índice que a View carrega.
38    ///
39    /// Um índice fora da faixa cai no valor neutro em vez de derrubar a
40    /// resposta: a linha guardada no banco não é entrada do cliente, e um
41    /// registro estranho não deve custar a página inteira a quem a pediu.
42    pub(crate) const fn of_index(index: i32) -> Self {
43        match index {
44            0 => Self::Class1Explosives,
45            1 => Self::Class2Gases,
46            2 => Self::Class3FlammableLiquids,
47            3 => Self::Class4FlammableSolids,
48            4 => Self::Class5OxidizingSubstances,
49            5 => Self::Class6ToxicSubstances,
50            6 => Self::Class7RadioactiveMaterials,
51            7 => Self::Class8CorrosiveSubstances,
52            8 => Self::Class9Miscellaneous,
53            _ => Self::None,
54        }
55    }
56
57    /// O índice que o domínio usa.
58    pub(crate) const fn as_index(self) -> i32 {
59        match self {

61            Self::Class2Gases => 1,
62            Self::Class3FlammableLiquids => 2,
63            Self::Class4FlammableSolids => 3,
64            Self::Class5OxidizingSubstances => 4,

67            Self::Class8CorrosiveSubstances => 7,
68            Self::Class9Miscellaneous => 8,
69            Self::None => 9,
70        }
71    }
