1//! O VO de `UserCreateRequest`.
2
3use crate::ports::error::api_error::ApiError;
4use crate::wire::dto::json::admin::user_create_request_json::UserCreateRequestJson;
5use crate::wire::tables as fbs;
6use crate::wire::x::request_x::RequestX;
7use planus::ReadAsRoot as _;
8
9/// O que a rota de `UserCreateRequest` recebe.
10///
11/// Os campos são `Option` embora o `.fbs` marque alguns `required`: é o
12/// que faz um campo ausente virar 422 nomeando-o, e não um 400 genérico.
13#[derive(Debug, Clone, Default)]
14pub(crate) struct UserCreateXRequest {
15    /// O nome do usuário novo.
16    pub(crate) name: Option<String>,
17    /// O e-mail, que também é a credencial de login.
18    pub(crate) email: Option<String>,
19    /// A senha inicial. Morre no `TableModule`, que guarda só o hash.
20    pub(crate) initial_password: Option<String>,
21    /// Os papéis com que ele nasce.
22    pub(crate) role_ids: Option<Vec<String>>,
23}
24
25impl RequestX for UserCreateXRequest {
26    type Json = UserCreateRequestJson;
27
28    fn of_json(dto: Self::Json) -> Self {
29        Self {
30            name: dto.name,
31            email: dto.email,
32            initial_password: dto.initial_password,
33            role_ids: dto.role_ids,
34        }
35    }
36
37    /// Lê a mensagem do buffer, tolerando campo ausente.
38    ///
39    /// Campo declarado `required` no `.fbs` que não veio é buffer truncado —
40    /// ilegível, não incompleto no sentido de negócio. O `ok()` deixa o `None`
41    /// seguir para o `TableModule`, que dirá qual campo falta.
42    fn of_fbs(bytes: &[u8]) -> Result<Self, ApiError> {
43        let table = fbs::admin::UserCreateRequestRef::read_as_root(bytes)
44            .map_err(|e| ApiError::unreadable_body(format!("corpo FlatBuffers inválido: {e}")))?;
45
46        Ok(Self {
47            name: table.name().ok().map(str::to_owned),
48            email: table.email().ok().map(str::to_owned),
49            initial_password: table.initial_password().ok().map(str::to_owned),
50            role_ids: table.role_ids().ok().flatten().map(|items| {
51                items
52                    .into_iter()
53                    .filter_map(|item| item.map(str::to_owned).ok())
54                    .collect()
55            }),
56        })
57    }
58}
