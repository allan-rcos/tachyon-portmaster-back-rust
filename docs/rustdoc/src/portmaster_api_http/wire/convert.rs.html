1//! As conversões que não são movimento direto entre a origem e o fio.
2//!
3//! Três lugares onde os tipos não batem, e os três moram aqui em vez de
4//! espalhados pelas factories:
5//!
6//! * **Contagens**: as Views usam `i64` (é o que `COUNT(*)` devolve); as tabelas
7//!   usam `int32`, porque foi o que o `.fbs` fixou. A conversão **satura** em vez
8//!   de truncar — um total absurdo aparece como absurdo, não como negativo.
9//! * **Enums**: a View carrega o índice; o wire tem o seu próprio enum. Um
10//!   índice fora da faixa cai no valor neutro em vez de derrubar a resposta.
11//! * **Timestamp**: a View guarda epoch em ms; o schema pede `string`. Sai em
12//!   RFC 3339, que é o que um cliente lê sem conhecer a convenção de quem
13//!   escreveu — e em UTC, como todo horário do sistema.
14
15use chrono::{DateTime, Utc};
16
17use crate::wire::tables as fbs;
18
19/// Converte um valor de origem no que a tabela do wire declara.
20pub(crate) struct Convert;
21
22impl Convert {
23    /// Uma contagem, saturada na faixa que o wire comporta.
24    pub(crate) fn count(value: i64) -> i32 {
25        i32::try_from(value).unwrap_or(i32::MAX)
26    }
27
28    /// O índice de classe de risco, no enum do wire.
29    pub(crate) fn risk_class(index: i32) -> fbs::common::RiskClass {
30        u8::try_from(index)
31            .ok()
32            .and_then(|index| fbs::common::RiskClass::try_from(index).ok())
33            .unwrap_or(fbs::common::RiskClass::None)
34    }
35
36    /// O índice de status, no enum do wire.
37    pub(crate) fn container_status(index: i32) -> fbs::common::ContainerStatus {
38        u8::try_from(index)
39            .ok()
40            .and_then(|index| fbs::common::ContainerStatus::try_from(index).ok())
41            .unwrap_or(fbs::common::ContainerStatus::Empty)
42    }
43
44    /// O índice de evento, no enum do wire.
45    pub(crate) fn telemetry_event(index: i32) -> fbs::common::TelemetryEvent {
46        u8::try_from(index)
47            .ok()
48            .and_then(|index| fbs::common::TelemetryEvent::try_from(index).ok())
49            .unwrap_or(fbs::common::TelemetryEvent::Load)
50    }
51
52    /// Epoch em ms → RFC 3339, em UTC.
53    pub(crate) fn timestamp(epoch_ms: i64) -> Option<String> {
54        DateTime::<Utc>::from_timestamp_millis(epoch_ms).map(|at| at.to_rfc3339())
55    }
56}

58#[cfg(test)]
59mod tests {
60    use super::*;
61    use pretty_assertions::assert_eq;
62
63    /// Um `COUNT(*)` acima de 2^31 é absurdo, mas truncá-lo produziria um
64    /// negativo — que o cliente exibiria como se fosse um dado.
65    #[test]
66    fn a_contagem_satura_em_vez_de_truncar() {
67        assert_eq!(Convert::count(i64::from(i32::MAX) + 1), i32::MAX);
68        assert_eq!(Convert::count(7), 7);
69    }
70
71    #[test]
72    fn um_indice_de_enum_fora_da_faixa_cai_no_neutro() {
73        assert_eq!(Convert::risk_class(99), fbs::common::RiskClass::None);
74        assert_eq!(
75            Convert::container_status(-1),
76            fbs::common::ContainerStatus::Empty
77        );
78        assert_eq!(
79            Convert::risk_class(2),
80            fbs::common::RiskClass::Class3FlammableLiquids
81        );
82    }
83
84    #[test]
85    fn o_timestamp_sai_em_utc_e_interpretavel() {
86        assert_eq!(
87            Convert::timestamp(0).as_deref(),
88            Some("1970-01-01T00:00:00+00:00")
89        );
90    }
