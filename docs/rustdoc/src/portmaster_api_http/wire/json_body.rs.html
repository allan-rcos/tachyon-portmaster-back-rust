1//! O corpo que não passa pela negociação.
2
3use axum::body::Bytes;
4use axum::extract::{FromRequest, Request};
5use axum::http::StatusCode;
6
7use crate::error::api_error::ApiError;
8
9/// Teto do corpo de uma requisição.
10///
11/// O mesmo que o [`Body`](crate::wire::body::Body) negociado aplica. Está
12/// repetido em vez de compartilhado porque os dois extractors são caminhos
13/// independentes, e um `const` público só para uni-los criaria um export a mais
14/// num módulo que já exporta o seu tipo.
15const MAX_BODY_BYTES: usize = 1024 * 1024;
16
17/// Um corpo lido **sempre** como JSON, qualquer que seja o `Content-Type`.
18///
19/// Existe para um endpoint só: `PUT /users/{id}/roles`. O payload dele nunca
20/// entrou nos `.fbs` — o PHP o lia com `json_decode` direto do corpo, sem tabela
21/// e sem negociação — e os `.fbs` são contrato publicado com os clientes, não
22/// algo a estender por conveniência nossa. Enquanto a lista de papéis não tiver
23/// tabela própria, ler JSON aqui é o que mantém o endpoint respondendo ao mesmo
24/// corpo de antes.
25///
26/// O `Content-Type` é ignorado de propósito: o cliente de referência anuncia
27/// `FlatBuffers` em toda requisição e mesmo assim manda JSON **neste** corpo, que
28/// é exatamente a situação que o PHP tolerava.
29#[derive(Debug)]
30pub struct JsonBody<T>(pub(crate) T);
31
32impl<S, T> FromRequest<S> for JsonBody<T>
33where
34    T: serde::de::DeserializeOwned,
35    S: Send + Sync,
36{
37    type Rejection = ApiError;
38
39    async fn from_request(request: Request, state: &S) -> Result<Self, Self::Rejection> {
40        let bytes = Bytes::from_request(request, state)
41            .await
42            .map_err(|e| ApiError::unreadable_body(format!("não foi possível ler o corpo: {e}")))?;
43
44        if bytes.len() > MAX_BODY_BYTES {
45            return Err(ApiError::new(
46                StatusCode::PAYLOAD_TOO_LARGE,
47                "corpo grande demais",
48            ));
49        }
50
51        serde_json::from_slice(&bytes)
52            .map(Self)
53            .map_err(|e| ApiError::unreadable_body(format!("corpo JSON inválido: {e}")))
54    }
55}
