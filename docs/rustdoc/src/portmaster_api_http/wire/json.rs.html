1//! A leitura de campos de um objeto JSON.
2//!
3//! É o `JsonHelper` do PHP: um namespace de leitores que uma factory usa para
4//! tirar cada campo do `Map` sem repetir a mesma dança de `get` + `as_str` +
5//! `to_owned` em cada uma das 15 mensagens de requisição.
6//!
7//! ## Ausente e inválido são a mesma coisa aqui
8//!
9//! Todo leitor devolve `Option`, e um campo com o tipo errado vira `None` em vez
10//! de erro. Não é desleixo: quem recusa é o `TableModule`, que sabe **quais**
11//! campos são obrigatórios e devolve todos os problemas de uma vez. Um erro
12//! levantado aqui interromperia na primeira falha e devolveria um por vez.
13
14use serde_json::{Map, Value};
15
16/// Lê campos de um objeto JSON.
17pub(crate) struct Json;
18
19impl Json {
20    /// Um texto.
21    pub(crate) fn text(source: &Map<String, Value>, field: &str) -> Option<String> {
22        source.get(field).and_then(Value::as_str).map(str::to_owned)
23    }
24
25    /// Um inteiro de 64 bits.
26    pub(crate) fn number(source: &Map<String, Value>, field: &str) -> Option<i64> {
27        source.get(field).and_then(Value::as_i64)
28    }
29
30    /// Um inteiro de 32 bits, saturado na faixa.
31    pub(crate) fn int(source: &Map<String, Value>, field: &str) -> Option<i32> {
32        Self::number(source, field).map(|value| {
33            i32::try_from(value).unwrap_or_else(|_| {
34                if value.is_negative() {
35                    i32::MIN
36                } else {
37                    i32::MAX
38                }
39            })
40        })
41    }
42
43    /// Um número com casas decimais.
44    pub(crate) fn real(source: &Map<String, Value>, field: &str) -> Option<f64> {
45        source.get(field).and_then(Value::as_f64)
46    }
47
48    /// Uma lista de textos.
49    ///
50    /// Entradas que não são texto são descartadas — a lista existe para ser
51    /// consumida inteira, e um elemento inválido no meio não deve derrubar os
52    /// demais.
53    pub(crate) fn texts(source: &Map<String, Value>, field: &str) -> Option<Vec<String>> {
54        source.get(field).and_then(Value::as_array).map(|items| {
55            items
56                .iter()
57                .filter_map(Value::as_str)
58                .map(str::to_owned)
59                .collect()
60        })
61    }
62}

64#[cfg(test)]
65mod tests {
66    use super::*;
67    use pretty_assertions::assert_eq;
68
69    fn source() -> Map<String, Value> {
70        match serde_json::json!({
71            "name": "Cimento",
72            "density": 1.44,
73            "count": 7,
74            "roles": ["a", 2, "b"],
75        }) {
76            Value::Object(map) => map,
77            _ => unreachable!("o literal acima é um objeto"),
78        }
79    }
80
81    #[test]
82    fn le_cada_tipo_no_seu_leitor() {
83        let s = source();
84
85        assert_eq!(Json::text(&s, "name"), Some("Cimento".to_owned()));
86        assert_eq!(Json::real(&s, "density"), Some(1.44));
87        assert_eq!(Json::number(&s, "count"), Some(7));
88    }
89
90    #[test]
91    fn campo_ausente_ou_do_tipo_errado_e_none() {
92        // Quem decide se isso é problema é o `TableModule`.
93        let s = source();
94
95        assert_eq!(Json::text(&s, "inexistente"), None);
96        assert_eq!(Json::text(&s, "density"), None);
97        assert_eq!(Json::number(&s, "name"), None);
98    }
99
100    #[test]
101    fn a_lista_descarta_o_que_nao_e_texto() {
102        assert_eq!(
103            Json::texts(&source(), "roles"),
104            Some(vec!["a".to_owned(), "b".to_owned()])
105        );
106    }
107}
