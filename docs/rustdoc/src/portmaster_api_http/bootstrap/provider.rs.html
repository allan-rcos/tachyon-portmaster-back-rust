17/// Os factories dos controllers e do que a pilha de middlewares consome.
18///
19/// Fecha o mesmo padrão das camadas de baixo aqui em cima: cada método devolve
20/// `impl Trait` — contrato, nunca tipo concreto — e o grafo inteiro é
21/// monomorfizado.
22///
23/// ## Os controllers são construídos uma vez
24///
25/// Ao contrário dos casos de uso, que são baratos de reconstruir a cada chamada,
26/// os controllers nascem no boot e são **clonados** por requisição. É o que o
27/// [`register`](crate::bootstrap::register::register()) faz: ele consome o `AppProvider`, tira
28/// dele tudo que as rotas precisam, e o descarta. Depois do boot não existe
29/// provider em memória, nem `Arc` para mantê-lo vivo — só os controllers que as
30/// rotas seguram, cada um um punhado de handles.
31pub(crate) trait ApiProvider {
32    /// A conta do próprio usuário.
33    fn account_controller(&self) -> impl AccountController + use<Self> + 'static;
34
35    /// Sessão: login, setup, refresh e logout.
36    fn auth_controller(&self) -> impl AuthController + use<Self> + 'static;
37
38    /// Contêineres.
39    fn container_controller(&self) -> impl ContainerController + use<Self> + 'static;
40
41    /// Carga e telemetria.
42    fn manifest_controller(&self) -> impl ManifestController + use<Self> + 'static;
43
44    /// Metadados de sistema.
45    fn metadata_controller(&self) -> impl MetadataController + use<Self> + 'static;
46
47    /// O painel do pátio.
48    fn metrics_controller(&self) -> impl MetricsController + use<Self> + 'static;
49
50    /// Produtos.
51    fn product_controller(&self) -> impl ProductController + use<Self> + 'static;
52
53    /// Papéis.
54    fn role_controller(&self) -> impl RoleController + use<Self> + 'static;
55
56    /// O estado do próprio processo.
57    fn server_controller(&self) -> impl ServerController + use<Self> + 'static;
58
59    /// Usuários.
60    fn user_controller(&self) -> impl UserController + use<Self> + 'static;
61
62    /// Quem emite e confere o access token, para o middleware de sessão.
63    fn token_service(&self) -> impl TokenService + use<Self> + 'static;
64
65    /// Fábrica de loggers, para os middlewares.
66    fn logger_factory(&self) -> impl LoggerFactory + use<Self> + 'static;
67
68    /// Um logger nomeado, para quem não é middleware.
69    fn logger(&self, name: &str) -> impl Logger + use<Self> + 'static;
70
71    /// Gerador de id ordenável, para o `request_id`.
72    fn sequential_id_generator(&self) -> impl SequentialIdGenerator + use<Self> + 'static;
73
74    /// Por quanto tempo uma requisição pode demorar.
75    fn request_timeout(&self) -> Duration;
76
77    /// As origens que o CORS libera.
78    ///
79    /// Vazio quando a API e o front saem do mesmo host, que é o padrão seguro.
80    fn cors_origins(&self) -> &[String];
81}
