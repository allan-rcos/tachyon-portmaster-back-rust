1//! A junção de todas as versões publicadas num router só.
2
3use std::collections::HashSet;
4
5use axum::http::Method;
6use axum::Router;
7
8use crate::bootstrap::provider::ApiProvider;
9use crate::router::intern::v1_router::V1Router;
10use crate::router::versioned_router::VersionedRouter;
11
12/// Junta todas as versões publicadas num `Router`.
13///
14/// Cada versão é montada sob o próprio grupo `/v<n>`, tirado do número que ela
15/// mesma declara — o prefixo não tem segunda fonte. Além disso o espaço sem
16/// versão também é servido: `/products` alcança a **versão mais nova que a
17/// publica**, decidido rota a rota e não globalmente. Uma rota carregada da v1
18/// para a v2 responde pela v2 na raiz; uma que só existiu na v1 continua
19/// respondendo pela v1.
20///
21/// A raiz é conveniência, não contrato. O que ela aponta muda no dia em que uma
22/// versão nova publicar a mesma rota, então um cliente que quer dizer v1 deve
23/// pedir `/v1` — que é o que o `swagger.json` diz a ele no bloco `servers`.
24///
25/// Struct-namespace: as duas funções são a mesma montagem vista de dois lados.
26pub(crate) struct RouterHub;
27
28impl RouterHub {
29    /// Monta o router sobre todas as versões que este binário publica.
30    ///
31    /// **Da mais nova para a mais velha.** É a ordem de que o alias sem versão
32    /// depende: a primeira ocorrência de cada par `(verbo, caminho)` é a que
33    /// fica, e percorrer nesta ordem é o que a faz ser a mais nova.
34    ///
35    /// Uma versão nova é uma linha aqui e um arquivo em `intern`.
36    pub(crate) fn build<P: ApiProvider>(provider: &P) -> anyhow::Result<Router> {
37        let mut mounted = Mounted::default();
38
39        Self::mount::<V1Router, P>(&mut mounted, provider)?;
40
41        Ok(mounted.router)
42    }
43
44    /// Monta uma versão sob o prefixo dela, e o que sobrar dela na raiz.
45    fn mount<V: VersionedRouter, P: ApiProvider>(
46        mounted: &mut Mounted,
47        provider: &P,
48    ) -> anyhow::Result<()> {
49        anyhow::ensure!(
50            mounted.versions.insert(V::VERSION),
51            "duas tabelas declaram a versão {}: um número endereça exatamente uma delas",
52            V::VERSION
53        );
54
55        let prefix = format!("/v{}", V::VERSION);
56        let mut versioned = Router::new();
57
58        for route in V::routes(provider) {
59            versioned = versioned.route(route.path, route.handler);
60        }
61
62        let mut router = std::mem::take(&mut mounted.router).nest(&prefix, versioned);
63
64        for route in V::routes(provider) {
65            if mounted.published.insert((route.method, route.path)) {
66                router = router.route(route.path, route.handler);
67            }
68        }
69
70        mounted.router = router;
71
72        Ok(())
73    }
74}
75
76/// O que a montagem acumula enquanto percorre as versões.
77///
78/// O `published` é o que faz o merge, e é por isso que ele é um conjunto de
79/// `(verbo, caminho)` e não uma lista: essa é a identidade que o axum recusa
80/// ver duas vezes, então descartar a repetição **antes** de registrar não é
81/// economia — é a única forma de registrar, já que o axum entra em pânico ao
82/// receber o par repetido em vez de deixar escolher qual fica.
83#[derive(Default)]
84struct Mounted {
85    /// O que já foi montado.
86    router: Router,
87    /// Os números de versão já vistos, para recusar um repetido.
88    versions: HashSet<u16>,
89    /// Os pares já publicados na raiz.
90    published: HashSet<(Method, &'static str)>,
91}
