1//! A primeira versão publicada do contrato REST.

3use crate::bootstrap::api_provider::ApiProvider;

9use crate::router::versioned_router::VersionedRouter;
10
11/// A v1, montada sob `/v1` e — enquanto for a mais nova — também na raiz.
12///
13/// **Esta tabela é congelada.** Uma mudança no que qualquer uma destas rotas
14/// significa é uma `V2Router` ao lado deste arquivo, não uma edição nele — é o
15/// ponto inteiro de a versão ser um tipo.
16///
17/// Ela não lista caminho nenhum. Cada recurso traz as próprias rotas, e o que
18/// este arquivo faz é dizer **quais recursos** a v1 publica: o encanamento de
19/// extractor mora ao lado do controller que o consome, e aqui não aparece nome
20/// de tipo interno nenhum.
21pub(crate) struct V1Router;
22
23impl VersionedRouter for V1Router {
24    const VERSION: u16 = 1;
25
26    fn routes() -> anyhow::Result<Vec<Route>> {
27        Ok([
28            server_routes::routes(ApiProvider::server_controller()),
29            auth_routes::routes(ApiProvider::auth_controller()?),
30            account_routes::routes(ApiProvider::account_controller()?),
31            product_routes::routes(ApiProvider::product_controller()?),
32            role_routes::routes(ApiProvider::role_controller()?),
33            container_routes::routes(ApiProvider::container_controller()?),
34            manifest_routes::routes(ApiProvider::manifest_controller()?),
35            user_routes::routes(ApiProvider::user_controller()?),
36            metadata_routes::routes(ApiProvider::metadata_controller()),
37            metrics_routes::routes(ApiProvider::metrics_controller()?),
38        ]
39        .into_iter()
40        .flatten()
41        .collect())
42    }
43}
