28/// Monta o router completo.
29///
30/// Cada recurso traz as **próprias** rotas, e este arquivo só as junta. É o que
31/// faz ele caber numa tela: o encanamento de extractor — qual rota tem corpo,
32/// qual tem `Path`, qual mensagem cada corpo carrega — mora ao lado do
33/// controller que o consome, e aqui não aparece nome nenhum de tipo interno.
34///
35/// Antes havia duas `macro_rules!` neste arquivo. Elas existiam para esconder
36/// que cada rota reconstruía o seu controller a cada requisição, a partir de um
37/// `Arc` do provider clonado duas vezes por chamada. Agora os controllers vêm
38/// prontos do `ApiProvider`, são construídos **uma vez** aqui, e cada rota
39/// clona o seu — um punhado de handles.
40pub async fn router<P: AppProvider>(
41    app: P,
42    config: ApiConfig,
43    jwt: &JwtConfig,
44) -> anyhow::Result<Router> {
45    let provider = crate::bootstrap::register::register(app, config, jwt);
46
47    // O grupo de marcador da sessão precisa existir antes da primeira
48    // requisição, e quem sabe o nome dele é o controller que o usa.
49    provider
50        .auth_controller()
51        .register_marker_group()
52        .await
53        .map_err(|error| anyhow::anyhow!("{error:?}"))?;
54
55    Ok(routes(&provider))
56}

58/// Junta as rotas de cada recurso e aplica a pilha.
59///
60/// A pilha é aplicada de dentro para fora: o último `.layer` é o mais externo.
61/// Quatro ordens aqui não são gosto, e sim requisito — cada layer que abre um
62/// escopo tem de estar **fora** de quem lê aquele escopo:
63///
64/// * `RequestId` é o mais externo, porque o `Logging` lê o id dele.
65/// * `Logging` vem logo em seguida, para que o span envolva o resto da pilha e o
66///   handler inteiro.
67/// * `Encode` e `Decode` ficam fora do `Recover` e do `Timeout`, que produzem
68///   resposta por conta própria — um `500` e um `504` — e precisam do formato já
69///   negociado para escrevê-la.
70/// * `Cookie` fica fora do `Session`, que lê o access token por ele.
71fn routes<P: ApiProvider>(provider: &P) -> Router {
72    Router::new()
73        .merge(server_routes::routes(provider.server_controller()))
74        .merge(auth_routes::routes(provider.auth_controller()))
75        .merge(account_routes::routes(provider.account_controller()))
76        .merge(product_routes::routes(provider.product_controller()))
77        .merge(role_routes::routes(provider.role_controller()))
78        .merge(container_routes::routes(provider.container_controller()))
79        .merge(manifest_routes::routes(provider.manifest_controller()))
80        .merge(user_routes::routes(provider.user_controller()))
81        .merge(metadata_routes::routes(provider.metadata_controller()))
82        .merge(metrics_routes::routes(provider.metrics_controller()))
83        // De dentro para fora: o último `.layer` é o mais externo.
84        .layer(SessionLayer::new(provider.token_service()))
85        .layer(CookieLayer::new())
86        .layer(cors_layer(provider.cors_origins()))
87        .layer(TimeoutLayer::new(provider.request_timeout()))
88        .layer(RecoverLayer::new())
89        .layer(DecodeLayer::new())
90        .layer(EncodeLayer::new())
91        .layer(LoggingLayer::new(&provider.logger_factory()))
92        .layer(RequestIdLayer::new(provider.sequential_id_generator()))
93}

95/// A política de CORS.
96///
97/// Sem origens configuradas, a camada não libera nada — é o padrão seguro, e o
98/// que vale quando a API e o front saem do mesmo host. Com origens, libera
99/// credenciais: a sessão viaja em cookie, e sem `allow_credentials` o navegador
100/// não o mandaria.
101fn cors_layer(origins: &[String]) -> CorsLayer {
102    if origins.is_empty() {
103        return CorsLayer::new();
104    }
105
106    let allowed: Vec<HeaderValue> = origins
107        .iter()
108        .filter_map(|origin| HeaderValue::from_str(origin).ok())
109        .collect();
110
111    CorsLayer::new()
112        .allow_origin(AllowOrigin::list(allowed))
113        .allow_credentials(true)
114        .allow_methods([
115            Method::GET,
116            Method::POST,
117            Method::PUT,
118            Method::DELETE,
119            Method::OPTIONS,
120        ])
121        .allow_headers(Any)
122        .max_age(std::time::Duration::from_secs(CORS_MAX_AGE_SECONDS))
123}
