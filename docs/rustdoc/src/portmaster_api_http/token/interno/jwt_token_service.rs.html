1//! O token de sessão como JWT. Não sai do módulo.
2//!
3//! ## O principal viaja como `FlatBuffers` dentro da claim
4//!
5//! O JWT não carrega claims soltas por usuário. O principal inteiro — id, nome,
6//! e-mail, papéis e permissões — é serializado na tabela `TokenUser` de
7//! `token.fbs`, codificado em base64url e posto numa claim só. Duas razões: o
8//! token fica menor que o JSON equivalente, e o payload deixa de ser legível
9//! para quem espia o cookie por curiosidade.
10//!
11//! Não é sigilo — base64 não esconde nada de quem queira ler. O que a assinatura
12//! garante é **integridade**: o conteúdo não pode ser alterado sem o segredo.
13

29/// As claims do access token.
30#[derive(Debug, Serialize, Deserialize)]
31struct Claims {
32    /// Id do usuário, em base62.
33    sub: String,
34    /// Quem emitiu.
35    iss: String,
36    /// Quando expira, em epoch de segundos.
37    exp: u64,
38    /// Quando foi emitido.
39    iat: u64,
40    /// O principal inteiro, em `FlatBuffers` e base64url.
41    user_data: String,
42}
43
44/// Emite e confere tokens assinados em HS256.
45#[derive(Clone)]
46pub(crate) struct JwtTokenService<K> {
47    /// A chave de assinatura.
48    encoding: EncodingKey,
49    /// A chave de verificação.
50    decoding: DecodingKey,
51    /// Quem emitiu, gravado e conferido na claim `iss`.
52    issuer: String,
53    /// Validade do access token.
54    ttl: Duration,
55    /// De onde sai o `iat`/`exp`.
56    clock: K,
57}
58
59impl<K: Clock> JwtTokenService<K> {
60    /// Monta o serviço a partir da configuração.
61    pub(crate) fn new(config: &JwtConfig, clock: K) -> Self {
62        let secret = config.secret.expose_secret().as_bytes();
63
64        Self {
65            encoding: EncodingKey::from_secret(secret),
66            decoding: DecodingKey::from_secret(secret),
67            issuer: config.issuer.clone(),
68            ttl: config.ttl,
69            clock,
70        }
71    }
72
73    /// O agora, em epoch de segundos.
74    ///
75    /// Vem do relógio injetado. O `exp` de um JWT precisa do tempo de parede
76    /// real — o mesmo que o cliente consulta —, e o `Clock` da `infra` entrega
77    /// exatamente isso, em UTC, sem que este arquivo chame `SystemTime::now`.
78    fn epoch_seconds(&self) -> u64 {
79        u64::try_from(self.clock.now().timestamp()).unwrap_or_default()
80    }
81
82    /// Serializa o principal na tabela `TokenUser`, em base64url.
83    ///
84    /// Método privado e não linha solta no [`Self::issue`]: são vinte linhas de
85    /// coreografia de builder cujo ponto — o que entra no token e o que fica de
86    /// fora — merece o nome e a explicação acima.
87    fn pack_principal(user: &dyn User) -> String {
88        let token_user = fbs::token::TokenUser {
89            id: Some(user.id().to_owned()),
90            name: Some(user.name().to_owned()),
91            email: Some(user.email().to_owned()),
92            roles: Some(
93                user.roles()
94                    .iter()
95                    .map(|role| fbs::token::TokenRole {
96                        id: Some(role.id().to_owned()),
97                        name: Some(role.name().to_owned()),
98                        permissions: Some(role.permissions().to_vec()),
99                    })
100                    .collect(),
101            ),
102        };
103
104        let mut builder = planus::Builder::new();
105
106        URL_SAFE_NO_PAD.encode(builder.finish(&token_user, None))
107    }
108
109    /// Desfaz o [`Self::pack_principal`].
110    ///
111    /// Tolera todo campo ausente menos os ids: um papel sem id não identifica
112    /// nada, e um principal sem id não é ninguém. Nome e permissões vazios
113    /// degradam a sessão sem invalidá-la.
114    fn unpack_principal(packed: &str) -> Option<UserContext> {
115        use planus::ReadAsRoot as _;
116
117        let bytes = URL_SAFE_NO_PAD.decode(packed).ok()?;
118        let token_user = fbs::token::TokenUserRef::read_as_root(&bytes).ok()?;
119
120        let roles = token_user
121            .roles()
122            .ok()
123            .flatten()
124            .map(|roles| {
125                roles
126                    .iter()
127                    .filter_map(|role| {
128                        let role = role.ok()?;
129
130                        Some(RoleContext {
131                            id: role.id().ok().flatten()?.to_owned(),
132                            name: role.name().ok().flatten().unwrap_or_default().to_owned(),
133                            permissions: role
134                                .permissions()
135                                .ok()
136                                .flatten()
137                                .map(|slugs| {
138                                    slugs
139                                        .iter()
140                                        .filter_map(|slug| Some(slug.ok()?.to_owned()))
141                                        .collect()
142                                })
143                                .unwrap_or_default(),
144                        })
145                    })
146                    .collect()
147            })
148            .unwrap_or_default();
149
150        Some(UserContext {
151            id: token_user.id().ok().flatten()?.to_owned(),
152            name: token_user
153                .name()
154                .ok()
155                .flatten()
156                .unwrap_or_default()
157                .to_owned(),
158            email: token_user
159                .email()
160                .ok()
161                .flatten()
162                .unwrap_or_default()
163                .to_owned(),
164            roles,
165        })
166    }
167}

169impl<K: Clock> TokenService for JwtTokenService<K> {
170    fn issue(&self, user: &dyn User) -> Result<String, ApiError> {
171        let now = self.epoch_seconds();
172
173        #[allow(
174            clippy::arithmetic_side_effects,
175            reason = "epoch em segundos mais um TTL de minutos: estourar u64 são ~500 bilhões de anos"
176        )]
177        let claims = Claims {
178            sub: user.id().to_owned(),
179            iss: self.issuer.clone(),
180            exp: now + self.ttl.as_secs(),
181            iat: now,
182            user_data: Self::pack_principal(user),
183        };
184
185        jsonwebtoken::encode(&Header::new(Algorithm::HS256), &claims, &self.encoding).map_err(|e| {
186            ApiError::new(axum::http::StatusCode::INTERNAL_SERVER_ERROR, e.to_string())
187        })
188    }
189
190    /// Confere a assinatura e devolve o principal que o token carrega.
191    ///
192    /// ## Sem tolerância de relógio
193    ///
194    /// O padrão da lib são 60 segundos, pensados para quando quem assina e quem
195    /// confere são máquinas diferentes — aqui são o mesmo processo, com o mesmo
196    /// relógio. Mantê-la só faria todo token valer um minuto a mais do que o
197    /// `exp` que ele mesmo declara.
198    fn verify(&self, token: &str) -> Option<UserContext> {
199        let mut validation = Validation::new(Algorithm::HS256);
200        validation.set_issuer(&[&self.issuer]);
201
202        validation.leeway = 0;
203
204        let claims = jsonwebtoken::decode::<Claims>(token, &self.decoding, &validation)
205            .ok()?
206            .claims;
207
208        Self::unpack_principal(&claims.user_data)
209    }
210}

335    /// É o que sustenta a autenticação stateless: tudo que a autorização
336    /// precisa saber viaja no token, e nenhum middleware toca o banco.

338    fn o_token_devolve_o_principal_inteiro() {
339        let service = service();
340        let token = service.issue(&user()).expect("o token deveria ser emitido");
341
342        let context = service
343            .verify(&token)
344            .expect("token recém-emitido deveria valer");
345
346        assert_eq!(context.id, "u1");
347        assert_eq!(context.email, "ana@portmaster.local");
348        assert_eq!(context.roles.len(), 1);
