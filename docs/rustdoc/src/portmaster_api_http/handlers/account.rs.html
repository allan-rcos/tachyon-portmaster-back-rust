1//! `/account` — o que um usuário faz com a própria conta.
2//!
3//! Nenhuma destas rotas pede permissão: o alvo é sempre quem está agindo, e o id
4//! vem do contexto da sessão — nunca do corpo. É o `app` que garante isso; aqui
5//! só se repassa o contexto.
6//!
7//! ## Por que a escrita relê antes de responder
8//!
9//! `PUT /account` responde com o **perfil**, que carrega quantos usuários cada
10//! papel tem. Esse número é consulta, não regra: o objeto de domínio que a
11//! escrita devolve não o conhece, e publicá-lo zerado seria informar um valor
12//! errado em vez de nenhum. Reler pelo lado de leitura é o que mantém a resposta
13//! igual à do `GET` que vem logo depois.
14
15use portmaster_app::commands::account::ChangePasswordCommand;
16use portmaster_app::commands::account::UpdateAccountCommand;
17use portmaster_app::context::UserContext;
18use portmaster_app::queries::account::GetAccountQuery;
19use portmaster_app::services::AccountUseCase;
20
21use crate::error::api_error::ApiError;
22use crate::session::Session;
23use crate::wire::api_response::ApiResponse;
24use crate::wire::body::Body;
25use crate::wire::dto::account::account_profile_response_factory::AccountProfileResponseFactory;
26use crate::wire::dto::account::account_update_request_factory::AccountUpdateRequestFactory;
27use crate::wire::dto::account::password_change_request_factory::PasswordChangeRequestFactory;
28use crate::wire::no_content::NoContent;
29use crate::wire::wire::Wire;
30
31/// Os handlers da conta própria.
32pub struct AccountHandlers<A> {
33    /// O caso de uso da conta própria.
34    account: A,
35}
36
37impl<A: AccountUseCase> AccountHandlers<A> {
38    /// Monta os handlers.
39    pub(crate) const fn new(account: A) -> Self {
40        Self { account }
41    }
42
43    /// `GET /account`
44    pub(crate) async fn get(&self, wire: Wire) -> Result<ApiResponse, ApiError> {
45        let context = Session::require_user()?;
46
47        Ok(ApiResponse::ok(wire, self.profile(context).await?))
48    }
49
50    /// `PUT /account`
51    pub(crate) async fn update(
52        &self,
53        wire: Wire,
54        Body(request): Body<AccountUpdateRequestFactory>,
55    ) -> Result<ApiResponse, ApiError> {
56        let context = Session::require_user()?;
57
58        self.account
59            .update(UpdateAccountCommand {
60                context: context.clone(),
61                name: request.name.unwrap_or_default(),
62                email: request.email.unwrap_or_default(),
63            })
64            .await
65            .map_err(ApiError::of_app)?;
66
67        Ok(ApiResponse::ok(wire, self.profile(context).await?))
68    }
69
70    /// `PUT /account/password`
71    /// Responde `204` sem corpo de propósito: o cookie de sessão continua
72    /// valendo, e a resposta não tem nada a dizer que o cliente já não saiba.
73    pub(crate) async fn change_password(
74        &self,
75        Body(request): Body<PasswordChangeRequestFactory>,
76    ) -> Result<NoContent, ApiError> {
77        let context = Session::require_user()?;
78
79        self.account
80            .change_password(ChangePasswordCommand {
81                context,
82                current_password: request.current_password.unwrap_or_default(),
83                new_password: request.new_password.unwrap_or_default(),
84            })
85            .await
86            .map_err(ApiError::of_app)?;
87
88        Ok(NoContent::new())
89    }
90
91    /// O perfil de quem está na sessão, pelo lado de leitura.
92    async fn profile(
93        &self,
94        context: UserContext,
95    ) -> Result<AccountProfileResponseFactory, ApiError> {
96        let view = self
97            .account
98            .get(GetAccountQuery { context })
99            .await
100            .map_err(ApiError::of_app)?;
101
102        Ok(AccountProfileResponseFactory::of(view))
103    }
104}
