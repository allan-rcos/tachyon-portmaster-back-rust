20/// Os layers da pilha, montados com o que cada um consome.
21///
22/// ## Por que estes não devolvem `impl Trait`
23///
24/// É a única exceção do grafo, e ela é do axum. O bound de `Router::layer` não
25/// pede só `L: Layer<Route>`: ele pede coisas sobre `L::Service` — que seja
26/// `Service<Request>`, `Clone`, `Send`, `'static`, e que a resposta dele vire
27/// `IntoResponse`. Um `impl Layer<Route>` esconde `L::Service` atrás de um tipo
28/// opaco, e não há como declarar bounds sobre o associado de um tipo opaco.
29/// Devolver o tipo concreto é o que mantém a pilha composta por generics, sem
30/// um `dyn` no meio dela.
31///
32/// Os tipos não saem do crate, então o nome de um layer alcança só o arquivo que
33/// monta o router — que é justamente quem precisa conhecê-lo.
34pub(crate) struct MiddlewareProvider;
35
36impl MiddlewareProvider {
37    /// Abre a sessão a partir do cookie de access.
38    pub(crate) fn session() -> anyhow::Result<SessionLayer<impl TokenService + use<>>> {
39        Ok(SessionLayer::new(TokenProvider::token_service()?))
40    }
41
42    /// Recolhe os cookies da requisição e publica os da resposta.
43    pub(crate) const fn cookies() -> CookieLayer {
44        CookieLayer::new()
45    }
46
47    /// Derruba a requisição que passar do teto de tempo.
48    pub(crate) const fn timeout(limit: Duration) -> TimeoutLayer {
49        TimeoutLayer::new(limit)
50    }
51
52    /// Transforma um pânico em resposta, em vez de queda.
53    pub(crate) const fn recover() -> RecoverLayer {
54        RecoverLayer::new()
55    }
56
57    /// Escolhe como o corpo da requisição é lido.
58    pub(crate) const fn decode() -> DecodeLayer {
59        DecodeLayer::new()
60    }
61
62    /// Escolhe em que formato a resposta sai.
63    pub(crate) const fn encode() -> EncodeLayer {
64        EncodeLayer::new()
65    }
66
67    /// Registra a requisição e o que ela respondeu.
68    pub(crate) fn logging() -> LoggingLayer<impl portmaster_app::Logger> {
69        LoggingLayer::new(&AppProvider::logger_factory())
70    }
71
72    /// Abre o escopo da pilha de eventos desta requisição.
73    ///
74    /// Precisa ser mais externo que todo layer que pergunte à pilha — hoje, o
75    /// [`Self::cache_status`].
76    pub(crate) fn meta_events(
77    ) -> MetaEventLayer<impl portmaster_app::MetaEventStackSubscriber + Clone + use<>> {
78        MetaEventLayer::new(AppProvider::meta_event_stack())
79    }
80
81    /// Publica se a resposta saiu do cache de leitura.
82    pub(crate) fn cache_status(
83    ) -> CacheStatusLayer<impl portmaster_app::MetaEventStackSubscriber + Clone + use<>> {
84        CacheStatusLayer::new(AppProvider::meta_event_stack())
85    }
86
87    /// Abre o escopo com o `request_id` desta requisição.
88    pub(crate) fn request_id() -> RequestIdLayer<impl portmaster_app::SequentialIdGenerator + use<>>
89    {
90        RequestIdLayer::new(AppProvider::sequential_id_generator())
91    }
92}
