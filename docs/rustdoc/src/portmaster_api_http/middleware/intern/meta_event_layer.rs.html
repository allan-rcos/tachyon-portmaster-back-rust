1//! O middleware que abre o escopo da pilha de eventos.
2
3use std::task::{Context, Poll};
4
5use axum::extract::Request;
6use axum::response::Response;
7use futures::future::BoxFuture;
8use portmaster_app::MetaEventStackSubscriber;
9use tower::{Layer, Service};
10
11/// Abre o escopo em que a pilha de eventos da requisição existe, e nada além.
12///
13/// ## Não faz nada com o que foi registrado
14///
15/// Ele **abre**; quem lê é outro layer, e quem escreve é um caso de uso. Somar
16/// as três coisas num middleware só faria este arquivo conhecer o
17/// [`MetaEvent`](portmaster_app::MetaEvent) que ele não tem por que conhecer, e
18/// um evento novo passaria a tocá-lo.
19///
20/// ## Quem lê tem de estar dentro
21///
22/// A leitura acontece depois que o handler responde, e um `task_local` só existe
23/// dentro do futuro que o escopo embrulha. Então todo layer que pergunte à pilha
24/// precisa ser **mais interno** que este — em `router/mod.rs`, onde o último
25/// `.layer` é o mais externo, isso quer dizer aplicado antes dele.
26///
27/// Errar essa ordem não quebra nada de forma visível: a pergunta responde
28/// `false` e a resposta sai sem a marca. É o teste de integração que pega, ao
29/// afirmar o acerto e não só a presença do cabeçalho.
30#[derive(Clone)]
31pub(crate) struct MetaEventLayer<E, S = ()> {
32    /// O serviço interno, que este envolve; `()` enquanto é só layer.
33    inner: S,
34    /// A pilha cujo escopo é aberto.
35    events: E,
36}
37
38impl<E> MetaEventLayer<E> {
39    /// Monta o layer com a pilha que o provider entregou.
40    pub(crate) const fn new(events: E) -> Self {
41        Self { inner: (), events }
42    }
43}
44
45impl<S, E: Clone> Layer<S> for MetaEventLayer<E> {
46    type Service = MetaEventLayer<E, S>;
47
48    fn layer(&self, inner: S) -> Self::Service {
49        MetaEventLayer {
50            inner,
51            events: self.events.clone(),
52        }
53    }
54}
55
56impl<S, E> Service<Request> for MetaEventLayer<E, S>
57where
58    S: Service<Request, Response = Response> + Clone + Send + 'static,
59    S::Future: Send + 'static,
60    E: MetaEventStackSubscriber + Clone + Send + Sync + 'static,
61{
62    type Response = Response;
63    type Error = S::Error;
64    type Future = BoxFuture<'static, Result<Response, Self::Error>>;
65
66    fn poll_ready(&mut self, cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
67        self.inner.poll_ready(cx)
68    }
69
70    fn call(&mut self, request: Request) -> Self::Future {
71        let clone = self.inner.clone();
72        let mut inner = std::mem::replace(&mut self.inner, clone);
73        let events = self.events.clone();
74
75        Box::pin(async move { events.scope(inner.call(request)).await })
76    }
77}
