1//! O formato de saída da tarefa corrente, e o Strategy que ele escolhe.
2//!
3//! > **Lacuna conhecida do `lint-exports`:** o `CURRENT` abaixo nasce de
4//! > `tokio::task_local!`, e itens gerados por macro são invisíveis ao `syn`.
5//! > O arquivo exporta dois itens na prática, não um.
6
7use std::future::Future;
8
9use axum::http::{header, StatusCode};
10use axum::response::{IntoResponse as _, Response};
11
12use crate::middleware::encode_port::EncodePort;
13use crate::wire::media_type::MediaType;
14use crate::wire::strategy::encode_strategy::EncodeStrategy as _;
15use crate::wire::strategy::flatbuffers_encode_strategy::FlatBuffersEncodeStrategy;
16use crate::wire::strategy::json_encode_strategy::JsonEncodeStrategy;
17use crate::wire::x::response_x::ResponseX;
18
19tokio::task_local! {
20    /// O formato que esta requisição negociou para a resposta.
21    static CURRENT: MediaType;
22}
23
24/// O contexto do Strategy pattern na saída.
25///
26/// O estado do contexto é a variante guardada no escopo, e as strategies são
27/// ZSTs imóveis: trocar de formato é trocar a variante, não realocar nada. Não
28/// há enum de strategy à parte porque não precisa haver — a
29/// [`MediaType`] **é** o seletor, e um segundo enum espelhando-a só criaria a
30/// chance de os dois discordarem.
31///
32/// ZST: o formato é da tarefa, não do objeto. Injetá-lo custa o mesmo que não
33/// injetar.
34#[derive(Debug, Clone, Copy, Default)]
35pub(crate) struct EncodeContext;
36
37impl EncodeContext {
38    /// Roda `future` com este formato instalado na tarefa.
39    ///
40    /// `pub(super)`: é o escritor, e só o layer irmão o alcança. De fora não há
41    /// como um handler responder num formato que a requisição não pediu.
42    pub(super) async fn scope<F: Future>(media: MediaType, future: F) -> F::Output {
43        CURRENT.scope(media, future).await
44    }
45
46    /// O formato desta tarefa, ou JSON.
47    ///
48    /// Cair em JSON fora do escopo é deliberado, e é o oposto do que a sessão
49    /// faz. Um erro que nasce **antes** do middleware — o pânico que o `Recover`
50    /// pega numa camada mais externa, um teste que exercita um controller solto
51    /// — ainda precisa virar resposta, e recusar aqui seria responder que não
52    /// sabemos responder, o que exigiria escolher um formato para dizê-lo.
53    fn current() -> MediaType {
54        CURRENT.try_with(|media| *media).unwrap_or(MediaType::Json)
55    }
56}
57
58impl EncodePort for EncodeContext {
59    fn respond<X: ResponseX>(&self, status: StatusCode, body: &X) -> Response {
60        let media = Self::current();
61
62        let encoded = match media {
63            MediaType::Json => JsonEncodeStrategy.encode(body),
64            MediaType::FlatBuffers => FlatBuffersEncodeStrategy.encode(body),
65        };
66
67        match encoded {
68            Ok(bytes) => (
69                status,
70                [(header::CONTENT_TYPE, media.header_value())],
71                bytes,
72            )
73                .into_response(),
74            Err(error) => error.into_response(),
75        }
76    }
77}
