17/// Transforma um pânico do handler em resposta.
18///
19/// ## O layer é o serviço antes de saber o que envolve
20///
21/// Um tipo só, e não um par. `RecoverLayer` sem parâmetro é o `Layer`, e
22/// `RecoverLayer<S>` é o `Service` que sai do `layer()` — o mesmo tipo, agora
23/// com o `S` que ele embrulha. Um par de tipos custaria um segundo arquivo e um
24/// nome que ninguém pronuncia: quem monta a pilha aplica o layer e mais nada.
25#[derive(Clone, Copy, Default)]
26pub(crate) struct RecoverLayer<S = ()> {
27    /// O serviço interno, que este envolve; `()` enquanto é só layer.
28    inner: S,
29}
30
31impl RecoverLayer {
32    /// Monta o layer.
33    pub(crate) const fn new() -> Self {
34        Self { inner: () }
35    }
36}
37
38impl<S> Layer<S> for RecoverLayer {
39    type Service = RecoverLayer<S>;
40
41    fn layer(&self, inner: S) -> Self::Service {
42        RecoverLayer { inner }
43    }
44}
45
46impl<S> Service<Request> for RecoverLayer<S>
47where
48    S: Service<Request, Response = Response> + Clone + Send + 'static,
49    S::Future: Send + 'static,
50{
51    type Response = Response;
52    type Error = S::Error;
53    type Future = BoxFuture<'static, Result<Response, Self::Error>>;
54
55    fn poll_ready(&mut self, cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
56        self.inner.poll_ready(cx)
57    }
58
59    /// Chama o serviço interno e transforma um pânico em resposta.
60    ///
61    /// O corpo de `500` sai pela
62    /// [`EncodePort`](crate::middleware::encode_port::EncodePort), no formato
63    /// que a requisição negociou. Este layer não relê o `Accept`: quem o
64    /// resolveu foi o `EncodeLayer`, que por isso tem de estar **fora** deste na
65    /// pilha. Cada middleware que responde por conta própria remontar a
66    /// negociação seria a mesma decisão tomada em três lugares.
67    ///
68    /// Um literal de bytes JSON aqui seria pior do que parece: num sistema cujo
69    /// cliente de produção fala `FlatBuffers`, é o único corpo que ele nunca
70    /// conseguiria ler.
71    fn call(&mut self, request: Request) -> Self::Future {
72        let clone = self.inner.clone();
73        let mut inner = std::mem::replace(&mut self.inner, clone);
74
75        Box::pin(async move {
76            let outcome = AssertUnwindSafe(inner.call(request)).catch_unwind().await;
77
78            Ok(match outcome {
79                Ok(response) => response?,
80                Err(panic) => {
81                    SystemLogger::get().error(
82                        "pânico capturado no controller",
83                        [("panic", &describe(&panic))],
84                    );
85
86                    let (status, problem) = ApiError::new(
87                        StatusCode::INTERNAL_SERVER_ERROR,
88                        "An unexpected error occurred.",
89                    )
90                    .into_parts();
91
92                    EncodeContext.respond(status, &problem)
93                }
94            })
95        })
96    }
97}

99/// O que dá para dizer sobre o payload de um pânico.
100fn describe(panic: &Box<dyn std::any::Any + Send>) -> String {
101    if let Some(message) = panic.downcast_ref::<&str>() {
102        return (*message).to_owned();
103    }
104
105    if let Some(message) = panic.downcast_ref::<String>() {
106        return message.clone();
107    }
108
109    "pânico sem mensagem legível".to_owned()
110}
