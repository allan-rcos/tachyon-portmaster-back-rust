1//! Os cookies da tarefa corrente: os que chegaram e os que vão sair.
2//!
3//! É o **único** arquivo do sistema que conhece o tipo `Cookie`. Havia uma trait
4//! `AuthCookie` tentando abstraí-lo, e ela não abstraía nada: quatro dos seis
5//! métodos devolviam um `Cookie` na assinatura, então o tipo interno atravessava
6//! o contrato e chegava aos controllers de qualquer forma.
7//!
8//! > **Lacuna conhecida do `lint-exports`:** o `CURRENT` abaixo nasce de
9//! > `tokio::task_local!`, e itens gerados por macro são invisíveis ao `syn`.
10//! > O arquivo exporta dois itens na prática, não um.
11
12use std::future::Future;
13use std::sync::Arc;
14
15use axum::http::{header, HeaderMap};
16use cookie::Cookie;
17use portmaster_app::{Logger as _, SystemLogger};
18
19use crate::middleware::cookie_port::CookiePort;
20use crate::ports::cookie::cookie_name::CookieName;
21use crate::ports::error::api_error::ApiError;
22use crate::ports::session_policy::SessionPolicy;
23
24tokio::task_local! {
25    /// Os cookies desta requisição.
26    static CURRENT: Arc<CookieJar>;
27}
28
29/// O que chegou no cabeçalho `Cookie` e o que sairá em `Set-Cookie`.
30///
31/// O `Arc` existe porque o layer precisa recolher os pendentes **depois** de o
32/// handler terminar, e um task-local é consumido ao abrir o escopo: ele guarda a
33/// própria referência antes de entrar.
34pub(super) struct CookieJar {
35    /// O que o cliente apresentou, já parseado.
36    incoming: Vec<(String, String)>,
37
38    /// O que ainda vai sair na resposta.
39    ///
40    /// Não há disputa aqui: o escopo é de uma tarefa e as chamadas do handler
41    /// são sequenciais. O `Mutex` existe para dar `Sync` ao `Arc` que o layer
42    /// segura do lado de fora — sem ele, o `Arc` não atravessa o `.await` do
43    /// handler.
44    #[allow(
45        clippy::disallowed_types,
46        reason = "o .clippy.toml admite Mutex em recurso de borda justificado; é o Sync do handle que o layer segura fora do escopo, não estado compartilhado entre tarefas"
47    )]
48    pending: std::sync::Mutex<Vec<Cookie<'static>>>,
49}
50
51/// O adaptador que serve os cookies do escopo corrente.
52///
53/// ZST: os cookies são da tarefa, não do objeto.
54#[derive(Debug, Clone, Copy, Default)]
55pub(crate) struct CookieContext;
56
57impl CookieContext {
58    /// Lê o cabeçalho `Cookie` e monta o jar da requisição.
59    ///
60    /// `pub(super)`: só o layer irmão abre um escopo.
61    pub(super) fn open(headers: &HeaderMap) -> Arc<CookieJar> {
62        let incoming = headers
63            .get_all(header::COOKIE)
64            .iter()
65            .filter_map(|value| value.to_str().ok())
66            .flat_map(Cookie::split_parse)
67            .filter_map(Result::ok)
68            .map(|cookie| (cookie.name().to_owned(), cookie.value().to_owned()))
69            .collect();
70
71        Arc::new(CookieJar {
72            incoming,
73            #[allow(
74                clippy::disallowed_types,
75                reason = "ver o campo `pending` do CookieJar"
76            )]
77            pending: std::sync::Mutex::new(Vec::new()),
78        })
79    }
80
81    /// Roda `future` com este jar instalado na tarefa.
82    pub(super) async fn scope<F: Future>(jar: Arc<CookieJar>, future: F) -> F::Output {
83        CURRENT.scope(jar, future).await
84    }
85
86    /// O que o handler escreveu, para o layer carimbar na resposta.
87    ///
88    /// `unwrap_or_else(into_inner)` em vez de `unwrap`: um handler que entrou em
89    /// pânico segurando o lock envenena o mutex, e perder os cookies por causa
90    /// disso trocaria um `500` por um `500` mais uma sessão corrompida.
91    pub(super) fn drain(jar: &CookieJar) -> Vec<Cookie<'static>> {
92        std::mem::take(
93            &mut *jar
94                .pending
95                .lock()
96                .unwrap_or_else(std::sync::PoisonError::into_inner),
97        )
98    }
99
100    /// Monta o cookie com a política que vale para todos.
101    ///
102    /// `HttpOnly` sempre. É o que impede um XSS de ler o token por JavaScript —
103    /// a diferença entre um script injetado poder incomodar o usuário e poder
104    /// roubar a sessão dele inteira.
105    fn build(name: CookieName, value: &str, ttl: std::time::Duration) -> Cookie<'static> {
106        let seconds = i64::try_from(ttl.as_secs()).unwrap_or(i64::MAX);
107
108        Cookie::build((name.as_str(), value.to_owned()))
109            .path("/")
110            .max_age(cookie::time::Duration::seconds(seconds))
111            .http_only(true)
112            .secure(SessionPolicy::SECURE)
113            .same_site(SessionPolicy::SAME_SITE)
114            .build()
115    }
116
117    /// Enfileira um cookie para sair na resposta.
118    fn push(cookie: Cookie<'static>) -> Result<(), ApiError> {
119        Self::jar()?
120            .pending
121            .lock()
122            .unwrap_or_else(std::sync::PoisonError::into_inner)
123            .push(cookie);
124
125        Ok(())
126    }
127
128    /// O jar desta tarefa.
129    ///
130    /// Falha fora do escopo em vez de descartar o cookie em silêncio. Descartar
131    /// seria um login que responde `200` sem logar ninguém — o modo de falha
132    /// mais caro que este desenho pode ter.
133    fn jar() -> Result<Arc<CookieJar>, ApiError> {
134        CURRENT.try_with(Arc::clone).map_err(|_| {
135            SystemLogger::get().error(
136                "o middleware de cookie não executou: a ordem dos layers do router está errada",
137                [],
138            );
139
140            ApiError::new(
141                axum::http::StatusCode::INTERNAL_SERVER_ERROR,
142                "cookies indisponíveis",
143            )
144        })
145    }
146
147    /// Quanto tempo o cookie deste nome vale.
148    const fn ttl_of(name: CookieName) -> std::time::Duration {
149        match name {
150            CookieName::Access => SessionPolicy::ACCESS_TTL,
151            CookieName::Refresh => SessionPolicy::REFRESH_TTL,
152        }
153    }
154}
155
156impl CookiePort for CookieContext {
157    fn read(&self, name: CookieName) -> Result<Option<String>, ApiError> {
158        Ok(Self::jar()?
159            .incoming
160            .iter()
161            .find(|(key, _)| key == name.as_str())
162            .map(|(_, value)| value.clone())
163            .filter(|value| !value.is_empty()))
164    }
165
166    fn set(&self, name: CookieName, value: &str) -> Result<(), ApiError> {
167        Self::push(Self::build(name, value, Self::ttl_of(name)))
168    }
169
170    /// Apagar é emitir o mesmo cookie já vencido.
171    ///
172    /// Reaproveitar o [`Self::build`] fecha a chance de o cookie de logout sair
173    /// com `Path` ou `SameSite` diferente do de login — caso em que o navegador
174    /// guarda os dois e a sessão não morre.
175    fn clear(&self, name: CookieName) -> Result<(), ApiError> {
176        Self::push(Self::build(name, "", std::time::Duration::ZERO))
177    }
178}
