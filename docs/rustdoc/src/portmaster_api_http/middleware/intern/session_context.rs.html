1//! A sessão da tarefa corrente.
2//!
3//! > **Lacuna conhecida do `lint-exports`:** os task-locals abaixo nascem de
4//! > `tokio::task_local!`, e itens gerados por macro são invisíveis ao `syn`.
5//! > O arquivo exporta três itens na prática, não um.
6
7use std::future::Future;
8
9use portmaster_app::context::UserContext;
10use portmaster_app::{Logger as _, SystemLogger};
11
12use crate::middleware::session_port::SessionPort;
13use crate::ports::error::api_error::ApiError;
14
15tokio::task_local! {
16    /// Se o middleware de token já rodou nesta requisição.
17    ///
18    /// Separado do usuário de propósito: "o token foi conferido e não havia
19    /// sessão" e "ninguém conferiu o token ainda" são situações diferentes, e
20    /// guardar só o usuário as tornaria indistinguíveis.
21    static VALIDATED: bool;
22
23    /// O usuário da requisição, quando há sessão.
24    static USER: Option<UserContext>;
25}
26
27/// O adaptador que serve a sessão do escopo corrente.
28///
29/// ZST: a sessão é da tarefa, não do objeto.
30#[derive(Debug, Clone, Copy, Default)]
31pub(crate) struct SessionContext;
32
33impl SessionContext {
34    /// Roda `future` com o resultado da validação de token instalado.
35    ///
36    /// `pub(super)`: é o escritor, e só o layer irmão o alcança. É a garantia
37    /// que o desenho existe para dar — de fora deste módulo não há como
38    /// acrescentar um usuário à sessão sem passar pela conferência do token.
39    pub(super) async fn scope<F: Future>(user: Option<UserContext>, future: F) -> F::Output {
40        VALIDATED.scope(true, USER.scope(user, future)).await
41    }
42
43    /// Confirma que o escopo de sessão está instalado.
44    ///
45    /// Não há caminho que marque `false` hoje: o middleware sempre entra no
46    /// escopo com `true`, com ou sem sessão. O braço existe para o dia em que
47    /// alguém acrescentar um estado intermediário.
48    fn gate() -> Result<(), ApiError> {
49        match VALIDATED.try_with(|validated| *validated) {
50            Ok(true) => Ok(()),
51            Ok(false) => Err(ApiError::unauthenticated()),
52            Err(_) => {
53                SystemLogger::get().error(
54                    "o middleware de token não executou: a ordem dos layers do router está errada",
55                    [],
56                );
57
58                Err(ApiError::new(
59                    axum::http::StatusCode::INTERNAL_SERVER_ERROR,
60                    "sessão indisponível",
61                ))
62            }
63        }
64    }
65}
66
67impl SessionPort for SessionContext {
68    fn current_user(&self) -> Result<Option<UserContext>, ApiError> {
69        Self::gate()?;
70
71        Ok(USER.try_with(Clone::clone).ok().flatten())
72    }
73
74    fn require_user(&self) -> Result<UserContext, ApiError> {
75        self.current_user()?.ok_or_else(ApiError::unauthenticated)
76    }
77}

79#[cfg(test)]
80mod tests {
81    use super::*;
82    use axum::http::StatusCode;
83    use portmaster_app::context::RoleContext;
84    use pretty_assertions::assert_eq;
85
86    fn context() -> UserContext {
87        UserContext {
88            id: "u1".into(),
89            name: "Ana".into(),
90            email: "ana@portmaster.local".into(),
91            roles: vec![RoleContext {
92                id: "r1".into(),
93                name: "Operador".into(),
94                permissions: vec!["container:seal".into()],
95            }],
96        }
97    }
98
99    #[tokio::test]
100    async fn dentro_do_escopo_a_sessao_esta_disponivel() {
101        SessionContext::scope(Some(context()), async {
102            let user = SessionContext.require_user().expect("há sessão");
103            assert_eq!(user.id, "u1");
104        })
105        .await;
106    }
107
108    /// O middleware rodou e não achou token — que é diferente de não ter
109    /// rodado.
110    #[tokio::test]
111    async fn rota_publica_tem_escopo_mas_nao_usuario() {
112        SessionContext::scope(None, async {
113            assert_eq!(SessionContext.current_user().unwrap(), None);
114            assert_eq!(
115                SessionContext.require_user().err().map(|e| e.status()),
116                Some(StatusCode::UNAUTHORIZED)
117            );
118        })
119        .await;
120    }
121
122    /// 500 e não 401: o cliente não fez nada errado — a pilha do router está
123    /// montada fora de ordem, e responder 401 esconderia isso atrás de um
124    /// "faça login" que nunca vai funcionar.
125    #[tokio::test]
126    async fn sem_o_middleware_o_erro_e_nosso_e_nao_do_cliente() {
127        assert_eq!(
128            SessionContext.current_user().err().map(|e| e.status()),
129            Some(StatusCode::INTERNAL_SERVER_ERROR)
130        );
131    }
132
133    #[tokio::test]
134    async fn escopos_de_requisicoes_diferentes_nao_se_misturam() {
135        let primeira = tokio::spawn(SessionContext::scope(Some(context()), async {
136            SessionContext.current_user().unwrap().map(|u| u.id)
137        }));
138
139        let segunda = tokio::spawn(SessionContext::scope(None, async {
140            SessionContext.current_user().unwrap().map(|u| u.id)
141        }));
142
143        let (primeira, segunda) = tokio::join!(primeira, segunda);
144
145        assert_eq!(primeira.unwrap(), Some("u1".to_owned()));
146        assert_eq!(segunda.unwrap(), None);
147    }
