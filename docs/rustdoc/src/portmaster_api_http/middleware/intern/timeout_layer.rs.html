17/// Desiste da requisição depois do prazo.
18///
19/// ## O layer é o serviço antes de saber o que envolve
20///
21/// Um tipo só, e não um par. `TimeoutLayer` sem parâmetro é o `Layer` — o teto
22/// configurado, ainda sem serviço interno —, e `TimeoutLayer<S>` é o `Service`
23/// que sai do `layer()`: o mesmo teto, agora com o `S` que ele embrulha. Eram
24/// dois tipos em dois arquivos, e o segundo nunca foi nomeado por ninguém — quem
25/// monta a pilha aplica o layer e mais nada.
26#[derive(Clone, Copy)]
27pub(crate) struct TimeoutLayer<S = ()> {
28    /// O serviço interno, que este envolve; `()` enquanto é só layer.
29    inner: S,
30    /// Teto de tempo de uma requisição.
31    limit: Duration,
32}
33
34impl TimeoutLayer {
35    /// Monta o layer com o teto configurado.
36    pub(crate) const fn new(limit: Duration) -> Self {
37        Self { inner: (), limit }
38    }
39}
40
41impl<S> Layer<S> for TimeoutLayer {
42    type Service = TimeoutLayer<S>;
43
44    fn layer(&self, inner: S) -> Self::Service {
45        TimeoutLayer {
46            inner,
47            limit: self.limit,
48        }
49    }
50}
51
52impl<S> Service<Request> for TimeoutLayer<S>
53where
54    S: Service<Request, Response = Response> + Clone + Send + 'static,
55    S::Future: Send + 'static,
56{
57    type Response = Response;
58    type Error = S::Error;
59    type Future = BoxFuture<'static, Result<Response, Self::Error>>;
60
61    fn poll_ready(&mut self, cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
62        self.inner.poll_ready(cx)
63    }
64
65    fn call(&mut self, request: Request) -> Self::Future {
66        let clone = self.inner.clone();
67        let mut inner = std::mem::replace(&mut self.inner, clone);
68        let limit = self.limit;
69
70        Box::pin(async move {
71            tokio::select! {
72                biased;
73
74                response = inner.call(request) => response,
75
76                () = tokio::time::sleep(limit) => {
77                    SystemLogger::get()
78                        .warn(
79                            "requisição excedeu o teto de tempo",
80                            [("limit_ms", &limit.as_millis().to_string())],
81                        );
82
83                    let (status, problem) = ApiError::new(
84                        StatusCode::GATEWAY_TIMEOUT,
85                        "The request took too long to complete.",
86                    )
87                    .into_parts();
88
89                    Ok(EncodeContext.respond(status, &problem))
90                }
91            }
92        })
93    }
94}

96#[cfg(test)]
97mod tests {
98    use super::*;
99    use axum::response::IntoResponse as _;
100    use tower::{ServiceBuilder, ServiceExt};
101
102    #[tokio::test]
103    async fn o_que_demora_demais_vira_504() {
104        let service = ServiceBuilder::new()
105            .layer(TimeoutLayer::new(Duration::from_millis(10)))
106            .service_fn(|_: Request| async move {
107                tokio::time::sleep(Duration::from_secs(30)).await;
108                Ok::<_, std::convert::Infallible>(Response::new(axum::body::Body::empty()))
109            });
110
111        let response = service
112            .oneshot(Request::new(axum::body::Body::empty()))
113            .await
114            .unwrap();
115
116        assert_eq!(response.status(), StatusCode::GATEWAY_TIMEOUT);
117    }
118
119    #[tokio::test]
120    async fn o_que_responde_a_tempo_atravessa() {
121        let service = ServiceBuilder::new()
122            .layer(TimeoutLayer::new(Duration::from_secs(30)))
123            .service_fn(|_: Request| async move {
124                Ok::<_, std::convert::Infallible>((StatusCode::CREATED, "rápido").into_response())
125            });
126
127        let response = service
128            .oneshot(Request::new(axum::body::Body::empty()))
129            .await
130            .unwrap();
131
132        assert_eq!(response.status(), StatusCode::CREATED);
133    }
