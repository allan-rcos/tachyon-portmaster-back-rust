1//! O contrato de quem escreve a resposta no formato negociado.

3use axum::http::StatusCode;
4use axum::response::Response;
5
6use crate::wire::x::response_x::ResponseX;

8/// Escreve um VO de resposta no formato que esta requisição negociou.
9///
10/// Genérica sobre o VO, e por isso **não é object-safe** — que é o ponto. Cada
11/// par (formato, VO) é monomorfizado: o que varia em tempo de execução é só a
12/// variante de formato guardada no escopo, e escolher entre elas é um `match`,
13/// não uma vTable.
14///
15/// Repare no que **não** está aqui: nada que revele o formato. Quem tem esta
16/// porta na mão consegue responder, e é só isso — não descobre em que formato
17/// nem escolhe outro. Escolher é do middleware, uma vez por requisição.
18///
19/// ## Por que ela substituiu um extractor
20///
21/// O `Encoder` era um extractor que toda rota declarava e repassava para o
22/// `ApiResponse`. Trinta assinaturas carregando um argumento que ninguém lia, só
23/// para levá-lo do axum até o construtor da resposta. Agora o formato está no
24/// escopo da requisição, e quem responde pede a porta.
25pub(crate) trait EncodePort: Clone + Send + Sync + 'static {
26    /// A resposta completa: corpo codificado e tipo de mídia.
27    ///
28    /// É o único jeito de tirar bytes daqui, e é de propósito: se codificar e
29    /// carimbar o cabeçalho fossem duas chamadas, existiria um caminho em que a
30    /// primeira acontece sem a segunda — um corpo saindo com o tipo de mídia
31    /// errado, que é exatamente o defeito que este desenho fecha.
32    ///
33    /// Cookie não passa por aqui. Quem os carimba é o layer de cookie, depois do
34    /// handler e para toda resposta — inclusive a de erro.
35    ///
36    /// Falha na serialização vira `502` e sai como corpo de problema — pelo
37    /// mesmo formato, para que nem o erro do erro escape da negociação.
38    fn respond<X: ResponseX>(&self, status: StatusCode, body: &X) -> Response;
39}
