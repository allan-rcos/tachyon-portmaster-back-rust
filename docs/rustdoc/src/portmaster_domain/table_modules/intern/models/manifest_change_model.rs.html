1//! A implementação de domínio do efeito de um movimento de carga.
2
3use crate::domain::{Container, ManifestCargo, ManifestChange};
4use crate::enums::TelemetryEvent;
5
6/// A implementação do domínio de [`ManifestChange`].
7pub struct ManifestChangeModel {
8    /// O contêiner **depois** do movimento, com peso e status já recalculados.
9    container: Box<dyn Container>,
10    /// O produto movimentado, em base62.
11    product_id: String,
12    /// A linha de manifesto resultante, ou `None` quando o movimento a zerou.
13    cargo: Option<Box<dyn ManifestCargo>>,
14    /// Se a linha deve sair do manifesto em vez de ser atualizada.
15    ///
16    /// É o desembarque que levou a quantidade a zero: manter a linha com
17    /// zero faria o manifesto crescer com registros que não são carga.
18    clear_manifest: bool,
19    /// O evento a registrar na telemetria — embarque ou desembarque.
20    event: TelemetryEvent,
21}
22
23impl ManifestChangeModel {
24    /// Monta o efeito de um movimento de carga.
25    pub(crate) fn new(
26        container: Box<dyn Container>,
27        product_id: String,
28        cargo: Option<Box<dyn ManifestCargo>>,
29        clear_manifest: bool,
30        event: TelemetryEvent,
31    ) -> Self {
32        Self {
33            container,
34            product_id,
35            cargo,
36            clear_manifest,
37            event,
38        }
39    }
40}
41
42impl ManifestChange for ManifestChangeModel {
43    fn container(&self) -> &dyn Container {
44        self.container.as_ref()
45    }
46
47    fn product_id(&self) -> &str {
48        &self.product_id
49    }
50
51    fn cargo(&self) -> Option<&dyn ManifestCargo> {
52        self.cargo.as_deref()
53    }
54
55    fn clear_manifest(&self) -> bool {
56        self.clear_manifest
57    }
58
59    fn event(&self) -> TelemetryEvent {
60        self.event
61    }
62
63    fn into_container(self: Box<Self>) -> Box<dyn Container> {
64        self.container
65    }
66}
