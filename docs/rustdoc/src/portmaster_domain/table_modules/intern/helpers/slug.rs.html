1//! O slug de metadado que permissão, grupo de marcador e marcador compartilham.
2//!
3//! Um slug é o nome estável de um metadado de sistema — `metrics`,
4//! `refresh-token`, `container:create`. Permissão tem um formato composto (dois
5//! tokens separados por `:`), os outros dois usam um token só, mas as regras de
6//! vazio, de comprimento e de como um segmento kebab é formado são as mesmas nos
7//! três. Ficam aqui uma vez só.
8
9use nutype::nutype;
10
11/// Um token de metadado em lower-kebab: `refresh-token`, `metrics`.
12///
13/// Quem valida um slug composto — a permissão — usa este tipo em cada metade,
14/// e confere o todo por conta própria.
15#[nutype(validate(not_empty, len_char_max = 64, predicate = Slug::is_kebab_token))]
16pub(crate) struct Slug(String);
17
18impl Slug {
19    /// Comprimento máximo de um slug.
20    ///
21    /// Casa com o limite que o modelo anterior validava, e mantém a chave curta
22    /// — ela é lida a cada verificação de permissão.
23    pub(crate) const MAX_LENGTH: usize = 64;
24
25    /// Se o valor é um token em lower-kebab.
26    ///
27    /// Começa por letra minúscula, segue com minúsculas e dígitos, e admite
28    /// hífen entre segmentos — nunca no começo, no fim, nem dobrado.
29    pub(crate) fn is_kebab_token(value: &str) -> bool {
30        let mut segments = value.split('-');
31
32        // O primeiro segmento tem a regra extra de começar por letra.
33        let Some(first) = segments.next() else {
34            return false;
35        };
36        if !first.starts_with(|c: char| c.is_ascii_lowercase()) {
37            return false;
38        }
39        if !first
40            .chars()
41            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit())
42        {
43            return false;
44        }
45
46        segments.all(|segment| {
47            !segment.is_empty()
48                && segment
49                    .chars()
50                    .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit())
51        })
52    }
53}

55#[cfg(test)]
56mod tests {
57    use super::*;
58
59    #[test]
60    fn aceita_tokens_kebab_validos() {
61        for good in ["metrics", "refresh-token", "rate-limit", "a1", "x-9-y"] {
62            assert!(Slug::try_new(good).is_ok(), "deveria aceitar {good}");
63        }
64    }
65
66    #[test]
67    fn recusa_o_que_nao_e_kebab() {
68        for bad in [
69            "",
70            "-leading",
71            "trailing-",
72            "double--hyphen",
73            "Upper",
74            "9leading",
75            "with_underscore",
76            "with space",
77        ] {
78            assert!(Slug::try_new(bad).is_err(), "deveria recusar {bad:?}");
79        }
80    }
81
82    #[test]
83    fn recusa_token_longo_demais() {
84        let long = "a".repeat(Slug::MAX_LENGTH.saturating_add(1));
85
86        assert!(matches!(
87            Slug::try_new(long),
88            Err(SlugError::LenCharMaxViolated)
89        ));
90    }
