11/// O slug de uma permissão: `domain:action`, cada lado um token kebab.
12///
13/// O comprimento é conferido no todo, e não em cada metade: são as duas juntas
14/// que viram a chave lida a cada verificação de acesso.
15#[nutype(validate(not_empty, len_char_max = 64, predicate = is_domain_action))]
16struct PermissionSlug(String);
17
18/// A implementação. Sem helpers: é validação de formato e nada mais.
19#[derive(Clone)]
20pub(crate) struct PermissionTMImpl;
21
22impl PermissionTMImpl {
23    /// Monta o `TableModule`.
24    pub(crate) const fn new() -> Self {
25        Self
26    }
27}
28
29impl PermissionTM for PermissionTMImpl {
30    fn create(&self, slug: String) -> Result<Box<dyn Permission>, MetadataError> {
31        let slug = PermissionSlug::try_new(slug)
32            .map_err(|error| MetadataError::Validation(vec![refused(&error)]))?;
33
34        Ok(Box::new(PermissionModel::new(slug.into_inner())))
35    }
36}
37
38/// Se o valor é `domain:action`, com os dois lados em lower-kebab.
39fn is_domain_action(value: &str) -> bool {
40    let mut parts = value.split(':');
41
42    matches!(
43        (parts.next(), parts.next(), parts.next()),
44        (Some(domain), Some(action), None)
45            if Slug::is_kebab_token(domain) && Slug::is_kebab_token(action)
46    )
47}
48
49/// Traduz a recusa do slug na mensagem que o cliente lê.
50fn refused(error: &PermissionSlugError) -> FieldError {
51    match *error {
52        PermissionSlugError::NotEmptyViolated => FieldError::new("slug", "Slug is required."),
53        PermissionSlugError::LenCharMaxViolated => FieldError::new(
54            "slug",
55            format!("Slug must not exceed {} characters.", Slug::MAX_LENGTH),
56        ),
57        PermissionSlugError::PredicateViolated => FieldError::new(
58            "slug",
59            r#"Slug must follow "domain:action" in lower-kebab (e.g. product:create)."#,
60        ),
61    }
62}
