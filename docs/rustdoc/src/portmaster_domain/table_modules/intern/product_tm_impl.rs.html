12/// O nome de um produto.
13#[nutype(sanitize(trim), validate(not_empty, len_char_max = 255))]
14struct ProductName(String);
15
16/// A densidade de um produto, em toneladas por unidade.
17///
18/// O `finite` vem antes do `greater`, e não é redundante: `f64::NAN > 0.0` é
19/// **falso**, então um NaN passaria pela comparação sozinha — e contaminaria
20/// todo cálculo de peso do contêiner adiante.
21#[nutype(validate(finite, greater = 0.0))]
22struct Density(f64);
23
24/// A implementação, genérica sobre o gerador de id.
25#[derive(Clone)]
26pub(crate) struct ProductTMImpl<G> {
27    /// De onde sai a identidade de um produto novo.
28    id_generator: G,
29}
30
31impl<G: DatabaseIdGenerator> ProductTMImpl<G> {
32    /// Monta o `TableModule` com o seu gerador de id.
33    pub(crate) const fn new(id_generator: G) -> Self {
34        Self { id_generator }
35    }
36
37    /// Examina nome e densidade, acumulando o que estiver errado.
38    ///
39    /// Os dois `try_new` acontecem antes de qualquer retorno: quem errou os dois
40    /// campos recebe os dois de volta.
41    fn checked(name: String, density: f64) -> Result<(ProductName, Density), ProductError> {
42        let checked_name = ProductName::try_new(name);
43        let checked_density = Density::try_new(density);
44
45        let mut errors = Vec::new();
46        if let Err(error) = &checked_name {
47            errors.push(name_refused(error));
48        }
49        if checked_density.is_err() {
50            errors.push(FieldError::new(
51                "density",
52                "Density must be greater than zero.",
53            ));
54        }
55
56        let (Ok(name), Ok(density)) = (checked_name, checked_density) else {
57            return Err(ProductError::Validation(errors));
58        };
59
60        Ok((name, density))
61    }
62}
63
64impl<G: DatabaseIdGenerator> ProductTM for ProductTMImpl<G> {
65    fn create(
66        &self,
67        name: String,
68        density: f64,
69        risk_class: RiskClass,
70    ) -> Result<Box<dyn Product>, ProductError> {
71        let (name, density) = Self::checked(name, density)?;
72
73        Ok(Box::new(ProductModel::new(
74            self.id_generator.next(),
75            name.into_inner(),
76            density.into_inner(),
77            risk_class,
78        )))
79    }
80
81    fn update(
82        &self,
83        product: &dyn Product,
84        name: String,
85        density: f64,
86        risk_class: RiskClass,
87    ) -> Result<Box<dyn Product>, ProductError> {
88        let (name, density) = Self::checked(name, density)?;
89
90        let mut model = ProductModel::from_domain(product);
91        model.set_details(name.into_inner(), density.into_inner(), risk_class);
92        Ok(Box::new(model))
93    }
94}

96/// Comprimento máximo do nome, casando com a coluna `VARCHAR(255)`.
97const MAX_NAME_LENGTH: usize = 255;
98
99/// Traduz a recusa do nome na mensagem que o cliente lê.
100fn name_refused(error: &ProductNameError) -> FieldError {
101    match *error {
102        ProductNameError::NotEmptyViolated => FieldError::new("name", "Name is required."),
103        ProductNameError::LenCharMaxViolated => FieldError::new(
104            "name",
105            format!("Name must not exceed {MAX_NAME_LENGTH} characters."),
106        ),
107    }
108}
