3use crate::error::{RoleError, Validation};
4use crate::id::IntIdGenerator;

9/// A implementação, genérica sobre o gerador de id.
10pub(crate) struct RoleTMImpl<G> {
11    /// De onde sai a identidade de um papel novo.
12    id_generator: G,
13}
14
15impl<G: IntIdGenerator> RoleTMImpl<G> {
16    /// Monta o `TableModule` com o seu gerador de id.
17    pub(crate) const fn new(id_generator: G) -> Self {
18        Self { id_generator }
19    }
20
21    /// Examina o nome, acumulando o que estiver errado.
22    fn validate(name: &str) -> Validation {
23        let mut errors = Validation::new();
24
25        if name.trim().is_empty() {
26            errors.add("name", "Name is required.");
27        } else if name.chars().count() > MAX_NAME_LENGTH {
28            errors.add(
29                "name",
30                format!("Name must not exceed {MAX_NAME_LENGTH} characters."),
31            );
32        }
33
34        errors
35    }
36}
37
38impl<G: IntIdGenerator> RoleTM for RoleTMImpl<G> {
39    fn create(&self, name: String, permissions: Vec<String>) -> Result<Box<dyn Role>, RoleError> {
40        Self::validate(&name)
41            .into_result(())
42            .map_err(RoleError::Validation)?;
43
44        Ok(Box::new(RoleModel::new(
45            self.id_generator.next(),
46            name,
47            permissions,
48        )))
49    }
50
51    /// Substitui o conjunto de permissões do papel.
52    ///
53    /// Os slugs **não** são validados aqui: quem os produz é o registro de
54    /// permissões do próprio sistema, não o cliente. Conferir se um slug existe
55    /// é trabalho do `app`, que tem o catálogo à mão.
56    fn update_permissions(
57        &self,
58        role: &dyn Role,
59        permissions: Vec<String>,
60    ) -> Result<Box<dyn Role>, RoleError> {
61        let mut model = RoleModel::from_domain(role);
62        model.set_permissions(permissions);
63        Ok(Box::new(model))
64    }
65}
66
67/// Comprimento máximo do nome, casando com a coluna `VARCHAR(255)`.
68const MAX_NAME_LENGTH: usize = 255;

70#[cfg(test)]
71mod tests {
72    use super::*;
73    use crate::table_modules::interno::fields_of::fields_of;
74    use pretty_assertions::assert_eq;
75
76    struct FixedIdGenerator;
77    impl IntIdGenerator for FixedIdGenerator {
78        fn next(&self) -> String {
79            "9Z8y".to_string()
80        }
81    }
82
83    fn table_module() -> RoleTMImpl<FixedIdGenerator> {
84        RoleTMImpl::new(FixedIdGenerator)
85    }
86
87    #[test]
88    fn cria_papel_valido() {
89        let role = table_module()
90            .create("Operador".into(), vec!["container:read".into()])
91            .expect("os dados são válidos");
92
93        assert_eq!(role.id(), "9Z8y");
94        assert_eq!(role.name(), "Operador");
95        assert_eq!(role.permissions(), ["container:read"]);
96    }
97
98    #[test]
99    fn recusa_nome_vazio_ou_longo_demais() {
100        for bad in ["", "   ", &"x".repeat(MAX_NAME_LENGTH + 1)] {
101            let error = table_module()
102                .create(bad.into(), Vec::new())
103                .err()
104                .expect("nome inválido deve falhar");
105
106            let RoleError::Validation(fields) = error;
107            assert_eq!(fields_of(&fields), vec!["name"]);
108        }
109    }
110
111    /// Um slug omitido é uma permissão **revogada**.
112    ///
113    /// Se isto virasse merge, não haveria como tirar uma permissão de um papel.
114    #[test]
115    fn atualizar_permissoes_substitui_em_vez_de_somar() {
116        let role = table_module()
117            .create("Operador".into(), vec!["a:read".into(), "a:write".into()])
118            .expect("os dados são válidos");
119
120        let updated = table_module()
121            .update_permissions(role.as_ref(), vec!["a:read".into()])
122            .expect("a substituição não valida slugs");
123
124        assert_eq!(updated.permissions(), ["a:read"]);
125        assert_eq!(role.permissions(), ["a:read", "a:write"]);
126    }
