1//! A validação de slug que permissão e grupo de marcador compartilham.
2//!
3//! Um slug é o nome estável de um metadado de sistema — `metrics`,
4//! `refresh-token`, `container.create`. Permissão e grupo têm formatos
5//! diferentes (uma admite ponto, o outro não), mas as regras de vazio, de
6//! comprimento e de como um segmento kebab é formado são as mesmas nos dois.
7//! Ficam aqui uma vez só.
8//!
9//! É um namespace, não um valor: o molde é o `SearchKey`/`Base62` do PHP —
10//! consts e funções associadas, sem construtor.
11
12use crate::error::{MetadataError, Validation};
13
14/// Regras de forma de um slug de metadado.
15pub(crate) struct Slug;
16
17impl Slug {
18    /// Comprimento máximo de um slug.
19    ///
20    /// Casa com o limite que o modelo anterior validava, e mantém a chave curta
21    /// — ela é lida a cada verificação de permissão.
22    pub(crate) const MAX_LENGTH: usize = 64;
23
24    /// Valida um token em lower-kebab: `refresh-token`, `metrics`.
25    ///
26    /// Começa por letra minúscula, segue com minúsculas e dígitos, e admite
27    /// hífen entre segmentos — nunca no começo, no fim, nem dobrado.
28    pub(crate) fn is_kebab_token(value: &str) -> bool {
29        if value.is_empty() {
30            return false;
31        }
32
33        let mut segments = value.split('-');
34
35        // O primeiro segmento tem a regra extra de começar por letra.
36        let Some(first) = segments.next() else {
37            return false;
38        };
39        if !first.starts_with(|c: char| c.is_ascii_lowercase()) {
40            return false;
41        }
42        if !first
43            .chars()
44            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit())
45        {
46            return false;
47        }
48
49        segments.all(|segment| {
50            !segment.is_empty()
51                && segment
52                    .chars()
53                    .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit())
54        })
55    }
56
57    /// Valida um slug composto e devolve os erros acumulados.
58    ///
59    /// `shape` descreve o formato esperado na mensagem, para que permissão e
60    /// grupo digam ao cliente coisas diferentes sem duplicar a lógica.
61    pub(crate) fn validate(slug: &str, valid: bool, shape: &str) -> Result<(), MetadataError> {
62        let mut errors = Validation::new();
63
64        if slug.is_empty() {
65            errors.add("slug", "Slug is required.");
66        } else if !valid {
67            errors.add("slug", shape);
68        } else if slug.chars().count() > Self::MAX_LENGTH {
69            errors.add(
70                "slug",
71                format!("Slug must not exceed {} characters.", Self::MAX_LENGTH),
72            );
73        }
74
75        errors.into_result(()).map_err(MetadataError::Validation)
76    }
77}

79#[cfg(test)]
80mod tests {
81    use super::*;
82
83    #[test]
84    fn aceita_tokens_kebab_validos() {
85        for good in ["metrics", "refresh-token", "rate-limit", "a1", "x-9-y"] {
86            assert!(Slug::is_kebab_token(good), "deveria aceitar {good}");
87        }
88    }
89
90    #[test]
91    fn recusa_o_que_nao_e_kebab() {
92        for bad in [
93            "",
94            "-leading",
95            "trailing-",
96            "double--hyphen",
97            "Upper",
98            "9leading",
99            "with_underscore",
100            "with space",
101        ] {
102            assert!(!Slug::is_kebab_token(bad), "deveria recusar {bad:?}");
103        }
104    }
105}
