8/// Os factories do `domain`.
9///
10/// Cada método devolve `impl Trait`: o consumidor recebe o **contrato**, nunca o
11/// tipo concreto. O despacho é estático — o compilador monomorfiza o grafo
12/// inteiro, então um serviço que não pode ser construído é erro de compilação,
13/// não surpresa em produção.
14pub trait DomainProvider {
15    /// `TableModule` de usuário.
16    fn user_table_module(&self) -> impl UserTM + Send + Sync;
17
18    /// `TableModule` de papel.
19    fn role_table_module(&self) -> impl RoleTM + Send + Sync;
20
21    /// `TableModule` de produto.
22    fn product_table_module(&self) -> impl ProductTM + Send + Sync;
23
24    /// `TableModule` de contêiner.
25    fn container_table_module(&self) -> impl ContainerTM + Send + Sync;
26
27    /// `TableModule` de manifesto.
28    fn manifest_table_module(&self) -> impl ManifestTM + Send + Sync;
29
30    /// `TableModule` de autenticação.
31    fn auth_table_module(&self) -> impl AuthTM + Send + Sync;
32
33    /// `TableModule` de permissão.
34    fn permission_table_module(&self) -> impl PermissionTM + Send + Sync;
35
36    /// `TableModule` de grupo de marcador.
37    fn marker_group_table_module(&self) -> impl MarkerGroupTM + Send + Sync;
38
39    /// `TableModule` de marcador.
40    fn marker_table_module(&self) -> impl MarkerTM + Send + Sync;
41}
