1//! O contrato de leitura de `User`.
2
3use chrono::{DateTime, Utc};
4
5use crate::domain::Role;
6
7/// Alguém que faz login, com os papéis que carrega.
8///
9/// Permissão nunca é concedida direto a uma pessoa — ela vem pelo papel, e é a
10/// lista de slugs do papel que o guarda de autorização confere.
11///
12/// O trait é **somente-leitura**: só existem getters. Quem o recebe consegue ler
13/// o usuário mas não alterá-lo, e é isso que impede o `app` ou o `api` de
14/// mudarem um e-mail sem passar pela validação do `TableModule`.
15pub trait User: Send + Sync {
16    /// Id em base62.
17    fn id(&self) -> &str;
18
19    /// Nome de exibição.
20    fn name(&self) -> &str;
21
22    /// E-mail, que é também o identificador de login.
23    fn email(&self) -> &str;
24
25    /// Hash Argon2id da senha — nunca a senha.
26    ///
27    /// A senha em claro atravessa só o `TableModule`, o tempo de ser validada e
28    /// hasheada; nada mais no sistema a retém.
29    fn password_hash(&self) -> &str;
30
31    /// Papéis atribuídos, na ordem em que foram concedidos.
32    fn roles(&self) -> &[Box<dyn Role>];
33
34    /// Quando a linha nasceu.
35    fn created_at(&self) -> DateTime<Utc>;
36
37    /// Quando mudou pela última vez.
38    fn updated_at(&self) -> DateTime<Utc>;
39
40    /// Quando foi removido, ou `None` enquanto vivo.
41    ///
42    /// Usuário é entidade forte: remover grava a data em vez de apagar a linha,
43    /// e toda leitura filtra por `None`.
44    fn deleted_at(&self) -> Option<DateTime<Utc>>;
45}
