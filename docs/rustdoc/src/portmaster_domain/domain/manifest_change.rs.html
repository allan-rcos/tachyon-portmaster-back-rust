1//! O contrato de leitura do efeito de um movimento de carga.
2
3use crate::domain::{Container, ManifestCargo};
4use crate::enums::TelemetryEvent;
5
6/// O efeito completo de um embarque ou desembarque.
7///
8/// Um único movimento de carga toca três coisas — o peso e o status do
9/// contêiner, a linha do manifesto, e o registro de telemetria — e as três
10/// precisam ser gravadas juntas ou nenhuma. Devolvê-las num objeto só é o que
11/// impede o `app` de aplicar metade.
12pub trait ManifestChange: Send + Sync {
13    /// O contêiner como fica depois do movimento.
14    fn container(&self) -> &dyn Container;
15
16    /// Produto movimentado, em base62.
17    fn product_id(&self) -> &str;
18
19    /// A linha do manifesto como fica, ou `None` se ela deixou de existir.
20    fn cargo(&self) -> Option<&dyn ManifestCargo>;
21
22    /// Se o manifesto inteiro deve ser apagado.
23    ///
24    /// Verdadeiro quando o desembarque esvaziou o contêiner: em vez de remover
25    /// linha a linha, o manifesto vai junto.
26    fn clear_manifest(&self) -> bool;
27
28    /// O que registrar na telemetria.
29    fn event(&self) -> TelemetryEvent;
30
31    /// Desmonta a mudança e entrega o contêiner resultante.
32    ///
33    /// Existe porque quem responde ao movimento precisa **publicar** o contêiner
34    /// no estado novo, e não só gravá-lo: a resposta de embarque leva o peso e o
35    /// status atualizados. Consumir a mudança é o que evita reler do banco o que
36    /// já está em memória — e, por consumi-la, garante que ninguém a persista de
37    /// novo depois de tê-la publicado.
38    fn into_container(self: Box<Self>) -> Box<dyn Container>;
39}
