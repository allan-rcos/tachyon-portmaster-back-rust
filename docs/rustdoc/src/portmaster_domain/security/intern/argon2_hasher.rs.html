1//! Hash de senha com Argon2id.
2//!
3//! Argon2id é o padrão recomendado hoje: além de custo de CPU, ele impõe custo
4//! de **memória**, o que é o que arruína um ataque em GPU — onde milhares de
5//! núcleos compensam bem a lentidão de CPU, mas não a falta de RAM por núcleo.
6
7use argon2::password_hash::{rand_core::OsRng, PasswordHash, PasswordHasher as _, SaltString};
8use argon2::{Argon2, PasswordVerifier};
9
10use crate::security::PasswordHasher;
11
12/// Monta o hasher com os parâmetros padrão do Argon2id.
13///
14/// Os parâmetros são do build, não de runtime: mudá-los invalidaria os hashes
15/// já gravados, que carregam os seus próprios no formato PHC.
16///
17/// Montar é barato — ler constantes —, e dois deles não têm nada a
18/// compartilhar. O que custa é derivar o hash, e isso é por chamada de
19/// [`PasswordHasher::hash`].
20pub(in crate::security) fn argon2_hasher(
21) -> impl PasswordHasher + Send + Sync + Clone + use<> + 'static {
22    Argon2Hasher {
23        argon: Argon2::default(),
24    }
25}
26
27/// Hasher de senha para armazenamento.
28#[derive(Clone)]
29struct Argon2Hasher {
30    /// O Argon2 já configurado, com os parâmetros de custo do build.
31    argon: Argon2<'static>,
32}
33
34impl PasswordHasher for Argon2Hasher {
35    /// Deriva o hash, com salt novo a cada chamada.
36    ///
37    /// O salt por chamada é o que faz duas contas com a mesma senha terem
38    /// hashes diferentes, e o que inutiliza uma rainbow table.
39    ///
40    /// ## O que acontece quando o Argon2 falha
41    ///
42    /// Ele só falha por parâmetro inválido, e aqui os parâmetros são constantes
43    /// — na prática, nunca. Ainda assim o caminho existe, e devolver string
44    /// vazia é a saída segura: ela não é um hash PHC válido, então `verify`
45    /// recusa qualquer senha contra ela. A conta fica inacessível em vez de
46    /// aceitar qualquer um.
47    fn hash(&self, plain: &str) -> String {
48        let salt = SaltString::generate(&mut OsRng);
49
50        match self.argon.hash_password(plain.as_bytes(), &salt) {
51            Ok(hash) => hash.to_string(),
52            Err(_) => String::new(),
53        }
54    }
55
56    /// Confere a tentativa contra o hash guardado.
57    ///
58    /// Hash corrompido ou em formato desconhecido responde `false`: não há como
59    /// conferir, e "não sei" tem que significar "não passa".
60    fn verify(&self, plain: &str, hash: &str) -> bool {
61        let Ok(parsed) = PasswordHash::new(hash) else {
62            return false;
63        };
64
65        self.argon
66            .verify_password(plain.as_bytes(), &parsed)
67            .is_ok()
68    }
69}
