1//! Digest de indexação com xxh64.
2//!
3//! Não é hash criptográfico e não deve ser usado como tal. O que se quer aqui é
4//! uma chave curta, determinística e barata — um marcador é consultado a cada
5//! refresh de token, e pagar Argon2 nisso seria absurdo.
6//!
7//! A saída são 16 caracteres hexadecimais, o que casa com a coluna `CHAR(16)`
8//! que o modelo anterior usava.
9
10use xxhash_rust::xxh64::xxh64;
11
12use crate::security::IndexHasher;
13
14/// Semente fixa. Precisa ser constante entre execuções: uma semente aleatória
15/// por processo faria toda marca gravada antes de um restart virar inalcançável.
16const SEED: u64 = 0;
17
18/// Monta o digest de lookup para marcadores.
19///
20/// O que sai é o contrato: trocar o xxh64 por outro digest de indexação não
21/// muda nenhuma assinatura fora deste arquivo — o que **não** pode mudar é a
22/// [`SEED`], porque ela é o que liga o digest às marcas já gravadas.
23pub(crate) const fn xx_index_hasher() -> impl IndexHasher + Send + Sync + Clone + use<> + 'static {
24    XxIndexHasher
25}
26
27/// Digest de lookup para marcadores.
28#[derive(Clone)]
29struct XxIndexHasher;
30
31impl IndexHasher for XxIndexHasher {
32    fn hash(&self, plain: &str) -> String {
33        format!("{:016x}", xxh64(plain.as_bytes(), SEED))
34    }
35}
