27/// A implementação do provider. Privada: nenhum crate exporta impl.
28pub(crate) struct DomainProviderImpl {
29    /// Quem é este servidor na composição do Snowflake.
30    secrets: DomainSecrets,
31}
32
33impl DomainProviderImpl {
34    /// Guarda a identidade de deploy e nada mais — não há recurso caro aqui.
35    pub(crate) const fn new(secrets: DomainSecrets) -> Self {
36        Self { secrets }
37    }
38
39    /// Serve o gerador de id de entidade.
40    ///
41    /// A impl é escolhida por **feature de compilação** — decisão de
42    /// arquitetura, resolvida no build, sem ramo em runtime. Os parâmetros de
43    /// identidade vêm dos segredos.
44    ///
45    /// Não está no [`DomainProvider`] de propósito: quem emite identidade de
46    /// entidade é o `TableModule`, e um gerador destes fora do crate permitiria
47    /// montar uma linha sem passar pela regra que a valida.
48    fn database_id_generator(&self) -> impl DatabaseIdGenerator + Clone + use<> + 'static {
49        #[cfg(feature = "id-snowflake")]
50        SnowflakeIdGenerator::new(self.secrets.cluster_id, self.secrets.server_id)
51    }
52}
53
54impl DomainProvider for DomainProviderImpl {
55    fn sequential_id_generator(&self) -> impl SequentialIdGenerator + use<> {
56        XidGenerator::new()
57    }
58
59    fn random_id_generator(&self) -> impl RandomIdGenerator + use<> {
60        NanoIdGenerator::new()
61    }
62
63    fn user_table_module(&self) -> impl UserTM + Clone + use<> + 'static {
64        UserTMImpl::new(self.database_id_generator(), Argon2Hasher::new())
65    }
66
67    fn role_table_module(&self) -> impl RoleTM + Clone + use<> + 'static {
68        RoleTMImpl::new(self.database_id_generator())
69    }
70
71    fn product_table_module(&self) -> impl ProductTM + Clone + use<> + 'static {
72        ProductTMImpl::new(self.database_id_generator())
73    }
74
75    fn container_table_module(&self) -> impl ContainerTM + Clone + use<> + 'static {
76        ContainerTMImpl::new(self.database_id_generator())
77    }
78
79    fn manifest_table_module(&self) -> impl ManifestTM + Clone + use<> + 'static {
80        ManifestTMImpl::new()
81    }
82
83    fn auth_table_module(&self) -> impl AuthTM + Clone + use<> + 'static {
84        AuthTMImpl::new(Argon2Hasher::new())
85    }
86
87    fn permission_table_module(&self) -> impl PermissionTM + Clone + use<> + 'static {
88        PermissionTMImpl::new()
89    }
90
91    fn marker_group_table_module(&self) -> impl MarkerGroupTM + Clone + use<> + 'static {
92        MarkerGroupTMImpl::new()
93    }
94
95    fn marker_table_module(&self) -> impl MarkerTM + Clone + use<> + 'static {
96        MarkerTMImpl::new(XxIndexHasher::new())
97    }
