1//! Snowflake — id de entidade, `i64` por dentro, base62 por fora.
2//!
3//! Escolhido para id de banco porque cabe num `BIGINT` de 8 bytes e ordena por
4//! tempo: o índice primário cresce pela ponta em vez de fragmentar, o que um
5//! UUID v4 não dá. O que o resto do sistema vê é sempre a string base62.
6//!
7//! ## Por que um `std::sync::Mutex`, que está banido
8//!
9//! O ban do `.clippy.toml` existe para impedir que estado de requisição vire
10//! estado global sincronizado — o caminho certo para aquilo é o task-local.
11//! Aqui é outra coisa: um recurso do **processo**, cuja seção crítica é o
12//! incremento de um contador e não contém `.await` nenhum. É exatamente o caso
13//! em que o mutex síncrono é o certo, e o assíncrono seria pior.
14
15#[allow(
16    clippy::disallowed_types,
17    reason = "recurso de processo com seção crítica sem await — ver o doc de SnowflakeIdGenerator"
18)]
19use std::sync::Mutex;
20use std::time::{Duration, UNIX_EPOCH};
21
22use snowflake::SnowflakeIdGenerator as Generator;
23
24use crate::id::{Base62, IntIdGenerator};
25
26/// Era do gerador: 2024-01-01T00:00:00Z, a mesma do PHP.
27///
28/// É uma constante de build e não um segredo de runtime — dois deploys com
29/// epochs diferentes emitiriam ids que se sobrepõem no tempo.
30const EPOCH_MS: u64 = 1_704_067_200_000;
31
32/// Gerador Snowflake compartilhado pelo processo.
33///
34/// O `Mutex` existe porque o algoritmo é sequencial por natureza: ele compara o
35/// timestamp atual com o da última emissão e incrementa um contador quando os
36/// dois caem no mesmo milissegundo. Duas threads gerando sem coordenação no
37/// mesmo milissegundo produziriam o mesmo id.
38#[allow(
39    clippy::disallowed_types,
40    reason = "gerador de ID Snowflake sequencial exige sincronização thread-safe entre requisições"
41)]
42pub struct SnowflakeIdGenerator {
43    /// O gerador, atrás de um `Mutex`.
44    ///
45    /// Um Snowflake guarda o último instante e a sequência dentro dele para
46    /// não repetir id no mesmo milissegundo, então há estado a proteger — e
47    /// ele é o único do `domain`.
48    inner: Mutex<Generator>,
49}
50
51impl SnowflakeIdGenerator {
52    /// Monta o gerador para esta instância de deploy.
53    ///
54    /// `cluster_id` e `server_id` são o que distingue dois processos emitindo ao
55    /// mesmo tempo; vêm dos segredos, não do build.
56    pub(crate) fn new(cluster_id: i32, server_id: i32) -> Self {
57        #[allow(
58            clippy::arithmetic_side_effects,
59            reason = "EPOCH_MS é constante de build; UNIX_EPOCH + 2024 não chega perto de estourar SystemTime"
60        )]
61        let epoch = UNIX_EPOCH + Duration::from_millis(EPOCH_MS);
62        #[allow(
63            clippy::disallowed_types,
64            reason = "recurso de processo com seção crítica sem await — ver o doc do tipo"
65        )]
66        Self {
67            inner: Mutex::new(Generator::with_epoch(cluster_id, server_id, epoch)),
68        }
69    }
70
71    /// Toma o gerador, ignorando o envenenamento do lock.
72    ///
73    /// Um lock envenenado significa que uma thread entrou em pânico segurando o
74    /// gerador. O estado interno continua íntegro — é só um contador e um
75    /// timestamp —, então seguir com ele é preferível a derrubar toda a emissão
76    /// de id pelo resto da vida do processo.
77    fn locked(&self) -> std::sync::MutexGuard<'_, Generator> {
78        self.inner
79            .lock()
80            .unwrap_or_else(std::sync::PoisonError::into_inner)
81    }
82}
83
84impl IntIdGenerator for SnowflakeIdGenerator {
85    fn next(&self) -> String {
86        Base62::encode(self.locked().real_time_generate())
87    }
88}

90#[cfg(test)]
91mod tests {
92    use super::*;
93    use std::collections::HashSet;
94
95    /// O caso que o `Mutex` existe para cobrir.
96    ///
97    /// Várias threads emitindo dentro do mesmo milissegundo.
98    #[test]
99    fn ids_sao_unicos_sob_concorrencia() {
100        let generator = std::sync::Arc::new(SnowflakeIdGenerator::new(1, 1));
101        let mut handles = Vec::new();
102
103        for _ in 0..8 {
104            let generator = generator.clone();
105            handles.push(std::thread::spawn(move || {
106                (0..500).map(|_| generator.next()).collect::<Vec<_>>()
107            }));
108        }
109
110        let mut all = HashSet::new();
111        for handle in handles {
112            for id in handle.join().expect("thread de emissão entrou em pânico") {
113                assert!(all.insert(id.clone()), "id repetido: {id}");
114            }
115        }
116
117        assert_eq!(all.len(), 4_000);
118    }
119
120    #[test]
121    fn ids_crescem_com_o_tempo() {
122        let generator = SnowflakeIdGenerator::new(1, 1);
123        let first = Base62::decode(&generator.next()).expect("id emitido deve decodificar");
124        let second = Base62::decode(&generator.next()).expect("id emitido deve decodificar");
125        assert!(second > first, "{second} deveria vir depois de {first}");
126    }
