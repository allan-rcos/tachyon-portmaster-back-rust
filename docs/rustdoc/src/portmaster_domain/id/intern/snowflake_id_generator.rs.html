1//! Snowflake — id de entidade, inteiro por dentro, base62 por fora.
2//!
3//! Escolhido para id de banco porque cabe num `BIGINT` de 8 bytes e ordena por
4//! tempo: o índice primário cresce pela ponta em vez de fragmentar, o que um
5//! UUID v4 não dá. O que o resto do sistema vê é sempre a string base62.
6//!
7//! O gerador é do `snowflaked`, na variante `sync`: o estado — o último instante
8//! e a sequência dentro dele — mora num `AtomicU64`, e `generate` toma `&self`.
9//! É o que permite este arquivo ser só o que ele deveria ser, um adaptador de
10//! quinze linhas entre uma lib pronta e o contrato do domínio.
11
12use std::sync::Arc;
13use std::time::{Duration, UNIX_EPOCH};
14
15use snowflaked::sync::Generator;
16
17use crate::id::DatabaseIdGenerator;
18
19/// Era do gerador: 2024-01-01T00:00:00Z, a mesma do PHP.
20///
21/// É uma constante de build e não um segredo de runtime — dois deploys com
22/// epochs diferentes emitiriam ids que se sobrepõem no tempo.
23const EPOCH_MS: u64 = 1_704_067_200_000;
24
25/// Quantos servidores cabem num cluster, e a largura do campo de cada um.
26const NODE_COUNT: u16 = 32;
27
28/// O maior valor que cluster e servidor admitem.
29const NODE_MAX: u16 = 31;
30
31/// Gerador Snowflake compartilhado pelo processo.
32///
33/// O `Arc` é o que faz um clone deste gerador continuar sendo **o mesmo**
34/// gerador, e é a exceção de borda que a DI estática admite pela razão mais
35/// literal possível: dois geradores independentes com o mesmo
36/// `cluster_id`/`server_id` emitiriam ids repetidos. Ele não pode existir em
37/// mais de um lugar ao mesmo tempo, então é compartilhado em vez de copiado.
38#[derive(Clone)]
39pub(crate) struct SnowflakeIdGenerator {
40    /// O gerador da lib, compartilhado por todas as threads do processo.
41    inner: Arc<Generator>,
42}
43
44impl SnowflakeIdGenerator {
45    /// Monta o gerador para esta instância de deploy.
46    ///
47    /// `cluster_id` e `server_id` são o que distingue dois processos emitindo ao
48    /// mesmo tempo; vêm dos segredos, não do build.
49    pub(crate) fn new(cluster_id: i32, server_id: i32) -> Self {
50        let epoch = UNIX_EPOCH
51            .checked_add(Duration::from_millis(EPOCH_MS))
52            .unwrap_or(UNIX_EPOCH);
53
54        Self {
55            inner: Arc::new(
56                Generator::builder()
57                    .instance(instance_of(cluster_id, server_id))
58                    .epoch(epoch)
59                    .build(),
60            ),
61        }
62    }
63}
64
65impl DatabaseIdGenerator for SnowflakeIdGenerator {
66    fn next(&self) -> String {
67        base62::encode(self.inner.generate::<u64>())
68    }
69}
70
71/// Junta cluster e servidor nos dez bits de `instance` do Snowflake.
72///
73/// Cada metade vai de 0 a 31, o que ocupa cinco bits; juntas cabem exatamente
74/// nos dez que o algoritmo reserva para identificar quem emitiu. Valor fora da
75/// faixa é preso na borda: o construtor do `snowflaked` entra em pânico acima do
76/// máximo, e derrubar o boot por um segredo mal preenchido troca um problema de
77/// configuração por um processo que não sobe.
78fn instance_of(cluster_id: i32, server_id: i32) -> u16 {
79    let cluster = u16::try_from(cluster_id).unwrap_or_default().min(NODE_MAX);
80    let server = u16::try_from(server_id).unwrap_or_default().min(NODE_MAX);
81
82    cluster.saturating_mul(NODE_COUNT).saturating_add(server)
83}
