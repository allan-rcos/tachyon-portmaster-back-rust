1//! O id opaco, pelo `NanoID`.
2
3use crate::id::RandomIdGenerator;
4
5/// Comprimento do id aleatório.
6///
7/// 21 caracteres do alfabeto URL-safe dão cerca de 126 bits de entropia — longe
8/// de qualquer força bruta viável, e ainda cabendo num cookie sem incomodar.
9const RANDOM_ID_SIZE: usize = 21;
10
11/// Gerador de id opaco, sobre `NanoID`.
12#[derive(Clone, Copy)]
13pub(crate) struct NanoIdGenerator;
14
15impl NanoIdGenerator {
16    /// Monta o gerador.
17    pub(crate) const fn new() -> Self {
18        Self
19    }
20}
21
22impl RandomIdGenerator for NanoIdGenerator {
23    fn next(&self) -> String {
24        nanoid::nanoid!(RANDOM_ID_SIZE)
25    }
26}
27
28#[cfg(test)]
29mod tests {
30    use super::*;
31    use std::collections::HashSet;
32
33    #[test]
34    fn o_token_aleatorio_nao_se_repete() {
35        let generator = NanoIdGenerator::new();
36        let ids: HashSet<String> = (0..1_000).map(|_| generator.next()).collect();
37
38        assert_eq!(ids.len(), 1_000, "houve colisão em 1000 tokens");
39    }
40
41    #[test]
42    fn o_token_aleatorio_tem_a_entropia_esperada() {
43        let generator = NanoIdGenerator::new();
44        assert_eq!(generator.next().chars().count(), RANDOM_ID_SIZE);
45    }
46}
