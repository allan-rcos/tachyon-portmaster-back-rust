3/// Falha ao decodificar uma string base62.
4#[derive(Debug, thiserror::Error, PartialEq, Eq)]
5pub enum Base62Error {
6    /// String vazia não representa número nenhum.
7    #[error("base62 não decodifica string vazia")]
8    Empty,
9
10    /// Caractere fora do alfabeto.
11    #[error("caractere base62 inválido: {0:?}")]
12    InvalidCharacter(char),
13
14    /// Valor grande demais para caber num id.
15    ///
16    /// Nenhum id emitido por esta aplicação chega aqui — um Snowflake estourar
17    /// `i64` é problema de 2093. Um valor deste tamanho veio da URL, e é tão
18    /// inválido quanto um caractere fora do alfabeto.
19    #[error("valor base62 fora da faixa: {0}")]
20    OutOfRange(String),
21}
