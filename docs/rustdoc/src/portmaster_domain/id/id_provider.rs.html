1//! Quem serve os geradores de id.
2//!
3//! Três sabores com propósitos diferentes, e um só deles guardado. Este arquivo
4//! decide qual — como cada um emite é assunto de `intern/`.
5
6use std::sync::{PoisonError, RwLock};
7
8use crate::config::DomainSecrets;
9use crate::id::intern::nano_id_generator::nano_id_generator;
10use crate::id::intern::xid_generator::xid_generator;
11use crate::id::{DatabaseIdGenerator, RandomIdGenerator, SequentialIdGenerator};
12
13#[cfg(feature = "id-snowflake")]
14use crate::id::intern::snowflake_id_generator::{snowflake_id_generator, SnowflakeIdGenerator};
15
16/// O gerador de identidade de entidade do processo.
17///
18/// Guardado porque precisa ser um só: dois geradores com a mesma identidade
19/// emitem o mesmo id, e o porquê aritmético está em [`snowflake_id_generator`].
20///
21/// `RwLock` porque quem o constrói é configuração, e configuração se troca.
22#[cfg(feature = "id-snowflake")]
23static DATABASE: RwLock<Option<SnowflakeIdGenerator>> = RwLock::new(None);
24
25/// Os geradores de id, um por sabor.
26///
27/// Só o de identidade de entidade é guardado. Os outros dois não têm estado a
28/// compartilhar, então duas chamadas devolvem coisas indistinguíveis.
29pub(crate) struct IdProvider;
30
31impl IdProvider {
32    /// Fixa a identidade de deploy, e larga o gerador que estava em uso.
33    ///
34    /// Instalar de novo troca. Fazê-lo com o processo no ar é operação
35    /// consciente: a identidade nova precisa ser de fato nova, senão o processo
36    /// passa a emitir ids que se sobrepõem aos de quem já usa aquele par. O
37    /// lugar normal de chamar isto é o boot.
38    pub(crate) fn install(secrets: DomainSecrets) {
39        #[cfg(feature = "id-snowflake")]
40        Self::replace(snowflake_id_generator(
41            secrets.cluster_id,
42            secrets.server_id,
43        ));
44    }
45
46    /// O gerador de identidade de entidade do processo.
47    ///
48    /// Sem identidade instalada, vale a instância zero de
49    /// [`DomainSecrets::default`] — a mesma que o ambiente sem
50    /// `APP_CLUSTER_ID`/`APP_SERVER_ID` já produziria.
51    ///
52    /// Não é servido pelo [`DomainProvider`](crate::DomainProvider), e é o
53    /// único que não é: quem nomeia uma linha é o `TableModule`, e um gerador
54    /// destes fora do crate permitiria gravar uma sem passar pela regra que a
55    /// valida.
56    ///
57    /// Qual impl atende é **feature de compilação**, resolvida no build.
58    pub(crate) fn database() -> impl DatabaseIdGenerator + Send + Sync + Clone + use<> + 'static {
59        #[cfg(feature = "id-snowflake")]
60        Self::shared()
61    }
62
63    /// O gerador de `request_id`.
64    pub(crate) fn sequential() -> impl SequentialIdGenerator + use<> {
65        xid_generator()
66    }
67
68    /// O gerador de id opaco, para o refresh token.
69    pub(crate) fn random() -> impl RandomIdGenerator + use<> {
70        nano_id_generator()
71    }
72
73    /// Troca o gerador guardado, seja qual for o que estava lá.
74    ///
75    /// Lock envenenado não impede a troca: o que ele protege é um `Option`
76    /// escrito de uma vez, e não uma estrutura que um pânico no meio da escrita
77    /// deixaria pela metade.
78    #[cfg(feature = "id-snowflake")]
79    fn replace(created: SnowflakeIdGenerator) {
80        *DATABASE.write().unwrap_or_else(PoisonError::into_inner) = Some(created);
81    }
82
83    /// O gerador guardado, criando o da instância zero se ainda não houver um.
84    ///
85    /// A leitura vem primeiro porque é o caminho de sempre: só a primeira
86    /// chamada de um processo que não passou pelo boot pede o lock de escrita.
87    #[cfg(feature = "id-snowflake")]
88    fn shared() -> SnowflakeIdGenerator {
89        let installed = DATABASE
90            .read()
91            .unwrap_or_else(PoisonError::into_inner)
92            .clone();
93
94        if let Some(existing) = installed {
95            return existing;
96        }
97
98        let secrets = DomainSecrets::default();
99
100        DATABASE
101            .write()
102            .unwrap_or_else(PoisonError::into_inner)
103            .get_or_insert_with(|| snowflake_id_generator(secrets.cluster_id, secrets.server_id))
104            .clone()
105    }
106}
